function(create_scratch source_files)

    list(LENGTH source_files number_sources)
    if(number_sources EQUAL 0)
        return()
    endif()

    set(valid_sources "")

    foreach(source_file ${source_files})
        file(READ ${source_file} content)

        if(content MATCHES "int[ \t]*main[ \t]*\\(")
            list(APPEND valid_sources ${source_file})
        endif()
    endforeach()

    list(LENGTH valid_sources len)

    if(len EQUAL 0)
        return()
    endif()

    if(len GREATER 1)
        message(FATAL_ERROR "Multiple main() functions detected: ${valid_sources}")
    endif()

    list(GET valid_sources 0 main_file)

    get_filename_component(name ${main_file} NAME_WLE)

    add_executable(${target_prefix}${name}
        ${source_files}
    )

    target_include_directories(${target_prefix}${name} PRIVATE
        ${CMAKE_SOURCE_DIR}/scratch
        ${CMAKE_SOURCE_DIR}/scratch/vanetza_integration
        /usr/local/include/vanetza
    )

    target_link_libraries(${target_prefix}${name}
        ${ns3-libs}
        ${ns3-contrib-libs}
        ${VANETZA_LIBS}
        OpenSSL::SSL
        OpenSSL::Crypto
        $<TARGET_OBJECTS:vanetza_cam_encoder_obj>
    )

endfunction()

#pragma once

#include <memory>
#include <vector>

#include <ns3/packet.h>
#include <ns3/ptr.h>

namespace ns3
{

class VehicleState;
class VanetzaGeoNetAdapter;

struct PacketContext;

/**
 * @brief Runtime integration layer between ns-3 and the Vanetza wrapper.
 *
 * VanetzaRuntime is an orchestration layer.
 *
 * It owns:
 *
 *   - the authoritative VehicleState reference
 *   - PacketContext / RX bridge coordination
 *   - the VanetzaGeoNetAdapter
 *
 * All direct Vanetza protocol/runtime objects are owned by
 * VanetzaGeoNetAdapter.
 */
class VanetzaRuntime
{
public:

    explicit VanetzaRuntime(
        VehicleState& vehicle);

    ~VanetzaRuntime();

    VanetzaRuntime(
        const VanetzaRuntime&) = delete;

    VanetzaRuntime& operator=(
        const VanetzaRuntime&) = delete;

    VanetzaRuntime(
        VanetzaRuntime&&) = delete;

    VanetzaRuntime& operator=(
        VanetzaRuntime&&) = delete;

    bool Initialize();

    void Shutdown();

    bool IsInitialized() const;

    VehicleState& Vehicle();

    void ReceivePacket(
        Ptr<Packet> packet,
        const Address& from);

    bool SendBtpB(
        PacketContext& context,
        uint16_t destinationPort,
        const std::vector<uint8_t>& payload);

private:

    VehicleState& m_vehicle;

    bool m_initialized = false;

    std::unique_ptr<VanetzaGeoNetAdapter>
        m_geonet;
};

} // namespace ns3

//=============================================================================
// Hex dump helper
//=============================================================================

static std::string
HexDump(const std::vector<uint8_t>& data, std::size_t maxBytes = 48)
{
    std::ostringstream oss;

    oss << std::hex
        << std::setfill('0');

    const std::size_t limit =
        std::min(maxBytes, data.size());

    for (std::size_t i = 0; i < limit; ++i)
    {
        oss << std::setw(2)
            << static_cast<unsigned>(data[i]);

        if (i % 2 == 1)
        {
            oss << ' ';
        }
    }

    if (limit < data.size())
    {
        oss << "...";
    }

    return oss.str();
}


//=============================================================================
// RX GeoNetworking packet classification
//=============================================================================

static const char*
ClassifyRxGeoNetworking(const std::vector<uint8_t>& buffer)
{
    if (buffer.size() < 1)
    {
        return "EMPTY";
    }

    const uint8_t basicHeader = buffer[0];

    const uint8_t version =
        (basicHeader >> 4) & 0x0F;

    const uint8_t nextHeader =
        basicHeader & 0x0F;

    if (version != 1)
    {
        return "INVALID_GN_VERSION";
    }

    // Basic Header -> Common Header
    if (nextHeader == 1)
    {
        constexpr std::size_t commonHeaderOffset = 4;
        constexpr std::size_t commonHeaderSize = 8;

        if (buffer.size() <
            commonHeaderOffset + commonHeaderSize)
        {
            return "TRUNCATED_COMMON_HEADER";
        }

        const uint8_t headerType =
            buffer[commonHeaderOffset + 1];

        switch (headerType)
        {
            case 0x10:
                return "BEACON";

            case 0x20:
                return "GEO_UNICAST";

            case 0x30:
                return "GEO_ANYCAST_CIRCLE";

            case 0x31:
                return "GEO_ANYCAST_RECT";

            case 0x32:
                return "GEO_ANYCAST_ELLIPSE";

            case 0x40:
                return "GEO_BROADCAST_CIRCLE";

            case 0x41:
                return "GEO_BROADCAST_RECT";

            case 0x42:
                return "GEO_BROADCAST_ELLIPSE";

            case 0x50:
                return "TSB_SINGLE_HOP";

            case 0x51:
                return "TSB_MULTI_HOP";

            case 0x60:
                return "LS_REQUEST";

            case 0x61:
                return "LS_REPLY";

            default:
                return "UNKNOWN_COMMON_HEADER";
        }
    }

    if (nextHeader == 2)
    {
        return "SECURED_PACKET";
    }

    return "UNKNOWN_BASIC_NEXT_HEADER";
}

//=============================================================================
// Initialize
//=============================================================================

bool
VanetzaRuntime::Initialize()
{
    if (m_initialized)
    {
        return true;
    }

    //=========================================================================
    // Identity preconditions
    //
    // VehicleIdentity is authoritative and must already be initialized
    // by VehicleIdentityInitializer before this function is called.
    //=========================================================================

    if (!m_vehicle.node)
    {
        NS_LOG_UNCOND(
            "[VRT_INIT] ERROR: "
            "VanetzaRuntime requires a valid ns-3 Node");

        return false;
    }

    if (m_vehicle.identity.vehId.empty())
    {
        NS_LOG_UNCOND(
            "[VRT_INIT] ERROR: "
            "VanetzaRuntime requires a valid vehicle ID");

        return false;
    }

    if (m_vehicle.identity.stationId == 0)
    {
        NS_LOG_UNCOND(
            "[VRT_INIT] ERROR: "
            "VanetzaRuntime requires a valid stationId");

        return false;
    }

    if (m_vehicle.identity.nodeId !=
        m_vehicle.node->GetId())
    {
        NS_LOG_UNCOND(
            "[VRT_INIT] ERROR: nodeId mismatch: "
            "identity="
            << m_vehicle.identity.nodeId
            << " node="
            << m_vehicle.node->GetId());

        return false;
    }

    //=========================================================================
    // Initialization logging
    //=========================================================================

    NS_LOG_UNCOND(
        "[VRT_INIT] ===========================================");

    NS_LOG_UNCOND(
        "[VRT_INIT] Initializing VanetzaRuntime");

    NS_LOG_UNCOND(
        "[VRT_INIT] Vehicle    : "
        << m_vehicle.identity.vehId);

    NS_LOG_UNCOND(
        "[VRT_INIT] Node       : "
        << m_vehicle.identity.nodeId);

    NS_LOG_UNCOND(
        "[VRT_INIT] Station ID : "
        << m_vehicle.identity.stationId);

    //=========================================================================
    // Vanetza GeoNetworking boundary
    //=========================================================================
    //
    // All direct Vanetza runtime/protocol objects are owned by
    // VanetzaGeoNetAdapter.
    //
    // VanetzaRuntime remains responsible only for orchestration and for
    // connecting the neutral BTP RX result to PacketContext / RxPipeline.
    //=========================================================================

    m_geonet =
        std::make_unique<VanetzaGeoNetAdapter>(
            m_vehicle);

    if (!m_geonet->Initialize())
    {
        NS_LOG_UNCOND(
            "[VRT_INIT] ERROR: VanetzaGeoNetAdapter initialization failed");

        m_geonet.reset();

        return false;
    }

    NS_LOG_UNCOND(
        "[VRT_INIT] VanetzaGeoNetAdapter initialized");

    //=========================================================================
    // BTP RX boundary
    //=========================================================================
    //
    // The callback is deliberately neutral: no Vanetza type crosses into
    // this runtime layer.
    //=========================================================================

    m_geonet->SetBtpRxHandler(
        2001,
        [this](const VanetzaGeoNetAdapter::BtpRxData& rx)
        {
            NS_LOG_UNCOND(
                "[BTP_HANDLER] indicate() translated "
                "to application RxData"
                << " port="
                << rx.destinationPort
                << " payload="
                << rx.payload.size());

            PacketContext* ctx =
                RxBridgeContext::Current();

            if (!ctx)
            {
                NS_LOG_UNCOND(
                    "[BTP_HANDLER] ERROR: "
                    "RxBridgeContext is null");

                return;
            }

            //=================================================================
            // BTP information
            //=================================================================

            ctx->btpDestinationPort =
                rx.destinationPort;

            ctx->reason =
                "BTP dst_port=" +
                std::to_string(
                    rx.destinationPort);

            //=================================================================
            // GeoNetworking source position
            //=================================================================

            ctx->geonet.hasSourcePosition =
                rx.hasSourcePosition;

            if (rx.hasSourcePosition)
            {
                ctx->geonet.latitude =
                    rx.latitude;

                ctx->geonet.longitude =
                    rx.longitude;
            }

            //=================================================================
            // GeoNetworking transport
            //=================================================================

            ctx->geonet.upperProtocol =
                "BTP_B";

            ctx->geonet.transportType =
                "SHB";

            //=================================================================
            // Traffic class
            //=================================================================

            ctx->geonet.hasTrafficClass =
                rx.hasTrafficClass;

            if (rx.hasTrafficClass)
            {
                ctx->geonet.trafficClass =
                    rx.trafficClass;
            }

            //=================================================================
            // ITS-AID
            //=================================================================

            ctx->geonet.hasItsAid =
                rx.hasItsAid;

            if (rx.hasItsAid)
            {
                ctx->geonet.itsAid =
                    rx.itsAid;
            }

            //=================================================================
            // Permissions
            //=================================================================

            ctx->geonet.hasPermissions =
                rx.hasPermissions;

            if (rx.hasPermissions)
            {
                ctx->geonet.permissionsSize =
                    rx.permissionsSize;
            }

            //=================================================================
            // Lifetime
            //=================================================================

            ctx->geonet.hasLifetime =
                rx.hasLifetime;

            if (rx.hasLifetime)
            {
                ctx->geonet.lifetimeMs =
                    rx.lifetimeMs;
            }

            //=================================================================
            // GeoNetworking sender MID
            //=================================================================

            ctx->geonet.hasSenderMid =
                rx.hasSourceMid;

            if (rx.hasSourceMid)
            {
                ctx->geonet.senderMid =
                    rx.sourceMid;
            }

            //=================================================================
            // Preserve the existing single PacketContext.
            //=================================================================

            RxPipeline::ProcessDirect(
                *ctx,
                rx.payload);
        });

    //=========================================================================
    // Final initialization state
    //=========================================================================

    NS_LOG_UNCOND(
        "[VRT_INIT] GeoNet adapter : "
        << (m_geonet ? "OK" : "NULL"));

    NS_LOG_UNCOND(
        "[VRT_INIT] Identity authority: "
        "VehicleIdentity");

    NS_LOG_UNCOND(
        "[VRT_INIT] GN_ADDR authority: "
        "VanetzaRuntime");

    NS_LOG_UNCOND(
        "[VRT_INIT] VehicleRegistry role: "
        "reverse indexes only");

    NS_LOG_UNCOND(
        "[VRT_INIT] VanetzaRuntime initialization completed");

    NS_LOG_UNCOND(
        "[VRT_INIT] ===========================================");

    m_initialized = true;

    return true;
}


//=============================================================================
// Shutdown
//=============================================================================

void
VanetzaRuntime::Shutdown()
{
    if (!m_initialized)
    {
        return;
    }

    m_geonet.reset();

    m_initialized = false;
}


//=============================================================================

//=============================================================================

bool
VanetzaRuntime::IsInitialized() const
{
    return m_initialized;
}


VehicleState&
VanetzaRuntime::Vehicle()
{
    return m_vehicle;
}


//=============================================================================
// BTP TX
//=============================================================================

bool
VanetzaRuntime::SendBtpB(
    PacketContext& context,
    uint16_t destinationPort,
    const std::vector<uint8_t>& payload)
{
    if (!m_initialized || !m_geonet)
    {
        NS_LOG_UNCOND(
            "[VRT_TX] ERROR: GeoNetworking adapter not initialized");

        return false;
    }

    return m_geonet->SendBtpB(
        context,
        destinationPort,
        payload);
}


//=============================================================================
// Receive path
//=============================================================================

void
VanetzaRuntime::ReceivePacket(
    Ptr<Packet> packet,
    const Address& from)
{
    //-------------------------------------------------------------------------
    // Runtime validation
    //-------------------------------------------------------------------------

    if (!m_initialized || !m_geonet)
    {
        NS_LOG_UNCOND(
            "[VRT_RX] Dropped: runtime not initialized");

        return;
    }

    if (!packet || packet->GetSize() == 0)
    {
        NS_LOG_UNCOND(
            "[VRT_RX] Dropped: null or empty packet");

        return;
    }

    //-------------------------------------------------------------------------
    // Resolve UDP source
    //-------------------------------------------------------------------------

    NS_ABORT_MSG_IF(
        !InetSocketAddress::IsMatchingType(from),
        "[VRT_RX] Unsupported source address type");

    const InetSocketAddress source =
        InetSocketAddress::ConvertFrom(from);

    const Ipv4Address sourceIp =
        source.GetIpv4();

    VehicleState* sender =
        VehicleRegistry::Instance().GetByIpv4(sourceIp);

    NS_ABORT_MSG_IF(
        sender == nullptr,
        "[VRT_RX] Sender not found for source IPv4="
            << sourceIp);

    NS_ABORT_MSG_IF(
        !sender->linkMacInitialized,
        "[VRT_RX] Sender Link MAC not initialized: "
            << sender->identity.vehId);

    //-------------------------------------------------------------------------
    // Receive information
    //-------------------------------------------------------------------------

    const uint32_t size =
        packet->GetSize();

    NS_LOG_UNCOND(
        "[VRT_RX] ReceivePacket called, size="
        << size
        << " receiver_veh="
        << m_vehicle.identity.vehId
        << " receiver_node="
        << m_vehicle.identity.nodeId
        << " source_ip="
        << sourceIp
        << " sender_veh="
        << sender->identity.vehId
        << " sender_node="
        << sender->identity.nodeId
        << " sender_station="
        << sender->identity.stationId
        << " sender_link_mac="
        << sender->linkMac);

    //-------------------------------------------------------------------------
    // Extract complete received frame
    //-------------------------------------------------------------------------

    std::vector<uint8_t> buffer(size);

    packet->CopyData(
        buffer.data(),
        buffer.size());

    NS_LOG_UNCOND(
        "[VRT_RX] First bytes: "
        << HexDump(buffer, 48));

    const char* rxClassification =
        ClassifyRxGeoNetworking(buffer);

    NS_LOG_UNCOND(
        "[VRT_RX] GN classification="
        << rxClassification
        << " size="
        << buffer.size());

    if (std::string(rxClassification) == "BEACON")
    {
        NS_LOG_UNCOND(
            "[VRT_RX] GN auxiliary BEACON detected; "
            "no BTP/CAM payload expected");
    }

    if (!buffer.empty() &&
        (buffer[0] & 0xF0) != 0x10)
    {
        NS_LOG_UNCOND(
            "[VRT_RX] WARNING: First nibble is 0x"
            << std::hex
            << static_cast<int>(buffer[0] >> 4)
            << ", expected 0x1 for GeoNet version 1");
    }

    //-------------------------------------------------------------------------
    // Build the SINGLE RX PacketContext.
    //
    // This context is created at the receive boundary and is preserved
    // through Router -> GeoNetworking -> BTP adapter -> RxPipeline.
    //-------------------------------------------------------------------------

    PacketContext ctx =
        RxContextBuilder::Build(
            0,
            m_vehicle);

    //-------------------------------------------------------------------------
    // Establish sender identity already known at RX boundary.
    //-------------------------------------------------------------------------

    ctx.senderVehId =
        sender->identity.vehId;

    ctx.senderNodeId =
        sender->identity.nodeId;

    ctx.senderStationId =
        sender->identity.stationId;

    //-------------------------------------------------------------------------
    // Sender Link MAC.
    //
    // This is the ns-3 Link MAC and is independent from GN_ADDR.MID.
    //-------------------------------------------------------------------------

    ctx.geonet.senderLinkMac =
        sender->linkMac;

    ctx.geonet.hasSenderLinkMac =
        sender->linkMacInitialized;

    ctx.receiverVehId =
        m_vehicle.identity.vehId;

    ctx.receiverNodeId =
        m_vehicle.identity.nodeId;

    ctx.receiverStationId =
        m_vehicle.identity.stationId;

    //-------------------------------------------------------------------------
    // Allocate RX-local observability identifiers.
    //-------------------------------------------------------------------------

    auto& packetIds =
        PacketIdManager::Instance();

    ctx.packetId =
        packetIds.NextPacketId();

    ctx.vehiclePacketId =
        packetIds.NextVehiclePacketId(
            ctx.senderVehId);

    ctx.sequenceId =
        packetIds.NextSequenceId(
            ctx.senderVehId,
            PayloadType::CAM);

    //-------------------------------------------------------------------------
    // 1/9 RAW_RX
    //-------------------------------------------------------------------------

    ctx.AdvancePipeline(
        PipelineStage::RAW_RX);

    ctx.payload.type =
        PayloadType::RAW;

    ctx.payload.bytes =
        buffer;

    ctx.payload.signatureSize =
        0;

    ctx.reason =
        "RX_FRAME_RECEIVED";

    V2XWriteLog(
        ctx,
        V2XLogLevel::TRACE,
        "RX",
        "RAW_RX",
        "size=" +
            std::to_string(buffer.size()));

    //-------------------------------------------------------------------------
    // 2/9 ACCESS_RX
    //-------------------------------------------------------------------------

    ctx.AdvancePipeline(
        PipelineStage::ACCESS_RX);

    ctx.reason =
        "ACCESS_FRAME_RECEIVED";

    V2XWriteLog(
        ctx,
        V2XLogLevel::DEBUG,
        "ACCESS",
        "ACCESS_RX_BEGIN",
        "received_frame_bytes=" +
            std::to_string(buffer.size()));

    ctx.MarkSuccess();

    V2XWriteLog(
        ctx,
        V2XLogLevel::DEBUG,
        "ACCESS",
        "ACCESS_RX_OK",
        "bytes=" +
            std::to_string(buffer.size()));

    //-------------------------------------------------------------------------
    // GeoNetworking / Vanetza boundary.
    //
    // PacketContext has already passed through RAW_RX and ACCESS_RX.
    // The adapter owns the Vanetza Router and preserves the existing
    // synchronous Router -> BTP -> RxPipeline path.
    //-------------------------------------------------------------------------

    if (!m_geonet)
    {
        NS_LOG_UNCOND(
            "[VRT_RX] ERROR: GeoNetworking adapter not initialized");

        return;
    }

    if (!m_geonet->Receive(
            ctx,
            buffer,
            sender->linkMac))
    {
        NS_LOG_UNCOND(
            "[VRT_RX] GeoNetworking adapter rejected packet");
    }
}

//=============================================================================
// SIGN
//=============================================================================

std::vector<uint8_t> CamSecurity::Sign(const ns3::PacketContext& ctx, const std::vector<uint8_t>& payload, EVP_PKEY* privateKey) {
    
    ns3::PacketContext logCtx = ctx;

    logCtx.payload.type = ns3::PayloadType::CAM;
    logCtx.payload.bytes = payload;
    logCtx.payload.signatureSize = 0;

    LogBegin(logCtx, "SIGN_BEGIN");

    const auto start = V2XTimeNow();

    if (!privateKey) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "SIGN_FAIL", "missing_private_key", "private key is null");

        return {};
    }

    if (payload.empty()) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::WARN, "SIGN_FAIL", "empty_payload", "payload is empty");

        return {};
    }

    MdContext mdCtx(EVP_MD_CTX_new());

    if (!mdCtx) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "SIGN_FAIL", "md_context_creation_failed", "EVP_MD_CTX_new failed");

        return {};
    }

    std::vector<uint8_t> signature(EVP_PKEY_size(privateKey));
    size_t signatureSize = signature.size();

    const bool ok =
        EVP_DigestSignInit(mdCtx.get(), nullptr, EVP_sha256(), nullptr, privateKey) == 1 &&
        EVP_DigestSignUpdate(mdCtx.get(), payload.data(), payload.size()) == 1 &&
        EVP_DigestSignFinal(mdCtx.get(), signature.data(), &signatureSize) == 1;

    logCtx.durationUs = V2XElapsedMicros(start);

    if (!ok) {

        LogFailure(logCtx, V2XLogLevel::ERROR, "SIGN_FAIL", "digest_sign_failed", "DigestSign failed");

        return {};
    }

    signature.resize(signatureSize);

    logCtx.payload.signatureSize = static_cast<uint32_t>(signatureSize);

    LogSuccess(logCtx, "SIGN_OK", "signature_size=" + std::to_string(signatureSize));

    return signature;
}

//=============================================================================
// VERIFY
//=============================================================================

bool CamSecurity::Verify(const ns3::PacketContext& ctx, const std::vector<uint8_t>& payload, const std::vector<uint8_t>& signature, X509* certificate, const VehicleState&) {
    
    ns3::PacketContext logCtx = ctx;

    logCtx.payload.type = ns3::PayloadType::CAM;
    logCtx.payload.bytes = payload;
    logCtx.payload.signatureSize = static_cast<uint32_t>(signature.size());

    const auto start = V2XTimeNow();

    if (!certificate) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "missing_certificate", "certificate is null");

        return false;
    }

    if (payload.empty()) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "empty_payload", "payload is empty");

        return false;
    }

    if (signature.empty()) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "empty_signature", "signature is empty");

        return false;
    }

    PublicKey publicKey(X509_get_pubkey(certificate));

    if (!publicKey) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "public_key_not_found", "public key unavailable");

        return false;
    }

    MdContext mdCtx(EVP_MD_CTX_new());

    if (!mdCtx) {

        logCtx.durationUs = V2XElapsedMicros(start);

        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "digest_context_creation_failed", "EVP_MD_CTX_new failed");

        return false;
    }

    const bool ok =
        EVP_DigestVerifyInit(mdCtx.get(), nullptr, EVP_sha256(), nullptr, publicKey.get()) == 1 &&
        EVP_DigestVerifyUpdate(mdCtx.get(), payload.data(), payload.size()) == 1 &&
        EVP_DigestVerifyFinal(mdCtx.get(), signature.data(), signature.size()) == 1;

    logCtx.durationUs = V2XElapsedMicros(start);

    if (!ok) {
        LogFailure(logCtx, V2XLogLevel::ERROR, "VERIFY_FAIL", "signature_verification_failed", "signature verification failed");
        return false;
    }

    return true;   // <-- mantém só isto, sem o LogSuccess antes

}

#pragma GCC diagnostic ignored "-Wdeprecated-declarations"

#include <openssl/evp.h>
#include <openssl/x509.h>
#include <openssl/x509v3.h>

namespace ns3 {

std::pair<EVP_PKEY*, X509*> GenerateEcdsaKeyAndCert(const std::string& vehId){
    
    EVP_PKEY* pkey = EVP_PKEY_new();

    EC_KEY* eckey = EC_KEY_new_by_curve_name(NID_X9_62_prime256v1);

    EC_KEY_generate_key(eckey);

    EVP_PKEY_assign_EC_KEY(pkey, eckey);

    X509* cert = X509_new();

    ASN1_INTEGER_set(X509_get_serialNumber(cert),std::rand());

    X509_gmtime_adj(X509_get_notBefore(cert),0);

    X509_gmtime_adj(X509_get_notAfter(cert), 365 * 24 * 3600);

    X509_set_pubkey(cert, pkey);

    X509_NAME* name = X509_get_subject_name(cert);

    X509_NAME_add_entry_by_txt(name, "CN", MBSTRING_ASC, (unsigned char*)vehId.c_str(), -1, -1, 0);

    X509_set_issuer_name(cert, name);

    X509_sign(cert, pkey, EVP_sha256());

    return {pkey, cert};
}

        << "]"

        << "[EV="
        << event
        << "]";

        oss << " " << message;

}

}

// ============================================================================
// Populate sender context
// ============================================================================

bool RxContext::PopulateSenderContext(
    ns3::PacketContext& ctx,
    const ns3::VehicleState& sender)
{
    const bool hasSenderContext =
        !ctx.senderVehId.empty() ||
        ctx.senderNodeId != 0 ||
        ctx.senderStationId != 0;

    if (hasSenderContext)
    {
        const bool matches =
            ctx.senderVehId == sender.identity.vehId &&
            ctx.senderNodeId == sender.identity.nodeId &&
            ctx.senderStationId == sender.identity.stationId;

        if (!matches)
        {
            ctx.MarkFailure();
            ctx.reason = "sender_context_mismatch";

            V2XWriteLog(
                ctx,
                V2XLogLevel::ERROR,
                "RX",
                "PACKET_CONTEXT_INTEGRITY_FAIL",
                "sender context does not match resolved sender");

            return false;
        }

        return true;
    }

    ctx.senderVehId = sender.identity.vehId;
    ctx.senderNodeId = sender.identity.nodeId;
    ctx.senderStationId = sender.identity.stationId;

    return true;
}

// ============================================================================
// Populate receiver context
// ============================================================================

bool RxContext::PopulateReceiverContext(
    ns3::PacketContext& ctx,
    const ns3::VehicleState& receiver)
{
    // receiverNodeId is a resolution key supplied by RxContextBuilder.
    // It does not by itself mean that the authoritative receiver identity
    // has already been established in PacketContext.
    const bool hasReceiverContext =
        !ctx.receiverVehId.empty() ||
        ctx.receiverStationId != 0;

    if (hasReceiverContext)
    {
        const bool matches =
            ctx.receiverVehId == receiver.identity.vehId &&
            ctx.receiverNodeId == receiver.identity.nodeId &&
            ctx.receiverStationId == receiver.identity.stationId;

        if (!matches)
        {
            ctx.MarkFailure();
            ctx.reason = "receiver_context_mismatch";

            V2XWriteLog(
                ctx,
                V2XLogLevel::ERROR,
                "RX",
                "PACKET_CONTEXT_INTEGRITY_FAIL",
                "receiver context does not match resolved receiver");

            return false;
        }

        return true;
    }

    ctx.receiverVehId = receiver.identity.vehId;
    ctx.receiverNodeId = receiver.identity.nodeId;
    ctx.receiverStationId = receiver.identity.stationId;

    return true;
}

#include "rx_unpack.h"

#include "vanetza_integration/wrappers/vanetza/cam/uper/cam_uper_decoder.h"
#include "vanetza_integration/core/v2x_log.h"

bool
RxUnpack::ExtractRaw(
    ns3::PacketContext& ctx,
    const std::vector<uint8_t>& buffer,
    CamPayloadParts& frame)
{
    if (buffer.empty())
    {
        ctx.MarkFailure();
        ctx.reason = "empty_buffer";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            "empty buffer");

        return false;
    }

    //----------------------------------------------------------------------
    // Decode only the CAM prefix.
    //
    // uper_decode() returns consumed bits even when additional bytes follow
    // the CAM. This gives us the protocol-native CAM boundary without any
    // proprietary length/header field.
    //----------------------------------------------------------------------
    CamUperDecodeResult decodeResult =
        CamUperDecoder::Decode(buffer);

    if (!decodeResult.success || !decodeResult.cam)
    {
        ctx.MarkFailure();
        ctx.reason = "cam_decode_failed";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            decodeResult.error);

        return false;
    }

    if (decodeResult.consumedBits == 0)
    {
        ctx.MarkFailure();
        ctx.reason = "invalid_consumed_bits";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            "decoder consumed zero bits");

        return false;
    }

    const std::size_t camSize =
        (decodeResult.consumedBits + 7) / 8;

    if (camSize > buffer.size())
    {
        ctx.MarkFailure();
        ctx.reason = "cam_size_exceeds_buffer";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            "decoded CAM size exceeds buffer");

        return false;
    }

    //----------------------------------------------------------------------
    // Application wire format:
    //
    //   [CAM ASN.1 bytes][signature bytes]
    //
    // No proprietary header is present.
    //----------------------------------------------------------------------
    frame.payload.assign(
        buffer.begin(),
        buffer.begin() + camSize);

    frame.signature.assign(
        buffer.begin() + camSize,
        buffer.end());

    if (frame.payload.empty())
    {
        ctx.MarkFailure();
        ctx.reason = "empty_cam_payload";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            "CAM payload is empty");

        return false;
    }

    if (frame.signature.empty())
    {
        ctx.MarkFailure();
        ctx.reason = "empty_signature";

        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "EXTRACT_FAIL",
            "signature is empty");

        return false;
    }

    ctx.payload.type = ns3::PayloadType::CAM;
    ctx.payload.bytes = buffer;
    ctx.payload.signatureSize =
        static_cast<uint32_t>(frame.signature.size());

    return true;
}

            CamSender::SendRaw(packetContext, *request.vehicle, rawPayload, request.reason);

CamTransmissionRequest CamTransmission::PrepareTransmission(VehicleState& vehicle, double now) {
    
    CamTransmissionRequest req;

    req.vehicle = &vehicle;

    const double elapsed = (vehicle.lastCamTime < 0.0) ? 999.0 : (now - vehicle.lastCamTime);

    if (elapsed < 0.099)
        return req;

    req.firstCam = (vehicle.lastCamTime < 0.0);

    req.reason = "FIRST_CAM";

    if (!req.firstCam){

        if (!ShouldTransmit(vehicle, vehicle.lastCamSnapshot, now, req.reason)) {
            return req;
        }
    }

    ++vehicle.camSequenceCounter;
    vehicle.currentTxSeqId = vehicle.camSequenceCounter;

    req.shouldSend = true;

    return req;
}


bool CamTransmission::ShouldTransmit(const VehicleState& current, const CamTriggerSnapshot& previous, double now, std::string& reason) {
    
    //const uint64_t step = SimulationContext::simulationStep;

    const double dx = current.mobility.position.x - previous.position.x;
    const double dy = current.mobility.position.y - previous.position.y;
    const double dist = std::sqrt(dx * dx + dy * dy);

    if (dist > 4.0){

        //V2XWriteLog(step,V2XLogLevel::DEBUG,"CAM","TRIGGER_POS",current.identity.vehId,current.identity.nodeId,current.identity.stationId,current.currentTxSeqId,"dist=" + std::to_string(dist));

        reason = "DISTANCE";
        return true;
    }

    const double dSpeed = std::abs(current.mobility.speed - previous.speed);

    if (dSpeed > 0.5){

        //V2XWriteLog(step,V2XLogLevel::DEBUG, "CAM", "TRIGGER_SPEED", current.identity.vehId, current.identity.nodeId, current.identity.stationId, current.currentTxSeqId, "dspeed=" + std::to_string(dSpeed));

        reason = "SPEED";
        return true;
    }

    const double dHeading = std::abs(current.mobility.heading - previous.heading);

    if (dHeading > 4.0){

        //V2XWriteLog(step,V2XLogLevel::DEBUG,"CAM","TRIGGER_HEADING",current.identity.vehId,current.identity.nodeId,current.identity.stationId,current.currentTxSeqId,"heading=" + std::to_string(dHeading));

        reason = "HEADING";
        return true;
    }

    if (previous.lastCamTime < 0.0 || (now - previous.lastCamTime) >= 1.0){
        //V2XWriteLog(step,V2XLogLevel::DEBUG,"CAM","TRIGGER_MAXTIME",current.identity.vehId,current.identity.nodeId,current.identity.stationId,current.currentTxSeqId,"elapsed=" + std::to_string(now - previous.lastCamTime));

        reason = "TIME";
        return true;
    }

    reason.clear();
    return false;
}

void CamTransmission::CommitTransmission(VehicleState& vehicle, double now){
    
    vehicle.lastCamTime = now;
    vehicle.lastCamGenTime = now;

    vehicle.lastCamSnapshot.position = vehicle.mobility.position;
    vehicle.lastCamSnapshot.speed = vehicle.mobility.speed;
    vehicle.lastCamSnapshot.heading = vehicle.mobility.heading;
    vehicle.lastCamSnapshot.lastCamTime = now;
}

#include "vanetza_integration/context/tx_bridge/tx_bridge_context.h"

std::vector<uint8_t> CamService::GenerateRawPayload(ns3::CamGenerationContext& context, const std::string& reason) {
    ns3::PacketContext& ctx = *context.packetContext;

    auto& packetIds = ns3::PacketIdManager::Instance();
    ctx.packetId        = packetIds.NextPacketId();
    ctx.vehiclePacketId = packetIds.NextVehiclePacketId(ctx.senderVehId);
    ctx.sequenceId      = packetIds.NextSequenceId(ctx.senderVehId, ns3::PayloadType::CAM);

    //=========================================================================
    // Step 1/8: GENERATE
    //=========================================================================
    ctx.AdvancePipeline(ns3::PipelineStage::GENERATE);
    ctx.reason = reason;
    ctx.MarkSuccess();
    V2XWriteLog(ctx, V2XLogLevel::DEBUG, "CAM", "GENERATE_OK", "reason=" + reason);

    //=========================================================================
    // Step 2/8: ENCODE
    //=========================================================================
    ctx.AdvancePipeline(ns3::PipelineStage::ENCODE);
    std::vector<uint8_t> payload = CamEncoder::Encode(ctx, context.camInput);
    if (payload.empty()) {
        ctx.MarkFailure();
        ctx.reason = "encode_failed";
        V2XWriteLog(ctx, V2XLogLevel::ERROR, "CAM", "ENCODE_FAIL", "ASN.1 encoding failed");
        return {};
    }
    ctx.payload.type = ns3::PayloadType::CAM;
    ctx.payload.bytes = payload;
    ctx.payload.signatureSize = 0;
    ctx.MarkSuccess();

    //=========================================================================
    // Step 3/8: SIGN
    //=========================================================================
    ctx.AdvancePipeline(ns3::PipelineStage::SIGN);
    std::vector<uint8_t> signature = CamSecurity::Sign(ctx, payload, context.security.privateKey);
    if (signature.empty()) {
        ctx.MarkFailure();
        ctx.reason = "sign_failed";
        V2XWriteLog(ctx, V2XLogLevel::ERROR, "CAM", "SIGN_FAIL", "Signature generation failed");
        return {};
    }
    ctx.payload.signatureSize = static_cast<uint32_t>(signature.size());
    ctx.MarkSuccess();

    //=========================================================================
    // Application payload
    //
    // Fase 9: no proprietary framing.
    //
    // Wire format at BTP Application:
    //   [CAM ASN.1 bytes][signature bytes]
    //
    // packetId / vehiclePacketId / sequenceId remain internal PacketContext
    // metadata and are not serialized onto the wire.
    //=========================================================================
    std::vector<uint8_t> rawPayload;
    rawPayload.reserve(payload.size() + signature.size());

    rawPayload.insert(
        rawPayload.end(),
        payload.begin(),
        payload.end());

    rawPayload.insert(
        rawPayload.end(),
        signature.begin(),
        signature.end());

    if (rawPayload.empty()) {
        ctx.MarkFailure();
        ctx.reason = "raw_payload_empty";
        V2XWriteLog(
            ctx,
            V2XLogLevel::ERROR,
            "CAM",
            "RAW_PAYLOAD_FAIL",
            "CAM payload plus signature is empty");
        return {};
    }

    ctx.payload.type = ns3::PayloadType::CAM;
    ctx.payload.bytes = rawPayload;
    ctx.payload.signatureSize = static_cast<uint32_t>(signature.size());

    //=========================================================================
    // Publicar o PacketContext para a stack de transporte.
    //
    // CORREÇÃO O1: publicar o shared_ptr dono do contexto (não uma
    // referência). O lambda diferido da Ns3AccessAdapter captura este
    // shared_ptr por valor, mantendo o objeto vivo até TX_COMPLETE.
    //=========================================================================
    ns3::TxBridgeContext::Push(context.packetContext);

    // O pipeline permanece no step 3/8 (SIGN). Os steps 4-8
    // (BTP_TX -> GEONET_TX -> ACCESS_TX -> SEND -> TX_COMPLETE)
    // são executados pelo transporte (CamSender::SendRaw).
    return rawPayload;
}

#pragma once

// ======================================================
// CAM POSITION ENCODING
// ======================================================
// A rede SUMO atual não tem georreferenciação real
// (projParameter="!" em simple.net.xml), pelo que as posições
// vindas do SUMO/TraCI são coordenadas cartesianas locais (metros),
// não latitude/longitude WGS84 reais.
//
// Até existir georreferenciação real (fase futura), mapeamos essas
// coordenadas locais diretamente para os campos ASN.1
// referencePosition.latitude/longitude do CAM, usando uma escala fixa
// que mantém os valores dentro do intervalo válido (constrangido) do
// tipo Latitude/Longitude do ETSI ITS CDD (~±900000000 / ±1800000000),
// evitando overflow / encoding inválido.
//
// IMPORTANTE: este valor TEM de ser idêntico no encoder e no decoder.
// Está centralizado aqui exatamente para evitar a dessincronização
// que já causou bugs anteriormente.
//

// confortavelmente dentro do limite. Se o cenário crescer para

#include "cam_encoder.h"

#include "vanetza_integration/wrappers/vanetza/cam/vanetza_cam_codec.h"
#include "vanetza_integration/cam/validation/cam_validator.h"

#include "vanetza_integration/core/v2x_log.h"

#include <utility>

// ============================================================================
// Encode CAM
// ============================================================================

std::vector<uint8_t> CamEncoder::Encode(const ns3::PacketContext& ctx, const CamInput& input) {
    
    ns3::PacketContext logCtx = ctx;

    //=========================================================================
    // Context enrichment
    //=========================================================================
    //
    // O encoder confirma o Station ID do emissor local. A identidade remota
    // não existe durante a transmissão.
    //
    //=========================================================================

    logCtx.senderStationId = input.stationId;

    //=========================================================================
    // Validation
    //=========================================================================

    if (!CamValidator::Validate(input)) {

        logCtx.MarkFailure();
        logCtx.reason = "invalid_position";

        V2XWriteLog(logCtx, V2XLogLevel::DEBUG, "CAM", "ENCODE_FAIL", "Coordinates are too close to origin");

        return {};
    }

    //=========================================================================
    // Encode through Vanetza CAM wrapper
    //=========================================================================

    V2XWriteLog(logCtx, V2XLogLevel::DEBUG, "CAM", "ENCODE_BEGIN");

    const auto start = V2XTimeNow();

    auto result = VanetzaCamCodec::Encode(input);

    logCtx.durationUs = V2XElapsedMicros(start);

    if (!result.success) {

        logCtx.MarkFailure();
        logCtx.reason = result.failedType;

        V2XWriteLog(logCtx, V2XLogLevel::DEBUG, "CAM", "ENCODE_FAIL", "failed_type=" + result.failedType);

        return {};
    }

    //=========================================================================
    // Update context
    //=========================================================================

    logCtx.payload.type = ns3::PayloadType::CAM;
    logCtx.payload.bytes = result.bytes;
    logCtx.payload.signatureSize = 0;

    logCtx.MarkSuccess();

    V2XWriteLog(logCtx, V2XLogLevel::DEBUG, "CAM", "ENCODE_OK", "bytes=" + std::to_string(result.bytes.size()));

    return std::move(result.bytes);
}

CamAsn1Builder::CamPtr
CamAsn1Builder::Build(const CamInput& input)
{
    CamPtr cam(
        static_cast<CAM_t*>(calloc(1, sizeof(CAM_t))));

    if (!cam)
    {
        return nullptr;
    }

    // =========================================================================
    // Header
    // =========================================================================

    cam->header.protocolVersion = 1;
    cam->header.messageID = 2;
    cam->header.stationID = input.stationId;

    // =========================================================================
    // CAM
    // =========================================================================

    cam->cam.generationDeltaTime = input.generationDeltaTime;

    auto& parameters = cam->cam.camParameters;

    // -------------------------------------------------------------------------
    // Basic Container
    // -------------------------------------------------------------------------

    parameters.basicContainer.stationType =
        StationType_passengerCar;

    parameters.basicContainer.referencePosition.latitude =
        static_cast<long>(
            std::llround(
                input.latitude * kCamPositionScale));

    parameters.basicContainer.referencePosition.longitude =
        static_cast<long>(
            std::llround(
                input.longitude * kCamPositionScale));

    // -------------------------------------------------------------------------
    // High Frequency Container
    // -------------------------------------------------------------------------

    parameters.highFrequencyContainer.present =
        HighFrequencyContainer_PR_basicVehicleContainerHighFrequency;

    auto& hf =
        parameters.highFrequencyContainer
            .choice.basicVehicleContainerHighFrequency;

    hf.speed.speedValue =
        static_cast<long>(input.speed * 100);

    hf.speed.speedConfidence =
        SpeedConfidence_unavailable;

    hf.heading.headingValue =
        static_cast<long>(input.heading * 10);

    hf.heading.headingConfidence =
        HeadingConfidence_unavailable;

    hf.driveDirection =
        DriveDirection_forward;

    hf.longitudinalAcceleration.longitudinalAccelerationValue =
        static_cast<long>(input.acceleration * 10);

    hf.longitudinalAcceleration.longitudinalAccelerationConfidence =
        1;

    hf.curvature.curvatureValue = 0;

    hf.curvature.curvatureConfidence =
        CurvatureConfidence_onePerMeter_0_00002;

    hf.yawRate.yawRateValue = 0;
    hf.yawRate.yawRateConfidence = 1;

    hf.vehicleLength.vehicleLengthValue = 50;

    hf.vehicleLength.vehicleLengthConfidenceIndication =
        VehicleLengthConfidenceIndication_noTrailerPresent;

    hf.vehicleWidth = 20;

    return cam;
}

VanetzaCamEncodeResult
VanetzaCamCodec::Encode(const CamInput& input)
{
    VanetzaCamEncodeResult result;

    auto cam = CamAsn1Builder::Build(input);

    if (!cam)
    {
        result.failedType = "asn1_alloc_failed";
        return result;
    }

    const auto encoded = CamUperEncoder::Encode(*cam);

    {
        result.failedType = encoded.failedType;
        return result;
    }

    result.success = true;
    result.bytes = encoded.bytes;

    return result;
}

VanetzaCamDecodeResult
VanetzaCamCodec::Decode(
    const std::vector<uint8_t>& payload,
    uint64_t timestamp)
{
    VanetzaCamDecodeResult result;

    const auto decoded = CamUperDecoder::Decode(payload);

    if (!decoded.success || !decoded.cam)
    {
        result.error = decoded.error.empty()
            ? "ASN.1 decode failed"
            : decoded.error;

        return result;
    }

    result.message =
        CamConverter::ToMessage(*decoded.cam, timestamp);

    result.success = true;

    return result;
}

{
#include "asn_application.h"
#include "asn_internal.h"
}

CamUperDecodeResult
CamUperDecoder::Decode(
    const std::vector<uint8_t>& payload)
{
    CamUperDecodeResult result;

    if (payload.empty())
    {
        result.error = "empty payload";
        return result;
    }

    CAM_t* rawCam = nullptr;

    const asn_dec_rval_t rval =
        uper_decode(
            nullptr,
            &asn_DEF_CAM,
            reinterpret_cast<void**>(&rawCam),
            payload.data(),
            payload.size(),
            0,
            0);

    if (rval.code != RC_OK || rawCam == nullptr)
    {
        result.error = "ASN.1 decode failed";

        if (rawCam)
        {
            ASN_STRUCT_FREE(asn_DEF_CAM, rawCam);
        }

        return result;
    }

    result.success = true;
    result.consumedBits = rval.consumed;
    result.cam.reset(rawCam);

    return result;
}

#include "cam_uper_encoder.h"

#include <array>

extern "C"
{
#include "asn_application.h"
#include "asn_internal.h"
}

namespace
{
    constexpr std::size_t kMaxBufferSize = 2048;
}

CamUperResult
CamUperEncoder::Encode(
    const CAM_t& cam)
{
    CamUperResult result;

    std::array<uint8_t, kMaxBufferSize> buffer;

    const asn_enc_rval_t rval =
        uper_encode_to_buffer(
            &asn_DEF_CAM,
            nullptr,
            &cam,
            buffer.data(),
            buffer.size());

    if (rval.encoded <= 0)
    {
        result.success = false;

        if (rval.failed_type && rval.failed_type->name)
        {
            result.failedType = rval.failed_type->name;
        }
        else
        {
            result.failedType = "unknown";
        }

        return result;
    }

    result.success = true;
    result.encodedBits = rval.encoded;

    const std::size_t encodedSize =
        (rval.encoded + 7) / 8;

    result.bytes.assign(
        buffer.begin(),
        buffer.begin() + encodedSize);

    return result;
}

//=============================================================================
// Internal BTP RX indication handler
//
// This class is intentionally private to VanetzaBtpAdapter.
//
// It is the Vanetza-facing part of the BTP RX boundary.
// No Vanetza BTP type is exposed to the application.
//=============================================================================

class VanetzaBtpAdapter::RxIndicationHandler
    : public vanetza::btp::IndicationInterface
{
public:

    explicit RxIndicationHandler(VanetzaBtpAdapter& owner)
        : m_owner(owner)
    {
    }

    void indicate(
        const vanetza::btp::DataIndication& ind,
        std::unique_ptr<vanetza::UpPacket> packet) override
    {
        NS_LOG_UNCOND(
            "[BTP_HANDLER] indicate() called port="
            << ind.destination_port.get());

        if (!m_owner.m_handler)
        {
            NS_LOG_UNCOND(
                "[BTP_HANDLER] ERROR: RX handler not installed");
            return;
        }

        if (!packet)
        {
            NS_LOG_UNCOND(
                "[BTP_HANDLER] ERROR: UpPacket is null");
            return;
        }

        //----------------------------------------------------------------------
        // Create application-neutral RX result.
        //----------------------------------------------------------------------
        VanetzaBtpAdapter::RxData data;

        data.destinationPort =
            ind.destination_port.get();

        //----------------------------------------------------------------------
        // Extract application payload from Vanetza UpPacket.
        //----------------------------------------------------------------------
        vanetza::ByteBuffer payload;

        if (auto* chunk =
                boost::get<vanetza::ChunkPacket>(&*packet))
        {
            chunk->layer(
                vanetza::OsiLayer::Application).convert(payload);
        }
        else if (auto* cohesive =
                     boost::get<vanetza::CohesivePacket>(&*packet))
        {
            auto net =
                (*cohesive)[vanetza::OsiLayer::Network];

            auto tr =
                (*cohesive)[vanetza::OsiLayer::Transport];

            auto ses =
                (*cohesive)[vanetza::OsiLayer::Session];

            auto pre =
                (*cohesive)[vanetza::OsiLayer::Presentation];

            auto app =
                (*cohesive)[vanetza::OsiLayer::Application];

            NS_LOG_UNCOND(
                "[BTP_HANDLER] Network bytes    = "
                << std::distance(net.begin(), net.end()));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] size(Network)="
                << cohesive->size(vanetza::OsiLayer::Network));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] Transport bytes  = "
                << std::distance(tr.begin(), tr.end()));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] size(Transport)="
                << cohesive->size(vanetza::OsiLayer::Transport));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] Session bytes    = "
                << std::distance(ses.begin(), ses.end()));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] size(Session)="
                << cohesive->size(vanetza::OsiLayer::Session));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] Presentation bytes = "
                << std::distance(pre.begin(), pre.end()));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] size(Presentation)="
                << cohesive->size(vanetza::OsiLayer::Presentation));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] Application bytes = "
                << std::distance(app.begin(), app.end()));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] size(Application)="
                << cohesive->size(vanetza::OsiLayer::Application));

            NS_LOG_UNCOND(
                "[BTP_HANDLER] total="
                << cohesive->size());

            //------------------------------------------------------------------
            // Preserve the existing application payload extraction behavior.
            //
            // In the current Vanetza RX path, after BTP processing the CAM
            // payload is carried in the Session layer.
            //------------------------------------------------------------------
            payload.assign(
                ses.begin(),
                ses.end());
        }
        else
        {
            NS_LOG_UNCOND(
                "[BTP_HANDLER] ERROR: Unknown packet type in UpPacket");
            return;
        }

        //----------------------------------------------------------------------
        // Validate extracted payload.
        //----------------------------------------------------------------------
        if (payload.empty())
        {
            NS_LOG_UNCOND(
                "[BTP_HANDLER] WARNING: Extracted payload is empty");
            return;
        }

        data.payload.assign(
            payload.begin(),
            payload.end());

        NS_LOG_UNCOND(
            "[BTP_HANDLER] Payload extracted, size="
            << data.payload.size());

        //----------------------------------------------------------------------
        // GeoNetworking source position.
        //----------------------------------------------------------------------
        data.hasSourcePosition = true;

        data.latitude =
            ind.source_position.latitude.value() * 1e-7;

        data.longitude =
            ind.source_position.longitude.value() * 1e-7;

        //----------------------------------------------------------------------
        // GeoNetworking transport information.
        //----------------------------------------------------------------------
        data.hasTrafficClass = true;

        data.trafficClass =
            ind.traffic_class.raw();

        //----------------------------------------------------------------------
        // ITS-AID.
        //----------------------------------------------------------------------
        data.hasItsAid = false;

        if (ind.its_aid)
        {
            data.hasItsAid = true;
            data.itsAid = ind.its_aid.value();
        }

        //----------------------------------------------------------------------
        // Permissions.
        //----------------------------------------------------------------------
        data.hasPermissions = false;

        if (ind.permissions)
        {
            data.hasPermissions = true;
            data.permissionsSize =
                ind.permissions->size();
        }

        //----------------------------------------------------------------------
        // Remaining lifetime.
        //----------------------------------------------------------------------
        data.hasLifetime = false;

        if (ind.remaining_packet_lifetime)
        {
            data.hasLifetime = true;

            auto duration =
                ind.remaining_packet_lifetime->decode();

            data.lifetimeMs =
                duration.value() * 1000.0;
        }

        //----------------------------------------------------------------------
        // GeoNetworking sender MID.
        //
        // Vanetza representation stays inside this boundary.
        // The application receives only std::array<uint8_t, 6>.
        //----------------------------------------------------------------------
        const auto& sourceMid =
            ind.source_position.gn_addr.mid();

        for (std::size_t i = 0; i < 6; ++i)
        {
            data.sourceMid[i] =
                sourceMid.octets[i];
        }

        data.hasSourceMid = true;

        //----------------------------------------------------------------------
        // Deliver neutral RX data to the application/runtime.
        //----------------------------------------------------------------------
        m_owner.m_handler(data);
    }

private:

    VanetzaBtpAdapter& m_owner;
};

    //-------------------------------------------------------------------------
    m_router.set_transport_handler(
        vanetza::geonet::UpperProtocol::BTP_B,

//=============================================================================
// BTP TX
//=============================================================================

bool
VanetzaBtpAdapter::SendBtpB(
    PacketContext& context,
    uint16_t destinationPort,
    const std::vector<uint8_t>& payload)
{
    //----------------------------------------------------------------------
    // STEP 4 : BTP_TX
    //----------------------------------------------------------------------
    context.AdvancePipeline(PipelineStage::BTP_TX);

    V2XWriteLog(
        context,
        V2XLogLevel::DEBUG,
        "BTP",
        "BTP_TX_BEGIN",
        "port=" + std::to_string(destinationPort) +
        " payload=" + std::to_string(payload.size()));

    //----------------------------------------------------------------------
    // Build BTP request
    //----------------------------------------------------------------------
    vanetza::btp::DataRequestB request;

    request.destination_port =
        vanetza::uint16be_t(destinationPort);

    request.destination_port_info =
        vanetza::uint16be_t(0);

    request.gn.transport_type =
        vanetza::geonet::TransportType::SHB;

    request.gn.communication_profile =
        vanetza::geonet::CommunicationProfile::ITS_G5;

    request.gn.its_aid =
        vanetza::aid::CA;

    request.gn.traffic_class =
        vanetza::geonet::TrafficClass(0);

    //----------------------------------------------------------------------
    // BTP_TX successfully constructed
    //----------------------------------------------------------------------
    context.MarkSuccess();

    V2XWriteLog(
        context,
        V2XLogLevel::DEBUG,
        "BTP",
        "BTP_TX_OK",
        "port=" + std::to_string(destinationPort));

    //----------------------------------------------------------------------
    // STEP 5 : GEONET_TX
    //----------------------------------------------------------------------
    context.AdvancePipeline(PipelineStage::GEONET_TX);

    V2XWriteLog(
        context,
        V2XLogLevel::DEBUG,
        "GEONET",
        "GEONET_TX_BEGIN",
        "transport=SHB profile=ITS_G5 aid=CA");

    //----------------------------------------------------------------------
    // Build GeoNetworking request
    //----------------------------------------------------------------------
    vanetza::geonet::ShbDataRequest gnRequest(m_mib);

    vanetza::geonet::copy_request_parameters(
        request,
        gnRequest);

    gnRequest.upper_protocol =
        vanetza::geonet::UpperProtocol::BTP_B;

    //----------------------------------------------------------------------
    // BTP-B HEADER
    //
    // Construction belongs to this wrapper.
    // Serialization belongs to Vanetza.
    //----------------------------------------------------------------------
    vanetza::btp::HeaderB btpHeader;

    btpHeader.destination_port =
        request.destination_port;

    btpHeader.destination_port_info =
        request.destination_port_info;

    //----------------------------------------------------------------------
    // Build application payload
    //----------------------------------------------------------------------
    vanetza::ByteBuffer appPayload(
        payload.begin(),
        payload.end());

    //----------------------------------------------------------------------
    // Build complete DownPacket
    //----------------------------------------------------------------------
    auto packet =
        std::make_unique<vanetza::DownPacket>();

    packet->layer(
        vanetza::OsiLayer::Transport) =
        btpHeader;

    packet->layer(
        vanetza::OsiLayer::Application) =
        std::move(appPayload);

    //----------------------------------------------------------------------
    // GEONET_TX successfully constructed
    //----------------------------------------------------------------------
    context.MarkSuccess();

    V2XWriteLog(
        context,
        V2XLogLevel::DEBUG,
        "GEONET",
        "GEONET_TX_OK",
        "transport=SHB profile=ITS_G5 aid=CA");

    //----------------------------------------------------------------------
    // Send through GeoNetworking Router
    //----------------------------------------------------------------------
    auto confirm =
        m_router.request(
            gnRequest,
            std::move(packet));

    if (!confirm.accepted())
    {
        context.MarkFailure();

        context.reason =
            ResultCodeToString(
                confirm.result_code);

        V2XWriteLog(
            context,
            V2XLogLevel::ERROR,
            "GEONET",
            "GEONET_TX_FAIL",
            "result=" + context.reason);

        V2XWriteLog(
            context,
            V2XLogLevel::ERROR,
            "BTP",
            "BTP_TX_FAIL",
            "port=" +
                std::to_string(destinationPort));

        return false;
    }

    context.MarkSuccess();

    context.reason =
        ResultCodeToString(
            confirm.result_code);

    return true;
}

bool
Ns3AccessAdapter::ValidateSecuredMessageV3Tx(
    const std::vector<uint8_t>& networkBytes) const
{
    constexpr std::size_t basicHeaderSize = 4;

    if (networkBytes.size() <= basicHeaderSize)
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=NO_SECURED_PAYLOAD "
            "network_size="
            << networkBytes.size());

        return false;
    }

    // ------------------------------------------------------------------
    // Basic Header
    //
    // For a secured GeoNetworking packet:
    //
    //   byte 0 = 0x12
    //              version       = 1
    //              next_header   = 2
    //
    // The bytes starting at offset 4 contain the serialized
    // SecuredMessage.
    // ------------------------------------------------------------------

    const uint8_t firstOctet =
        networkBytes[0];

    const uint8_t version =
        (firstOctet >> 4) & 0x0F;

    const uint8_t nextHeader =
        firstOctet & 0x0F;

    if (version != 1)
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=INVALID_GN_VERSION "
            "version="
            << static_cast<unsigned>(version));

        return false;
    }

    if (nextHeader != 2)
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=NOT_SECURED_PACKET "
            "next_header="
            << static_cast<unsigned>(nextHeader));

        return false;
    }

    // ------------------------------------------------------------------
    // Extract serialized SecuredMessage
    // ------------------------------------------------------------------

    const std::vector<uint8_t> securedBytes(
        networkBytes.begin() + basicHeaderSize,
        networkBytes.end());

    if (securedBytes.empty())
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=EMPTY_SECURED_MESSAGE");

        return false;
    }

    // ------------------------------------------------------------------
    // Build input stream over the actual wire bytes
    // ------------------------------------------------------------------

    const std::string serialized(
        reinterpret_cast<const char*>(
            securedBytes.data()),
        securedBytes.size());

    std::istringstream input(
        serialized,
        std::ios::binary);

    vanetza::InputArchive archive(input);

    // ------------------------------------------------------------------
    // Deserialize Vanetza Security v3 SecuredMessage
    // ------------------------------------------------------------------

    vanetza::security::v3::SecuredMessage securedMessage;

    std::size_t consumed = 0;

    try
    {
        consumed =
            vanetza::security::v3::deserialize(
                archive,
                securedMessage);
    }
    catch (const std::exception& e)
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=DESERIALIZATION_EXCEPTION "
            "error="
            << e.what());

        return false;
    }
    catch (...)
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=DESERIALIZATION_EXCEPTION "
            "error=UNKNOWN");

        return false;
    }

    if (!archive.is_good())
    {
        NS_LOG_UNCOND(
            "[SECURED_TX] RESULT=FAIL "
            "reason=DESERIALIZATION_ARCHIVE_ERROR "
            "remaining="
            << archive.remaining_bytes());

        return false;
    }

    // ------------------------------------------------------------------
    // ITS-AID
    // ------------------------------------------------------------------

    const vanetza::ItsAid actualAid =
        securedMessage.its_aid();

    const vanetza::ItsAid expectedAid =
        vanetza::aid::CA;

    const bool aidOk =
        actualAid == expectedAid;

    // ------------------------------------------------------------------
    // Structural information
    // ------------------------------------------------------------------

    const std::size_t securedSize =
        securedBytes.size();

    const std::size_t remaining =
        archive.remaining_bytes();

    const bool completeMessageConsumed =
        remaining == 0;

    // ------------------------------------------------------------------
    // Diagnostic output
    // ------------------------------------------------------------------

    NS_LOG_UNCOND(
        "[SECURED_TX] "
        "version="
        << static_cast<unsigned>(version)

        << " next_header="
        << static_cast<unsigned>(nextHeader)

        << " network_size="
        << networkBytes.size()

        << " secured_size="
        << securedSize

        << " consumed="
        << consumed

        << " remaining="
        << remaining

        << " its_aid="
        << static_cast<unsigned>(actualAid)

        << " expected_its_aid="
        << static_cast<unsigned>(expectedAid)

        << " aid_ok="
        << (aidOk ? "YES" : "NO")

        << " complete="
        << (completeMessageConsumed ? "YES" : "NO")

        << " RESULT="
        << ((aidOk && completeMessageConsumed)
                ? "PASS"
                : "FAIL"));

    return aidOk &&
           completeMessageConsumed;
}

bool
Ns3AccessAdapter::ValidateCommonHeaderTx(
    const std::vector<uint8_t>& networkBytes,
    std::size_t transportSize,
    std::size_t applicationSize,
    std::size_t frameSize) const
{
    constexpr std::size_t basicHeaderSize = 4;
    constexpr std::size_t securedMessageHeaderSize = 8;
    constexpr std::size_t commonHeaderSize = 8;

    const bool securedPacket =
        m_mib.itsGnSecurity;

    // ------------------------------------------------------------------
    // Layout:
    //
    // Non-secured:
    //
    //   Basic Header       4
    //   Common Header      8
    //   ...
    //
    // Secured:
    //
    //   Basic Header       4
    //   SecuredMessage    8   <- current Vanetza v3 serialization
    //   Common Header      8
    //   ...
    // ------------------------------------------------------------------

    const std::size_t commonHeaderOffset =
        securedPacket
            ? basicHeaderSize + securedMessageHeaderSize
            : basicHeaderSize;

    const std::size_t requiredSize =
        commonHeaderOffset + commonHeaderSize;

    if (networkBytes.size() < requiredSize)
    {
        NS_LOG_UNCOND(
            "[GEONET_COMMON_TX] RESULT=FAIL "
            "reason=NETWORK_LAYER_TOO_SHORT "
            "secured="
            << (securedPacket ? "YES" : "NO")
            << " size="
            << networkBytes.size()
            << " required="
            << requiredSize);

        return false;
    }

    const std::size_t offset =
        commonHeaderOffset;

    // ------------------------------------------------------------------
    // GeoNetworking Common Header
    //
    // octet 0:
    //   high nibble = next_header
    //   low nibble  = reserved
    //
    // octet 1   = header_type
    // octet 2   = traffic_class
    // octet 3   = flags
    // octet 4-5 = payload length
    // octet 6   = maximum hop limit
    // octet 7   = reserved
    // ------------------------------------------------------------------

    const uint8_t octet0 =
        networkBytes[offset + 0];

    const uint8_t nextHeader =
        (octet0 >> 4) & 0x0F;

    const uint8_t reserved1 =
        octet0 & 0x0F;

    const uint8_t headerType =
        networkBytes[offset + 1];

    const uint8_t trafficClass =
        networkBytes[offset + 2];

    const uint8_t flags =
        networkBytes[offset + 3];

    const uint16_t payloadLength =
        static_cast<uint16_t>(
            (static_cast<uint16_t>(
                 networkBytes[offset + 4]) << 8) |
            static_cast<uint16_t>(
                 networkBytes[offset + 5]));

    const uint8_t maximumHopLimit =
        networkBytes[offset + 6];

    const uint8_t reserved2 =
        networkBytes[offset + 7];

    // ------------------------------------------------------------------
    // Packet type
    // ------------------------------------------------------------------

    const bool isBeacon =
        headerType == 0x10;

    const bool isTsbSingleHop =
        headerType == 0x50;

    // ------------------------------------------------------------------
    // Expected Common Header values
    // ------------------------------------------------------------------

    const uint8_t expectedNextHeader =
        isBeacon
            ? 0
            : (isTsbSingleHop ? 2 : 0);

    const uint8_t expectedHeaderType =
        isBeacon
            ? 0x10
            : (isTsbSingleHop ? 0x50 : 0x00);

    const uint8_t expectedTrafficClass =
        0x00;

    const uint8_t expectedFlags =
        m_mib.itsGnIsMobile
            ? 0x80
            : 0x00;

    const uint8_t expectedMaximumHopLimit =
        1;

    // ------------------------------------------------------------------
    // Payload length
    //
    // Non-secured:
    //
    //   Common Header
    //       -> Transport + Application
    //
    // Secured:
    //
    //   Common Header is inside SecuredMessage.
    //   The outer ChunkPacket has:
    //
    //       Network    = complete SecuredPacket
    //       Transport = 0
    //       Application = 0
    //
    //   Therefore the outer layer sizes MUST NOT be used here.
    //
    // For the current secured SHB:
    //
    //   BTP-B = 4 bytes
    //   CAM payload = CommonHeader.payloadLength - 4
    //
    // Thus the minimum valid payload is 4 bytes.
    // ------------------------------------------------------------------

    bool payloadLengthOk = false;
    std::size_t expectedPayloadLength = 0;

    if (securedPacket)
    {
        constexpr std::size_t btpBHeaderSize = 4;

        expectedPayloadLength =
            btpBHeaderSize;

        payloadLengthOk =
            payloadLength >= btpBHeaderSize;
    }
    else
    {
        expectedPayloadLength =
            transportSize + applicationSize;

        payloadLengthOk =
            payloadLength == expectedPayloadLength;
    }

    // ------------------------------------------------------------------
    // Frame size
    //
    // Non-secured:
    //
    //   network + transport + application
    //
    // Secured:
    //
    //   the complete SecuredMessage is already in Network.
    //
    // Therefore Transport/Application outside Network are zero.
    // ------------------------------------------------------------------

    std::size_t expectedFrameSize = 0;

    bool frameSizeOk = false;

    if (securedPacket)
    {
        expectedFrameSize =
            networkBytes.size();

        frameSizeOk =
            frameSize == expectedFrameSize;
    }
    else
    {
        expectedFrameSize =
            networkBytes.size() +
            expectedPayloadLength;

        frameSizeOk =
            frameSize == expectedFrameSize;
    }

    // ------------------------------------------------------------------
    // Field validation
    // ------------------------------------------------------------------

    const bool nextHeaderOk =
        nextHeader == expectedNextHeader;

    const bool reserved1Ok =
        reserved1 == 0;

    const bool headerTypeOk =
        headerType == expectedHeaderType;

    const bool trafficClassOk =
        trafficClass == expectedTrafficClass;

    const bool flagsOk =
        flags == expectedFlags;

    const bool maximumHopLimitOk =
        maximumHopLimit ==
        expectedMaximumHopLimit;

    const bool reserved2Ok =
        reserved2 == 0;

    // ------------------------------------------------------------------
    // Packet-specific validation
    // ------------------------------------------------------------------

    const bool packetTypeOk =
        isBeacon ||
        isTsbSingleHop;

    const bool beaconPayloadOk =
        !isBeacon ||
        (payloadLength == 0 &&
         (!securedPacket ||
          payloadLength == 0));

    // ------------------------------------------------------------------
    // Secured-packet structural validation
    //
    // The Common Header is inside the SecuredMessage.
    //
    // We therefore additionally require enough bytes after the Common
    // Header for:
    //
    //   SHB Extended Header = 28
    //   BTP-B               = 4
    //
    // CAM is accounted for by payloadLength >= 4 but its exact boundary
    // is validated by the SecuredMessage deserialization.
    // ------------------------------------------------------------------

    bool securedStructureOk = true;

    if (securedPacket && isTsbSingleHop)
    {
        constexpr std::size_t shbHeaderSize = 28;
        constexpr std::size_t btpBHeaderSize = 4;

        const std::size_t payloadOffset =
            offset + commonHeaderSize;

        const std::size_t minimumAfterCommon =
            shbHeaderSize +
            btpBHeaderSize;

        securedStructureOk =
            networkBytes.size() >=
            payloadOffset + minimumAfterCommon;
    }

    // ------------------------------------------------------------------
    // Final result
    // ------------------------------------------------------------------

    const bool result =
        nextHeaderOk &&
        reserved1Ok &&
        headerTypeOk &&
        trafficClassOk &&
        flagsOk &&
        payloadLengthOk &&
        maximumHopLimitOk &&
        reserved2Ok &&
        packetTypeOk &&
        beaconPayloadOk &&
        frameSizeOk &&
        securedStructureOk;

    // ------------------------------------------------------------------
    // Diagnostic
    // ------------------------------------------------------------------

    std::ostringstream oss;

    oss << "[GEONET_COMMON_TX]"
        << " secured="
        << (securedPacket ? "YES" : "NO")

        << " common_offset="
        << offset

        << " next_header="
        << static_cast<unsigned>(nextHeader)
        << " expected="
        << static_cast<unsigned>(expectedNextHeader)
        << " ok="
        << (nextHeaderOk ? "YES" : "NO")

        << " reserved1="
        << static_cast<unsigned>(reserved1)
        << " ok="
        << (reserved1Ok ? "YES" : "NO")

        << " header_type=0x"
        << std::hex
        << std::setw(2)
        << std::setfill('0')
        << static_cast<unsigned>(headerType)
        << std::dec

        << " expected=0x"
        << std::hex
        << std::setw(2)
        << std::setfill('0')
        << static_cast<unsigned>(expectedHeaderType)
        << std::dec

        << " ok="
        << (headerTypeOk ? "YES" : "NO")

        << " traffic_class="
        << static_cast<unsigned>(trafficClass)
        << " expected="
        << static_cast<unsigned>(expectedTrafficClass)
        << " ok="
        << (trafficClassOk ? "YES" : "NO")

        << " flags="
        << static_cast<unsigned>(flags)
        << " expected="
        << static_cast<unsigned>(expectedFlags)
        << " ok="
        << (flagsOk ? "YES" : "NO")

        << " payload_length="
        << payloadLength
        << " minimum_expected="
        << expectedPayloadLength
        << " transport_outer="
        << transportSize
        << " application_outer="
        << applicationSize
        << " payload_ok="
        << (payloadLengthOk ? "YES" : "NO")

        << " maximum_hop_limit="
        << static_cast<unsigned>(maximumHopLimit)
        << " expected="
        << static_cast<unsigned>(expectedMaximumHopLimit)
        << " ok="
        << (maximumHopLimitOk ? "YES" : "NO")

        << " reserved2="
        << static_cast<unsigned>(reserved2)
        << " ok="
        << (reserved2Ok ? "YES" : "NO")

        << " packet_type="
        << (isBeacon
                ? "BEACON"
                : (isTsbSingleHop
                       ? "TSB_SINGLE_HOP"
                       : "UNKNOWN"))

        << " packet_type_ok="
        << (packetTypeOk ? "YES" : "NO")

        << " secured_structure_ok="
        << (securedStructureOk ? "YES" : "NO")

        << " network_size="
        << networkBytes.size()

        << " expected_frame_size="
        << expectedFrameSize

        << " actual_frame_size="
        << frameSize

        << " frame_size_ok="
        << (frameSizeOk ? "YES" : "NO");

    oss << " raw=";

    for (std::size_t i = offset;
         i < offset + commonHeaderSize;
         ++i)
    {
        if (i != offset)
        {
            oss << " ";
        }

        oss << std::hex
            << std::setw(2)
            << std::setfill('0')
            << static_cast<unsigned>(
                   networkBytes[i]);
    }

    oss << std::dec
        << " RESULT="
        << (result ? "PASS" : "FAIL");

    NS_LOG_UNCOND(oss.str());

    return result;
}

bool
Ns3AccessAdapter::ValidateBasicHeaderTx(
    const std::vector<uint8_t>& networkBytes) const
{
    if (networkBytes.size() < 4)
    {
        NS_LOG_UNCOND(
            "[GEONET_BASIC_TX] RESULT=FAIL "
            "reason=NETWORK_LAYER_TOO_SHORT "
            "size="
            << networkBytes.size());

        return false;
    }

    // ------------------------------------------------------------------
    // GeoNetworking Basic Header
    //
    // Octet 0:
    //   bits 7..4 = version
    //   bits 3..0 = next_header
    //
    // Examples:
    //
    //   0x11 -> version=1, next_header=1 (COMMON_HEADER)
    //   0x12 -> version=1, next_header=2 (SECURED_PACKET)
    //
    // Octet 1:
    //   reserved
    //
    // Octet 2:
    //   lifetime
    //
    // Octet 3:
    //   remaining hop limit
    // ------------------------------------------------------------------

    const uint8_t octet0 =
        networkBytes[0];

    const uint8_t version =
        (octet0 >> 4) & 0x0F;

    const uint8_t nextHeader =
        octet0 & 0x0F;

    const uint8_t reserved =
        networkBytes[1];

    const uint8_t lifetimeRaw =
        networkBytes[2];

    const uint8_t remainingHopLimit =
        networkBytes[3];


    // ------------------------------------------------------------------
    // Expected values from the active Vanetza MIB
    // ------------------------------------------------------------------

    const uint8_t expectedVersion =
        static_cast<uint8_t>(
            m_mib.itsGnProtocolVersion);

    // next_header:
    //
    //   1 = Common Header
    //   2 = Secured Packet
    //
    // The expected value follows the active GeoNetworking
    // security configuration.
    const uint8_t expectedNextHeader =
        m_mib.itsGnSecurity
            ? 2
            : 1;


    // ------------------------------------------------------------------
    // Lifetime semantic decoding
    //
    // bits 0..1 = base
    // bits 2..7 = multiplier
    //
    // Base values:
    //   0 -> 50 ms
    //   1 -> 1 s
    //   2 -> 10 s
    //   3 -> 100 s
    // ------------------------------------------------------------------

    const uint8_t lifetimeBase =
        lifetimeRaw & 0x03;

    const uint8_t lifetimeMultiplier =
        lifetimeRaw >> 2;

    double lifetimeSeconds = 0.0;

    switch (lifetimeBase)
    {
        case 0:
            lifetimeSeconds =
                static_cast<double>(
                    lifetimeMultiplier) * 0.05;
            break;

        case 1:
            lifetimeSeconds =
                static_cast<double>(
                    lifetimeMultiplier);
            break;

        case 2:
            lifetimeSeconds =
                static_cast<double>(
                    lifetimeMultiplier) * 10.0;
            break;

        case 3:
            lifetimeSeconds =
                static_cast<double>(
                    lifetimeMultiplier) * 100.0;
            break;

        default:
            lifetimeSeconds = -1.0;
            break;
    }


    // ------------------------------------------------------------------
    // Basic Header validation
    // ------------------------------------------------------------------

    const bool versionOk =
        version == expectedVersion;

    const bool nextHeaderOk =
        nextHeader == expectedNextHeader;

    const bool reservedOk =
        reserved == 0;

    const bool lifetimeOk =
        lifetimeSeconds >= 0.0;

    // GeoNetworking conformance criterion:
    // remaining_hop_limit >= 1.
    const uint8_t minimumRemainingHopLimit =
        1;

    const bool remainingHopLimitOk =
        remainingHopLimit >=
        minimumRemainingHopLimit;

    const bool result =
        versionOk &&
        nextHeaderOk &&
        reservedOk &&
        lifetimeOk &&
        remainingHopLimitOk;


    // ------------------------------------------------------------------
    // Diagnostic output
    // ------------------------------------------------------------------

    NS_LOG_UNCOND(
        "[GEONET_BASIC_TX] "
        "version="
        << static_cast<unsigned>(version)
        << " expected="
        << static_cast<unsigned>(expectedVersion)
        << " ok="
        << (versionOk ? "YES" : "NO")

        << " next_header="
        << static_cast<unsigned>(nextHeader)
        << "("
        << NextHeaderBasicToString(nextHeader)
        << ")"
        << " expected="
        << static_cast<unsigned>(expectedNextHeader)
        << " ok="
        << (nextHeaderOk ? "YES" : "NO")

        << " reserved="
        << static_cast<unsigned>(reserved)
        << " ok="
        << (reservedOk ? "YES" : "NO")

        << " lifetime_raw="
        << static_cast<unsigned>(lifetimeRaw)
        << " lifetime_base="
        << static_cast<unsigned>(lifetimeBase)
        << " lifetime_multiplier="
        << static_cast<unsigned>(lifetimeMultiplier)
        << " lifetime_seconds="
        << lifetimeSeconds
        << " ok="
        << (lifetimeOk ? "YES" : "NO")

        << " remaining_hop_limit="
        << static_cast<unsigned>(remainingHopLimit)
        << " minimum="
        << static_cast<unsigned>(minimumRemainingHopLimit)
        << " ok="
        << (remainingHopLimitOk ? "YES" : "NO")

        << " raw_octet=0x"
        << std::hex
        << std::setw(2)
        << std::setfill('0')
        << static_cast<unsigned>(networkBytes[3])
        << std::dec

        << " raw="
        << std::hex
        << std::setw(2)
        << std::setfill('0')
        << static_cast<unsigned>(networkBytes[0])
        << " "
        << std::setw(2)
        << static_cast<unsigned>(networkBytes[1])
        << " "
        << std::setw(2)
        << static_cast<unsigned>(networkBytes[2])
        << " "
        << std::setw(2)
        << static_cast<unsigned>(networkBytes[3])
        << std::dec

        << " RESULT="
        << (result ? "PASS" : "FAIL"));

    return result;
}

bool
Ns3AccessAdapter::ValidateShbExtendedHeaderTx(
    const std::vector<uint8_t>& networkBytes) const
{
    constexpr std::size_t basicHeaderSize = 4;
    constexpr std::size_t securedMessageHeaderSize = 8;
    constexpr std::size_t commonHeaderSize = 8;
    constexpr std::size_t shbHeaderSize = 28;
    constexpr std::size_t longPositionVectorSize = 24;

    const bool securedPacket =
        m_mib.itsGnSecurity;

    const std::size_t commonHeaderOffset =
        securedPacket
            ? basicHeaderSize + securedMessageHeaderSize
            : basicHeaderSize;

    const std::size_t shbOffset =
        commonHeaderOffset +
        commonHeaderSize;

    const std::size_t dccOffset =
        shbOffset +
        longPositionVectorSize;

    const std::size_t minimumNetworkSize =
        shbOffset +
        shbHeaderSize;

    // ------------------------------------------------------------------
    // Size validation
    // ------------------------------------------------------------------

    if (networkBytes.size() < minimumNetworkSize)
    {
        NS_LOG_UNCOND(
            "[GEONET_SHB_TX] RESULT=FAIL "
            "reason=NETWORK_LAYER_TOO_SHORT "
            "secured="
            << (securedPacket ? "YES" : "NO")
            << " network_layer_size="
            << networkBytes.size()
            << " expected_min="
            << minimumNetworkSize);

        return false;
    }

    // ------------------------------------------------------------------
    // Verify Common Header type from the correct location
    // ------------------------------------------------------------------

    const uint8_t headerType =
        networkBytes[commonHeaderOffset + 1];

    const bool shbTypeOk =
        headerType == 0x50;

    if (!shbTypeOk)
    {
        NS_LOG_UNCOND(
            "[GEONET_SHB_TX] RESULT=FAIL "
            "reason=NOT_TSB_SINGLE_HOP "
            "header_type=0x"
            << std::hex
            << std::setw(2)
            << std::setfill('0')
            << static_cast<unsigned>(headerType)
            << std::dec);

        return false;
    }

    // ------------------------------------------------------------------
    // SHB Extended Header
    //
    //   bytes  0..23 = Long Position Vector
    //   bytes 24..27 = DCC
    // ------------------------------------------------------------------

    const std::size_t pvOffset =
        shbOffset;

    // ------------------------------------------------------------------
    // GN Address
    // ------------------------------------------------------------------

    const uint16_t addressControl =
        static_cast<uint16_t>(
            (static_cast<uint16_t>(
                 networkBytes[pvOffset + 0]) << 8) |
            static_cast<uint16_t>(
                 networkBytes[pvOffset + 1]));

    const bool manuallyConfigured =
        (addressControl & 0x8000) != 0;

    const uint8_t stationType =
        static_cast<uint8_t>(
            (addressControl & 0x7C00) >> 10);

    const uint16_t countryCode =
        addressControl & 0x03FF;

    std::ostringstream mid;

    mid << std::hex
        << std::setw(2)
        << std::setfill('0')
        << static_cast<unsigned>(
               networkBytes[pvOffset + 2])
        << ":"
        << std::setw(2)
        << static_cast<unsigned>(
               networkBytes[pvOffset + 3])
        << ":"
        << std::setw(2)
        << static_cast<unsigned>(
               networkBytes[pvOffset + 4])
        << ":"
        << std::setw(2)
        << static_cast<unsigned>(
               networkBytes[pvOffset + 5])
        << ":"
        << std::setw(2)
        << static_cast<unsigned>(
               networkBytes[pvOffset + 6])
        << ":"
        << std::setw(2)
        << static_cast<unsigned>(
               networkBytes[pvOffset + 7])
        << std::dec;

    // ------------------------------------------------------------------
    // Timestamp
    // ------------------------------------------------------------------

    const uint32_t timestamp =
        (static_cast<uint32_t>(
             networkBytes[pvOffset + 8]) << 24) |
        (static_cast<uint32_t>(
             networkBytes[pvOffset + 9]) << 16) |
        (static_cast<uint32_t>(
             networkBytes[pvOffset + 10]) << 8) |
        static_cast<uint32_t>(
            networkBytes[pvOffset + 11]);

    // ------------------------------------------------------------------
    // Timestamp validation
    //
    // The Router stores the LongPositionVector used to construct
    // the GeoNetworking SHB.
    //
    // GeoNetworking Timestamp:
    //   milliseconds since 2004-01-01 00:00:00
    //
    // The timestamp serialized on the wire must match the timestamp
    // of the Router's current local Position Vector.
    // ------------------------------------------------------------------

    const auto& localPositionVector = m_router.get_local_position_vector();

    const uint32_t routerTimestamp = localPositionVector.timestamp.raw();

    const bool timestampNonZero = timestamp != 0;

    const bool timestampOk = timestampNonZero && timestamp == routerTimestamp;

    const int64_t timestampDeltaMs = static_cast<int64_t>(timestamp) - static_cast<int64_t>(routerTimestamp);

    // ------------------------------------------------------------------
    // Latitude / Longitude
    // ------------------------------------------------------------------

    const int32_t latitudeRaw =
        static_cast<int32_t>(
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 12]) << 24) |
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 13]) << 16) |
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 14]) << 8) |
            static_cast<uint32_t>(
                networkBytes[pvOffset + 15]));

    const int32_t longitudeRaw =
        static_cast<int32_t>(
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 16]) << 24) |
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 17]) << 16) |
            (static_cast<uint32_t>(
                 networkBytes[pvOffset + 18]) << 8) |
            static_cast<uint32_t>(
                networkBytes[pvOffset + 19]));

    const double latitudeDegrees =
        static_cast<double>(latitudeRaw) /
        10000000.0;

    const double longitudeDegrees =
        static_cast<double>(longitudeRaw) /
        10000000.0;

    // ------------------------------------------------------------------
    // PAI + Speed
    // ------------------------------------------------------------------

    const uint16_t paiAndSpeed =
        static_cast<uint16_t>(
            (static_cast<uint16_t>(
                 networkBytes[pvOffset + 20]) << 8) |
            static_cast<uint16_t>(
                 networkBytes[pvOffset + 21]));

    const bool pai =
        (paiAndSpeed & 0x8000) != 0;

    const uint16_t speedRaw =
        paiAndSpeed & 0x7FFF;

    const double speedMps =
        static_cast<double>(speedRaw) / 100.0;

    // ------------------------------------------------------------------
    // Heading
    // ------------------------------------------------------------------

    const uint16_t headingRaw =
        static_cast<uint16_t>(
            (static_cast<uint16_t>(
                 networkBytes[pvOffset + 22]) << 8) |
            static_cast<uint16_t>(
                networkBytes[pvOffset + 23]));

    const double headingDegrees =
        static_cast<double>(headingRaw) * 0.1;

    // ------------------------------------------------------------------
    // DCC
    // ------------------------------------------------------------------

    const uint32_t dccRaw =
        (static_cast<uint32_t>(
             networkBytes[dccOffset + 0]) << 24) |
        (static_cast<uint32_t>(
             networkBytes[dccOffset + 1]) << 16) |
        (static_cast<uint32_t>(
             networkBytes[dccOffset + 2]) << 8) |
        static_cast<uint32_t>(
            networkBytes[dccOffset + 3]);

    // ------------------------------------------------------------------
    // Structural validation
    // ------------------------------------------------------------------

    const bool sizeOk =
        networkBytes.size() >=
        minimumNetworkSize;

    const bool latitudeOk =
        latitudeDegrees >= -90.0 &&
        latitudeDegrees <= 90.0;

    const bool longitudeOk =
        longitudeDegrees >= -180.0 &&
        longitudeDegrees <= 180.0;

    const bool headingOk =
        headingRaw <= 3600;

    const bool addressOk = manuallyConfigured;

    const bool structureOk = sizeOk && shbTypeOk && latitudeOk && longitudeOk && headingOk && addressOk && timestampOk;

    // ------------------------------------------------------------------
    // Raw SHB
    // ------------------------------------------------------------------

    std::ostringstream raw;

    raw << std::hex
        << std::setfill('0');

    for (std::size_t i = 0;
         i < shbHeaderSize;
         ++i)
    {
        if (i != 0)
        {
            raw << " ";
        }

        raw << std::setw(2)
            << static_cast<unsigned>(
                   networkBytes[
                       shbOffset + i]);
    }

    // ------------------------------------------------------------------
    // Diagnostic
    // ------------------------------------------------------------------

    std::ostringstream log;

    log << "[GEONET_SHB_TX] "
        << "secured="
        << (securedPacket ? "YES" : "NO")

        << " common_offset="
        << commonHeaderOffset

        << " shb_offset="
        << shbOffset

        << " size="
        << shbHeaderSize

        << " expected="
        << shbHeaderSize

        << " source_mid="
        << mid.str()

        << " manually_configured="
        << (manuallyConfigured ? "YES" : "NO")

        << " timestamp_raw="
        << timestamp

        << " router_timestamp="
        << routerTimestamp

        << " timestamp_delta_ms="
        << timestampDeltaMs

        << " timestamp_ok="
        << (timestampOk ? "YES" : "NO")

        << " station_type="
        << static_cast<unsigned>(
               stationType)

        << " country_code="
        << countryCode

        << " timestamp="
        << timestamp

        << " latitude_raw="
        << latitudeRaw

        << " latitude_deg="
        << latitudeDegrees

        << " longitude_raw="
        << longitudeRaw

        << " longitude_deg="
        << longitudeDegrees

        << " pai="
        << (pai ? "1" : "0")

        << " speed_raw="
        << speedRaw

        << " speed_mps="
        << speedMps

        << " heading_raw="
        << headingRaw

        << " heading_deg="
        << headingDegrees

        << " dcc_raw=0x"
        << std::hex
        << std::setw(8)
        << std::setfill('0')
        << dccRaw
        << std::dec

        << " address_ok="
        << (addressOk ? "YES" : "NO")

        << " latitude_ok="
        << (latitudeOk ? "YES" : "NO")

        << " longitude_ok="
        << (longitudeOk ? "YES" : "NO")

        << " heading_ok="
        << (headingOk ? "YES" : "NO")

        << " structure_ok="
        << (structureOk ? "YES" : "NO")

        << " raw="
        << raw.str()

        << " RESULT="
        << (structureOk ? "PASS" : "FAIL");

    NS_LOG_UNCOND(log.str());

    return structureOk;
}

Ns3AccessAdapter::Ns3AccessAdapter(VehicleState& vehicle, const vanetza::geonet::MIB& mib, vanetza::geonet::Router& router) : m_vehicle(vehicle), m_mib(mib), m_router(router) {}

std::vector<uint8_t>
Ns3AccessAdapter::SerializePacket(const vanetza::ChunkPacket& packet) const
{
    std::vector<uint8_t> bytes;

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] =================================================");

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] BEGIN ChunkPacket -> ns3 frame");

    for (int layer = static_cast<int>(vanetza::OsiLayer::Physical);
         layer <= static_cast<int>(vanetza::OsiLayer::Application);
         ++layer)
    {
        const auto osiLayer =
            static_cast<vanetza::OsiLayer>(layer);

        vanetza::ByteBuffer buffer;

        packet.layer(osiLayer).convert(buffer);

        const std::size_t offset = bytes.size();
        const std::size_t size = buffer.size();

        std::vector<uint8_t> layerBytes(
            buffer.begin(),
            buffer.end());

        NS_LOG_UNCOND(
            "[TX_BYTELEVEL] layer="
            << LayerName(osiLayer)
            << " offset="
            << offset
            << " size="
            << size);

        if (!layerBytes.empty())
        {
            NS_LOG_UNCOND(
                "[TX_BYTELEVEL] "
                << LayerName(osiLayer)
                << "_HEX_BEGIN");

            NS_LOG_UNCOND(HexDump(layerBytes));

            NS_LOG_UNCOND(
                "[TX_BYTELEVEL] "
                << LayerName(osiLayer)
                << "_HEX_END");
        }

        bytes.insert(
            bytes.end(),
            layerBytes.begin(),
            layerBytes.end());
    }

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] FINAL_FRAME_SIZE="
        << bytes.size());

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] FINAL_FRAME_HEX_BEGIN");

    NS_LOG_UNCOND(HexDump(bytes));

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] FINAL_FRAME_HEX_END");

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] =================================================");

    return bytes;
}


void
Ns3AccessAdapter::request(
    const vanetza::access::DataRequest&,
    std::unique_ptr<vanetza::ChunkPacket> packet)
{
    std::shared_ptr<PacketContext> ctx =
        TxBridgeContext::Current();

    //----------------------------------------------------------------------
    // Validation
    //----------------------------------------------------------------------

    if (!packet)
    {
        if (ctx)
        {
            ctx->AdvancePipeline(PipelineStage::ACCESS_TX);
            ctx->MarkFailure();
            ctx->reason = "null_packet";

            V2XWriteLog(
                *ctx,
                V2XLogLevel::ERROR,
                "ACCESS",
                "ACCESS_TX_FAIL",
                "Null ChunkPacket");

            TxBridgeContext::Pop();
        }

        return;
    }

    if (!m_vehicle.socket)
    {
        if (ctx)
        {
            ctx->AdvancePipeline(PipelineStage::ACCESS_TX);
            ctx->MarkFailure();
            ctx->reason = "socket_unavailable";

            V2XWriteLog(
                *ctx,
                V2XLogLevel::ERROR,
                "ACCESS",
                "ACCESS_TX_FAIL",
                "Socket unavailable");

            TxBridgeContext::Pop();
        }

        return;
    }

    //----------------------------------------------------------------------
    // Serialize complete Down/ChunkPacket
    //----------------------------------------------------------------------

    std::vector<uint8_t> frame =
        SerializePacket(*packet);

    //----------------------------------------------------------------------
     // GeoNetworking Network Layer
    //----------------------------------------------------------------------

    vanetza::ByteBuffer networkBuffer;

    packet->layer(vanetza::OsiLayer::Network)
        .convert(networkBuffer);

    std::vector<uint8_t> networkBytes(
        networkBuffer.begin(),
        networkBuffer.end());

    NS_LOG_UNCOND(
        "[GEONET_BASIC_TX] network_layer_size="
        << networkBytes.size());

    if (!networkBytes.empty())
    {
        NS_LOG_UNCOND(
            "[GEONET_BASIC_TX] NETWORK_HEX_BEGIN");

        NS_LOG_UNCOND(
            HexDump(networkBytes));

        NS_LOG_UNCOND(
            "[GEONET_BASIC_TX] NETWORK_HEX_END");
    }

    ValidateBasicHeaderTx(networkBytes);

    if (m_mib.itsGnSecurity) {
        // --------------------------------------------------------------
        // Secured Packet
        // --------------------------------------------------------------

        ValidateSecuredMessageV3Tx(networkBytes);

        ValidateCommonHeaderTx(networkBytes, 0, 0, frame.size());

        ValidateShbExtendedHeaderTx(networkBytes);
        
    }else{

        // --------------------------------------------------------------
        // Normal GeoNetworking Packet
        // --------------------------------------------------------------

        vanetza::ByteBuffer transportBuffer;

        packet->layer(vanetza::OsiLayer::Transport).convert(transportBuffer);

        const std::size_t transportSize = transportBuffer.size();

        vanetza::ByteBuffer applicationBuffer;

        packet->layer(vanetza::OsiLayer::Application).convert(applicationBuffer);

        const std::size_t applicationSize = applicationBuffer.size();

        ValidateCommonHeaderTx(networkBytes, transportSize, applicationSize, frame.size());

        const uint8_t headerType = (networkBytes.size() >= 6) ? networkBytes[5] : 0x00;

        if (headerType == 0x50) {

            ValidateShbExtendedHeaderTx(networkBytes);
        }
    }


    if (frame.empty()) {
        if (ctx) {

            ctx->AdvancePipeline(PipelineStage::ACCESS_TX);
            ctx->MarkFailure();
            ctx->reason = "empty_frame";

            V2XWriteLog(
                *ctx,
                V2XLogLevel::ERROR,
                "ACCESS",
                "ACCESS_TX_FAIL",
                "Serialized frame is empty");

            TxBridgeContext::Pop();
        }

        return;
    }


    //----------------------------------------------------------------------
     // STEP 6 : ACCESS_TX
    //----------------------------------------------------------------------

    if (ctx)
    {
        ctx->AdvancePipeline(PipelineStage::ACCESS_TX);

        ctx->payload.bytes = frame;

        V2XWriteLog(
            *ctx,
            V2XLogLevel::DEBUG,
            "ACCESS",
            "ACCESS_TX_BEGIN",
            "Serialized complete Vanetza frame");

        ctx->MarkSuccess();

        V2XWriteLog(
            *ctx,
            V2XLogLevel::DEBUG,
            "ACCESS",
            "ACCESS_TX_OK",
            "bytes=" + std::to_string(frame.size()));
    }

    //----------------------------------------------------------------------
    // Convert vector<uint8_t> -> ns3::Packet
    //----------------------------------------------------------------------

    Ptr<Packet> ns3Packet =
        Create<Packet>(
            frame.data(),
            frame.size());

    NS_LOG_UNCOND(
        "[TX_BYTELEVEL] ns3::Packet created size="
        << ns3Packet->GetSize());

    NS_ASSERT_MSG(
        ns3Packet->GetSize() == frame.size(),
        "ns3::Packet size differs from serialized frame");

    Ptr<Socket> socket =
        m_vehicle.socket;

    //----------------------------------------------------------------------
    // Deferred send
    //----------------------------------------------------------------------

    Simulator::ScheduleNow(
        [ctx, socket, ns3Packet]()
        {
            //--------------------------------------------------------------
            // Internal Router transmissions without application context
            //--------------------------------------------------------------

            if (!ctx)
            {
                socket->SendTo(
                    ns3Packet,
                    0,
                    InetSocketAddress(
                        Ipv4Address("10.0.0.255"),
                        5000));

                return;
            }

            //--------------------------------------------------------------
            // STEP 7 : SEND
            //--------------------------------------------------------------

            ctx->AdvancePipeline(
                PipelineStage::SEND);

            V2XWriteLog(
                *ctx,
                V2XLogLevel::DEBUG,
                "ACCESS",
                "SEND_BEGIN",
                "ns3_packet_bytes="
                + std::to_string(
                    ns3Packet->GetSize()));

            int sent =
                socket->SendTo(
                    ns3Packet,
                    0,
                    InetSocketAddress(
                        Ipv4Address("10.0.0.255"),
                        5000));

            if (sent < 0)
            {
                ctx->MarkFailure();
                ctx->reason = "SendTo failed";

                V2XWriteLog(
                    *ctx,
                    V2XLogLevel::ERROR,
                    "ACCESS",
                    "SEND_FAIL",
                    "SendTo returned "
                    + std::to_string(sent));

                TxBridgeContext::Pop();
                return;
            }

            ctx->MarkSuccess();
            ctx->reason = "SendTo OK";

            V2XWriteLog(
                *ctx,
                V2XLogLevel::INFO,
                "ACCESS",
                "SEND_OK",
                "bytes="
                + std::to_string(sent));

            //--------------------------------------------------------------
            // STEP 8 : TX_COMPLETE
            //--------------------------------------------------------------

            ctx->AdvancePipeline(
                PipelineStage::TX_COMPLETE);

            ctx->MarkSuccess();

            V2XWriteLog(
                *ctx,
                V2XLogLevel::INFO,
                "TX",
                "TX_COMPLETE",
                "Transmission completed");

            TxBridgeContext::Pop();
        });
}

bool
VanetzaGeoNetAdapter::Initialize()
{
    if (m_impl->initialized)
    {
        return true;
    }

    //=====================================================================
    // Manual runtime
    //=====================================================================

    m_impl->runtime =
        std::make_unique<vanetza::ManualRuntime>();

    //=====================================================================
    // GeoNetworking security
    //=====================================================================

    auto signService =
        std::make_unique<
            vanetza::security::v3::DummySignService>(
                *m_impl->runtime);

    auto verifyService =
        std::make_unique<
            vanetza::security::DummyVerifyService>(
                vanetza::security::VerificationReport::Success,
                vanetza::security::CertificateValidity::valid());

    m_impl->securityEntity =
        std::make_unique<
            vanetza::security::DelegatingSecurityEntity>(
                std::move(signService),
                std::move(verifyService));

    //=====================================================================
    // MIB
    //=====================================================================

    m_impl->mib =
        std::make_unique<vanetza::geonet::MIB>();

    m_impl->mib->itsGnSecurity = true;

    //=====================================================================
    // GN_ADDR
    //=====================================================================

    vanetza::MacAddress mac =
        vanetza::create_mac_address(
            m_impl->vehicle.identity.stationId);

    mac.octets[0] = 0x02;
    mac.octets[1] = 0x00;

    vanetza::geonet::Address gnAddr;

    gnAddr.mid(mac);

    gnAddr.station_type(
        vanetza::geonet::StationType::Passenger_Car);

    gnAddr.is_manually_configured(true);

    gnAddr.country_code(
        vanetza::BitNumber<unsigned, 10>(0));

    m_impl->mib->itsGnLocalGnAddr =
        gnAddr;

    //=====================================================================
    // Local MID -> neutral context representation
    //=====================================================================

    m_impl->localMid =
        VanetzaGeoNetMidAdapter::ToGeoNetMid(
            gnAddr.mid());

    VehicleRegistry::Instance().RegisterMid(
        m_impl->localMid,
        m_impl->vehicle);

    //=====================================================================
    // Router
    //=====================================================================

    m_impl->router =
        std::make_unique<vanetza::geonet::Router>(
            *m_impl->runtime,
            *m_impl->mib);

    m_impl->router->set_security_entity(
        m_impl->securityEntity.get());

    m_impl->router->set_address(gnAddr);

    //=====================================================================
    // Diagnostics
    //=====================================================================

    m_impl->router->packet_dropped =
        [](vanetza::geonet::Router::PacketDropReason reason)
    {
        NS_LOG_UNCOND(
            "[ROUTER_DROP] reason="
            << static_cast<int>(reason));
    };

    m_impl->router->forwarding_stopped =
        [](vanetza::geonet::Router::ForwardingStopReason reason)
    {
        NS_LOG_UNCOND(
            "[ROUTER_FORWARD_STOP] reason="
            << static_cast<int>(reason));
    };

    //=====================================================================
    // Access / DCC
    //
    // Still instantiated here temporarily, because these wrappers
    // currently require direct Router/MIB references.
    //=====================================================================

    m_impl->access =
        std::make_unique<Ns3AccessAdapter>(
            m_impl->vehicle,
            *m_impl->mib,
            *m_impl->router);

    m_impl->dccAdapter =
        std::make_unique<Ns3DccAdapter>(
            *m_impl->access);

    m_impl->router->set_access_interface(
        m_impl->dccAdapter.get());

    //=====================================================================
    // Position provider
    //=====================================================================

    m_impl->position =
        std::make_unique<Ns3PositionProvider>(
            m_impl->vehicle);

    m_impl->router->update_position(
        m_impl->position->position_fix());

    //=====================================================================
    // BTP
    //=====================================================================

    m_impl->btpAdapter =
        std::make_unique<VanetzaBtpAdapter>(
            *m_impl->router,
            *m_impl->mib);

    m_impl->initialized = true;

    return true;
}

void
VanetzaGeoNetAdapter::Shutdown()
{
    if (!m_impl || !m_impl->initialized)
    {
        return;
    }

    // Preserve the existing destruction dependency order.
    m_impl->btpAdapter.reset();
    m_impl->dccAdapter.reset();
    m_impl->access.reset();
    m_impl->router.reset();
    m_impl->securityEntity.reset();
    m_impl->position.reset();
    m_impl->mib.reset();
    m_impl->runtime.reset();

    m_impl->initialized = false;
}

VanetzaGeoNetAdapter::IsInitialized() const
{
    return m_impl &&
           m_impl->initialized;
}

VanetzaGeoNetAdapter::UpdatePosition()

    if (!IsInitialized())
    {
        return false;
    }

    m_impl->router->update_position(
        m_impl->position->position_fix());

    return true;
}

bool
VanetzaGeoNetAdapter::Receive(
    PacketContext& context,
    const std::vector<uint8_t>& buffer,
    const Mac48Address& senderMac)
{
    if (!IsInitialized() ||
        !m_impl->runtime ||
        !m_impl->router)
    {
        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: GeoNetworking adapter not initialized");

        return false;
    }

    if (buffer.empty())
    {
        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: empty GeoNetworking frame");

        return false;
    }

    try
    {
        //=====================================================================
        // Advance Vanetza's manual runtime to the current ns-3 time.
        //=====================================================================

        constexpr int64_t VANETZA_EPOCH_MS =
            1072915200000LL;

        const int64_t currentTimeMs =
            VANETZA_EPOCH_MS +
            Simulator::Now().GetMilliSeconds();

        const auto now =
            vanetza::Clock::time_point(
                std::chrono::milliseconds(
                    currentTimeMs));

        m_impl->runtime->trigger(now);

        //=====================================================================
        // Keep the Router position synchronized with the ns-3 vehicle.
        //=====================================================================

        if (m_impl->position)
        {
            m_impl->router->update_position(
                m_impl->position->position_fix());
        }

        //=====================================================================
        // Convert the received ns-3 frame into Vanetza's network packet.
        //=====================================================================

        vanetza::ByteBuffer bytes(
            buffer.begin(),
            buffer.end());

        auto packet =
            std::make_unique<vanetza::UpPacket>(
                vanetza::CohesivePacket(
                    std::move(bytes),
                    vanetza::OsiLayer::Network));

        //=====================================================================
        // Convert the ns-3 Link MAC into Vanetza's link-layer MAC.
        //=====================================================================

        uint8_t macBytes[vanetza::MacAddress::length_bytes] = {};
        senderMac.CopyTo(macBytes);

        vanetza::MacAddress sender;
        for (std::size_t i = 0;
             i < vanetza::MacAddress::length_bytes;
             ++i)
        {
            sender.octets[i] = macBytes[i];
        }

        //=====================================================================
        // Preserve the single PacketContext through Router -> BTP.
        //
        // The BTP indication handler retrieves this exact context through
        // RxBridgeContext::Current().
        //=====================================================================

        RxBridgeContext::Push(&context);

        try
        {
            m_impl->router->indicate(
                std::move(packet),
                sender,
                vanetza::cBroadcastMacAddress);
        }
        catch (...)
        {
            RxBridgeContext::Pop();
            throw;
        }

        RxBridgeContext::Pop();

        NS_LOG_UNCOND(
            "[GEO_RX] Router indicate accepted frame"
            << " bytes="
            << buffer.size()
            << " sender="
            << sender);

        return true;
    }
    catch (const std::exception& e)
    {
        RxBridgeContext::Pop();

        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: Router receive failed: "
            << e.what());

        return false;
    }
    catch (...)
    {
        RxBridgeContext::Pop();

        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: Router receive failed with unknown exception");

        return false;
    }
}

VanetzaGeoNetAdapter::SendBtpB(

void
VanetzaGeoNetAdapter::SetBtpRxHandler(
    uint16_t destinationPort,
    BtpRxHandler handler)
{
    if (!IsInitialized() ||
        !m_impl->btpAdapter)
    {
        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: BTP adapter not initialized");

        return;
    }

    if (!handler)
    {
        NS_LOG_UNCOND(
            "[GEO_RX] ERROR: null BTP RX handler for port="
            << destinationPort);

        return;
    }

    m_impl->btpAdapter->SetRxHandler(
        destinationPort,
        [handler = std::move(handler)](
            const VanetzaBtpAdapter::RxData& rx)
        {
            BtpRxData data;

            data.destinationPort =
                rx.destinationPort;

            data.payload =
                rx.payload;

            data.hasSourcePosition =
                rx.hasSourcePosition;

            data.latitude =
                rx.latitude;

            data.longitude =
                rx.longitude;

            data.hasTrafficClass =
                rx.hasTrafficClass;

            data.trafficClass =
                rx.trafficClass;

            data.hasItsAid =
                rx.hasItsAid;

            data.itsAid =
                rx.itsAid;

            data.hasPermissions =
                rx.hasPermissions;

            data.permissionsSize =
                rx.permissionsSize;

            data.hasLifetime =
                rx.hasLifetime;

            data.lifetimeMs =
                rx.lifetimeMs;

            data.hasSourceMid =
                rx.hasSourceMid;

            data.sourceMid =
                rx.sourceMid;

            handler(data);
        });

    NS_LOG_UNCOND(
        "[GEO_RX] BTP RX handler forwarded to adapter"
        << " port="
        << destinationPort);
}

        vehId,
        table,
        step);

void NeighborManager::PurgeOld(
    std::map<uint32_t, NeighborInfo>& table,
    uint64_t currentStep,
    uint64_t maxAge)
{
    for (auto it = table.begin(); it != table.end(); )
    {
        uint64_t age = currentStep - it->second.lastUpdateStep;

        if (age > maxAge)
        {
            NeighborLogger::LogPurge(
                std::to_string(it->first),
                it->first,
                age,
                currentStep);

            it = table.erase(it);
        }
        else
        {
            ++it;
        }
    }
}

void VehicleMobilityUpdater::Apply(VehicleState& state){

    if (!state.node){
        return;
    }

    auto mobility = state.node->GetObject<ConstantVelocityMobilityModel>();

    if (!mobility){
        return;
    }

    mobility->SetPosition(state.mobility.position);

    const double headingRad = (90.0 - state.mobility.heading) * M_PI / 180.0;

    const double speed = state.mobility.speed;

    mobility->SetVelocity(Vector(speed * std::cos(headingRad), speed * std::sin(headingRad),0.0));

// ============================================================================
// PacketIdManager
//
// Responsável pela geração de identificadores únicos dos pacotes V2X.
//
// Mantém três espaços de numeração independentes:
//
//  • packetId
//      Identificador global da simulação.
//
//  • vehiclePacketId
//      Contador monotónico por veículo.
//
//  • sequenceId
//      Contador monotónico por veículo e por tipo de mensagem.
//
// Exemplo:
//
// veh0
//      CAM
//          packetId        = 154
//          vehiclePacketId = 18
//          sequenceId      = 7
//
// veh1
//      CAM
//          packetId        = 155
//          vehiclePacketId = 7
//          sequenceId      = 7
//
// veh0
//      DENM
//          packetId        = 156
//          vehiclePacketId = 19
//          sequenceId      = 1
//
// ============================================================================

struct PayloadTypeHash {
    std::size_t operator()(PayloadType type) const noexcept {
        return static_cast<std::size_t>(type);
    }
};

class PacketIdManager {

public:

    static PacketIdManager& Instance();

    uint64_t NextPacketId();

    uint32_t NextVehiclePacketId(const std::string& vehId);

    uint16_t NextSequenceId(const std::string& vehId, PayloadType payloadType);

    void Reset();

private:

    using SequenceTable = std::unordered_map<PayloadType, uint16_t, PayloadTypeHash>;

    PacketIdManager() = default;

    //-------------------------------------------------------------------------
    // Global packet counter
    //-------------------------------------------------------------------------

    uint64_t m_nextPacketId = 1;

    //-------------------------------------------------------------------------
    // Packet counter per vehicle
    //-------------------------------------------------------------------------

    std::unordered_map<std::string, uint32_t> m_vehiclePacketCounters;

    //-------------------------------------------------------------------------
    // Sequence counter per vehicle and payload type
    //-------------------------------------------------------------------------

        << std::hex
        << std::setfill('0');

        std::min(maxBytes, bytes.size());

        oss << std::setw(2)
            << static_cast<unsigned>(bytes[i]);

        oss << "...";
    }

    return oss.str();

// ============================================================================
// PacketContext
//
// Contexto único associado a um pacote V2X.
//
// O PacketContext acompanha um pacote durante todo o seu ciclo de vida,
// constituindo a fonte única de informação para:
//
//   • logging
//   • tracing
//   • profiling
//   • auditoria
//   • correlação entre módulos
//
// Fluxo TX:
//
//   GENERATE -> ENCODE -> SIGN -> BTP_TX -> GEONET_TX -> ACCESS_TX -> SEND -> TX_COMPLETE
//
// Fluxo RX:
//
//   RAW_RX -> GEONET_RX -> BTP_RX -> VERIFY -> DECODE -> REPLAY -> RECEPTION -> DELIVER
//
// Durante TX:
//
//   sender == receiver
//
// Durante RX:
//
//   sender   = veículo que originou o pacote
//   receiver = veículo que recebeu o pacote
// ============================================================================

struct PacketContext
{
    //=========================================================================
    // Simulation context
    //=========================================================================

    int64_t simulationStep = 0;
    double simulationTime = 0.0;

    // Timestamp temporal imutável do PacketContext,
    // expresso em microssegundos do relógio de simulação ns-3.
    uint64_t timestamp = 0;

    //=========================================================================
    // Sender identity
    //=========================================================================

    std::string senderVehId;

    uint32_t senderNodeId = 0;
    uint32_t senderStationId = 0;

    //=========================================================================
    // Receiver identity
    //=========================================================================

    std::string receiverVehId;

    uint32_t receiverNodeId = 0;
    uint32_t receiverStationId = 0;

    //=========================================================================
    // Packet identity
    //=========================================================================
    //
    // Estes identificadores pertencem sempre ao emissor.
    //=========================================================================

    uint64_t packetId = 0;
    uint32_t vehiclePacketId = 0;
    uint16_t sequenceId = 0;

    //=========================================================================
    // Pipeline state
    //=========================================================================
    //
    // pipelineStep e pipelineTotal são calculados automaticamente a partir
    // da etapa (PipelineStage).
    //=========================================================================

    uint16_t pipelineStep = 0;
    uint16_t pipelineTotal = 0;

    PipelineStage pipelineStage = PipelineStage::NONE;
    PipelineResult pipelineResult = PipelineResult::NONE;

    //=========================================================================
    // Execution metadata
    //=========================================================================

    std::string reason;

    uint64_t durationUs = 0;

    //=========================================================================
    // Payload
    //=========================================================================

    PacketPayload payload;

    //=========================================================================
    // BTP RX context
    //=========================================================================

    uint16_t btpDestinationPort = 0;

    //=========================================================================
    // GeoNetworking context
    //=========================================================================
    //
    // Informação original proveniente da camada GeoNetworking.
    // É preenchida na receção e acompanha o pacote durante toda a RX pipeline.
    //=========================================================================

    // Pipeline control
    //=========================================================================

    /*
     * Avança para uma nova etapa da pipeline.

     *
     *   • pipelineStep
     *   • pipelineTotal
     *   • pipelineResult = BEGIN

    void AdvancePipeline(PipelineStage stage);

    //=========================================================================

    void MarkSuccess();

    void MarkFailure();

    void MarkDrop();

    void MarkResult(PipelineResult result);

    //=========================================================================
    // Execution helpers
    //=========================================================================

    /*

     * preservando a identidade do pacote.
     */
    void ClearExecutionState();

    //=========================================================================

    bool IsValid() const;

    bool IsEmpty() const;

#include "vanetza_integration/core/v2x_log.h"

namespace {

    inline int16_t CompareGenerationDeltaTime(uint16_t current, uint16_t previous) {
        return static_cast<int16_t>(current - previous);
    }

} // anonymous namespace

namespace ns3 {

    bool ReplayProtection::Process(PacketContext& ctx, const CamMessage& message, ReplayContext& replay) {
        if (IsReplay(replay, message)) {
            ctx.MarkDrop();
            ctx.reason = "replay_detected";
            V2XWriteLog(ctx, V2XLogLevel::WARN, "REPLAY", "REPLAY_DROP",
                        "station=" + std::to_string(message.stationId) +
                        " gdt=" + std::to_string(message.generationDeltaTime));
            return false;
        }

        Update(replay, message);
        return true;
    }

    bool ReplayProtection::IsReplay(const ReplayContext& replay, const CamMessage& message) {
        auto it = replay.database.find(message.stationId);
        if (it == replay.database.end()) {
            return false;
        }

        const ReplayEntry& entry = it->second;
        const int16_t diff = CompareGenerationDeltaTime(message.generationDeltaTime, entry.lastGenerationDeltaTime);

        if (diff > 0) {
            return false;
        }

        if (diff == 0) {
            return message.timestamp <= entry.lastSequenceId;
        }

        return true;
    }

    void ReplayProtection::Update(ReplayContext& replay, const CamMessage& message) {
        ReplayEntry& entry = replay.database[message.stationId];
        entry.lastGenerationDeltaTime = message.generationDeltaTime;
        entry.lastSequenceId = message.timestamp;
    }

} // namespace ns3

    VehicleState* VehicleBuilder::Create(const std::string& vehId, Ptr<Node> node, Ptr<Socket> socket) {
        
        VehicleState* state = InitializeVehicleState(vehId, node, socket);

        if (!state){
            return nullptr;
        }

        //----------------------------------------------------------------------
        // STEP 1: authoritative vehicle identity
        //
        // VehicleIdentityInitializer is the sole authority for:
        //   - nodeId
        //   - stationId
        //
        // vehId originates from SUMO / VehicleBuilder.
        //
        // Identity MUST be complete before VanetzaRuntime is created.
        //----------------------------------------------------------------------
        VehicleIdentityInitializer::Initialize(*state);

        if (state->identity.stationId == 0) {
            NS_LOG_UNCOND("[VEHICLE_BUILDER] ERROR: invalid stationId=0 for veh=" << state->identity.vehId);
            return nullptr;
        }

        //----------------------------------------------------------------------
        // STEP 2: security identity/context
        //----------------------------------------------------------------------
        VehicleSecurityInitializer::Initialize(*state);

        //----------------------------------------------------------------------
        // STEP 3: Vanetza Runtime
        //
        // At this point VehicleIdentity is definitive.
        //
        // Therefore VanetzaRuntime::Initialize() receives:
        //   vehId     = authoritative VehicleIdentity.vehId
        //   nodeId    = authoritative VehicleIdentity.nodeId
        //   stationId = authoritative VehicleIdentity.stationId
        //
        // GN_ADDR is materialized from this definitive stationId.
        //----------------------------------------------------------------------
        if (!state->runtime) {

            state->runtime = std::make_unique<VanetzaRuntime>(*state);

            if (!state->runtime->Initialize()){

                state->runtime.reset();
                return nullptr;
            }
        }

        LogVehicleCreation(*state);

        return state;
    }

    VehicleState* VehicleBuilder::InitializeVehicleState(const std::string& vehId, Ptr<Node> node, Ptr<Socket> socket) {

        auto& registry = VehicleRegistry::Instance();

        VehicleState* state = registry.CreateOrGet(SimulationContext::simulationStep, vehId);

        if (!state){
            return nullptr;
        }

        state->node = node;
        state->socket = socket;
        state->identity.vehId = vehId;

        //----------------------------------------------------------------------
        // IMPORTANT:
        //
        // Do NOT create VanetzaRuntime here.
        //
        // VehicleIdentityInitializer::Initialize() must run first so that
        // nodeId and stationId are authoritative before VanetzaRuntime
        // materializes GN_ADDR and creates the Router.
        //----------------------------------------------------------------------

        return state;
    }

    // Preconditions

    // Register reverse indexes
    //
    // VehicleRegistry is an index service only.
    //======================================================================

        VehicleRegistry::Instance();

    for (std::size_t i = 0; i < mid.size(); ++i)
    {
        if (i != 0)
        {
            oss << ":";
        }

        oss << std::hex
            << std::setw(2)
            << std::setfill('0')
            << static_cast<unsigned>(mid[i]);
    }

    return oss.str();
}

{
    static VehicleRegistry instance;
    return instance;

VehicleState*
VehicleRegistry::CreateOrGet(
    int64_t step,
    const std::string& vehId)
{
    (void)step;

    auto it = vehiclesById.find(vehId);

    if (it != vehiclesById.end()) {
        return it->second.get();
    }

    auto vs = std::make_unique<VehicleState>();

    /*
     * VehicleRegistry é responsável apenas pela gestão e indexação
     * dos VehicleState.
     *
     * A autoridade sobre a identidade pertence ao
     * VehicleIdentityInitializer.
     *
     * Neste ponto apenas o vehId é conhecido, pois é o identificador
     * proveniente do SUMO / VehicleBuilder.
     */
    vs->identity.vehId = vehId;
    vs->initialized = false;

    VehicleState* ptr = vs.get();

    vehiclesById[vehId] = std::move(vs);

    return ptr;
}

    auto it = vehiclesById.find(vehId);

    return it->second.get();
}

void
VehicleRegistry::RegisterNodeId(
    int64_t step,
    const std::string& vehId,
    uint32_t nodeId)
{
    (void)step;

    VehicleState* vs = GetByVehId(vehId);

    NS_ABORT_MSG_IF(
        vs == nullptr,
        "[REGISTRY] RegisterNodeId: vehicle not found: "
            << vehId);

    /*
     * VehicleIdentityInitializer é a autoridade sobre nodeId.
     * O Registry apenas verifica a consistência e cria o índice reverso.
     */
    NS_ABORT_MSG_IF(
        vs->identity.nodeId != nodeId,
        "[REGISTRY] nodeId mismatch for vehicle "
            << vehId
            << ": identity.nodeId="
            << vs->identity.nodeId
            << " requested="
            << nodeId);

    auto existing = nodeToVeh.find(nodeId);

    NS_ABORT_MSG_IF(
        existing != nodeToVeh.end()
            && existing->second != vehId,
        "[REGISTRY] nodeId collision: nodeId="
            << nodeId
            << " already belongs to vehicle "
            << existing->second
            << ", cannot register for "
            << vehId);

    nodeToVeh[nodeId] = vehId;
}

// ============================================================================
// Bind Link MAC to permanent VehicleState
// ============================================================================

void
VehicleNetworkBuilder::BindLinkMac(
    VehicleState& vehicle)
{
    NS_ABORT_MSG_IF(
        vehicle.node == nullptr,
        "[NET] Cannot bind Link MAC: vehicle node is null");

    const uint32_t deviceCount =
        vehicle.node->GetNDevices();

    NS_ABORT_MSG_IF(
        deviceCount == 0,
        "[NET] Cannot bind Link MAC: vehicle has no NetDevice");

    bool foundWifi = false;

    for (uint32_t i = 0; i < deviceCount; ++i)
    {
        Ptr<NetDevice> device =
            vehicle.node->GetDevice(i);

        if (device == nullptr)
        {
            continue;
        }

        Ptr<WifiNetDevice> wifiDevice =
            DynamicCast<WifiNetDevice>(device);

        if (wifiDevice == nullptr)
        {
            continue;
        }

        const Address address =
            wifiDevice->GetAddress();

        NS_ABORT_MSG_IF(
            !Mac48Address::IsMatchingType(address),
            "[NET] WifiNetDevice does not expose Mac48Address");

        vehicle.linkMac =
            Mac48Address::ConvertFrom(address);

        vehicle.linkMacInitialized = true;

        Ptr<Ipv4> ipv4 =
            vehicle.node->GetObject<Ipv4>();

        NS_ABORT_MSG_IF(
            ipv4 == nullptr,
            "[NET] Vehicle node has no Ipv4 object");

        const int32_t interfaceIndex =
            ipv4->GetInterfaceForDevice(
                wifiDevice);

        NS_ABORT_MSG_IF(
            interfaceIndex < 0,
            "[NET] Could not resolve IPv4 interface for WifiNetDevice");

        NS_ABORT_MSG_IF(
            ipv4->GetNAddresses(interfaceIndex) == 0,
            "[NET] IPv4 interface has no address");

        vehicle.ipv4Address =
            ipv4->GetAddress(
                interfaceIndex,
                0).GetLocal();

        vehicle.ipv4Initialized = true;

        NS_LOG_UNCOND(
            "[NET] node="
            << vehicle.identity.nodeId
            << " veh="
            << vehicle.identity.vehId
            << " wifiDeviceIndex="
            << i
            << " ip="
            << vehicle.ipv4Address
            << " linkMac="
            << vehicle.linkMac);

        VehicleRegistry::Instance().RegisterIpv4(
            vehicle.ipv4Address,
            vehicle);

        foundWifi = true;
        break;
    }

    NS_ABORT_MSG_IF(
        !foundWifi,
        "[NET] No WifiNetDevice found for vehicle "
            << vehicle.identity.vehId);
}


// ============================================================================
// Configure networking
// ============================================================================

void
VehicleNetworkBuilder::ConfigureNetworking(
    Ptr<Node> node)
{
    NS_ABORT_MSG_IF(
        node == nullptr,
        "[NET] Cannot configure networking: node is null");

    InternetStackHelper internet;
    internet.Install(node);

    YansWifiPhyHelper phy;
    WifiHelper wifi;
    WifiMacHelper mac;

    phy.SetChannel(m_channel);

    wifi.SetStandard(WIFI_STANDARD_80211p);

    wifi.SetRemoteStationManager(
        "ns3::ConstantRateWifiManager",
        "DataMode",
        StringValue("OfdmRate6MbpsBW10MHz"),
        "ControlMode",
        StringValue("OfdmRate6MbpsBW10MHz"));

    mac.SetType("ns3::AdhocWifiMac");



    NetDeviceContainer devices =
        wifi.Install(
            phy,
            mac,
            node);

    TrafficControlHelper tch;

    tch.SetRootQueueDisc(
        "ns3::PfifoFastQueueDisc");

    tch.Install(devices);


    for (uint32_t i = 0; i < devices.GetN(); ++i)
    {
        Ptr<NetDevice> device = devices.Get(i);

        NS_ABORT_MSG_IF(
            device == nullptr,
            "[NET] Null NetDevice at index "
                << i);

        const Address address =
            device->GetAddress();

        NS_LOG_UNCOND(
            "[NET_DEBUG] node="
            << node->GetId()
            << " device="
            << i
            << " type="
            << device->GetInstanceTypeId().GetName()
            << " address="
            << address);
    }



    /*
     * Assign IPv4 only once.
     */
    Ipv4InterfaceContainer interfaces =
        m_ipv4.Assign(devices);

    NS_LOG_UNCOND(
        "[NET] node="
        << node->GetId()
        << " ip="
        << interfaces.GetAddress(0));
}

// ============================================================================
// Create socket
// ============================================================================

Ptr<Socket>
VehicleNetworkBuilder::CreateSocket(
    Ptr<Node> node)
{
    NS_ABORT_MSG_IF(
        node == nullptr,
        "[NET] Cannot create socket: node is null");

    Ptr<Socket> socket =
        Socket::CreateSocket(
            node,
            UdpSocketFactory::GetTypeId());

    NS_ABORT_MSG_IF(
        socket == nullptr,
        "[NET] Failed to create UDP socket");

    socket->SetAllowBroadcast(true);

    const int result =
        socket->Bind(
            InetSocketAddress(
                Ipv4Address::GetAny(),
                5000));

    NS_ABORT_MSG_IF(
        result != 0,
        "[NET] Failed to bind UDP socket on port 5000");

    socket->SetRecvCallback(
        m_rxCallback);

    return socket;
}

//==========================================================================
// TxBridgeContext
//
// CORREÇÃO O1 (use-after-free):
//   A versão anterior guardava um ponteiro cru (PacketContext*) para um
//   objeto cujo dono é o CamGenerationContext local da stack em
//   CamScheduler::Run. Como a transmissão real para o canal é diferida
//   (Simulator::ScheduleNow na Ns3AccessAdapter), o lambda corria DEPOIS
//   de o CamGenerationContext — e portanto o PacketContext — já ter sido
//   destruído, lendo/escrevendo memória libertada.

//   camada Access captura este shared_ptr por valor, mantendo o objeto
//   vivo até TX_COMPLETE, independentemente do tempo de vida do
//   CamGenerationContext que o criou.
//==========================================================================
class TxBridgeContext {
public:
    static void Push(std::shared_ptr<PacketContext> context);
    static std::shared_ptr<PacketContext> Current();
    static void Pop();

private:
    static thread_local std::shared_ptr<PacketContext> m_context;
};

//==========================================================================
// RAII helper
//==========================================================================
class TxBridgeScope {
public:
    explicit TxBridgeScope(std::shared_ptr<PacketContext> context) {
        TxBridgeContext::Push(std::move(context));
    }

    ~TxBridgeScope() {
        TxBridgeContext::Pop();
    }

    TxBridgeScope(const TxBridgeScope&) = delete;
    TxBridgeScope& operator=(const TxBridgeScope&) = delete;
};

    int localPort,

{

        m_socketFd,
        m_localAddr,
        m_sumoAddr,

        sumoPort);
}

// ============================================================================
// Poll
// ============================================================================

void
SumoAdapter::Poll()
{
    if (m_socketFd < 0) {
        return;
    }

    char buffer[512];

    sockaddr_in sender{};

    socklen_t senderLen =
        sizeof(sender);

    while (true) {

        const int bytes =
            SumoSocket::Receive(
                m_socketFd,
                buffer,
                sizeof(buffer) - 1,
                sender,
                senderLen);

        if (bytes <= 0) {
            break;
        }

        buffer[bytes] = '\0';

        auto vehicle =
            ns3::SumoParser::Parse(buffer);

        if (!vehicle) {
            continue;
        }

        // ---------------------------------------------------------------------
        // SUMO state
        //
        // O parser produz um VehicleState temporário contendo apenas:
        //   - vehId
        //   - mobility
        //
        // O SumoAdapter não resolve nem modifica nodeId/stationId.
        // A resolução da identidade permanente é responsabilidade da camada
        // de sincronização/VehicleFactory.
        // ---------------------------------------------------------------------

        V2XWriteLog(
            SimulationContext::simulationStep,
            V2XLogLevel::DEBUG,
            "SUMO",
            "RX_VEHICLE_STATE",
            vehicle->identity.vehId,
            vehicle->identity.nodeId,
            vehicle->identity.stationId,
            -1,
            "pos=("
                + std::to_string(
                    vehicle->mobility.position.x)
                + ","
                + std::to_string(
                    vehicle->mobility.position.y)
                + ") speed="
                + std::to_string(
                    vehicle->mobility.speed));

        // ---------------------------------------------------------------------
        // Forward vehicle state
        //
        // O callback recebe o estado temporário produzido pelo SUMO.
        // A camada seguinte decide:
        //
        //   1. se o veículo já existe;
        //   2. se precisa ser criado;
        //   3. qual VehicleState permanente deve ser atualizado.
        //
        // Desta forma o SumoAdapter não possui autoridade sobre a identidade.
        // ---------------------------------------------------------------------

        if (m_vehicleCallback) {
            m_vehicleCallback(*vehicle);
        }
    }
}

// ============================================================================
// Send
// ============================================================================

void
SumoAdapter::SendToSumo(
    const ns3::VehicleState& vehicle)
{
    if (m_socketFd < 0) {
        return;
    }

    SumoSocket::Send(
        m_socketFd,
        m_sumoAddr,
        SumoSerializer::Serialize(vehicle));

    // TODO:
    // Quando a transmissão utilizar PacketContext desde o início
    // da pipeline, este log deverá migrar para a API baseada em
    // PacketContext.
}

// ============================================================================
// Callback
// ============================================================================

void
SumoAdapter::SetVehicleUpdateCallback(
    VehicleUpdateCallback callback)
{
    m_vehicleCallback =
        std::move(callback);
}

// ============================================================================
// Close
// ============================================================================

void
SumoAdapter::Close()
{
    SumoSocket::Close(m_socketFd);
}

=== ./vanetza_integration/sumo_adapter/sumo_socket.cc ===
#include "sumo_socket.h"

#include "vanetza_integration/context/simulation_context/simulation_context.h"
#include "vanetza_integration/core/v2x_log.h"

#include <arpa/inet.h>
#include <sys/socket.h>
#include <unistd.h>

#include <cstring>

bool SumoSocket::Init(int& socketFd,sockaddr_in& localAddr, sockaddr_in& sumoAddr, int localPort, const std::string& sumoIp, int sumoPort){

    // =====================================================
    // Create the UDP socket used by the ns-3 <-> SUMO bridge. The socket remains open for the entire lifetime of the adapter and is released by Close().
    // =====================================================

    socketFd = ::socket(AF_INET, SOCK_DGRAM, 0);

    if (socketFd < 0){

        V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::ERROR, "SUMO", "SOCKET_FAIL", "-", 0, 0, -1, "Unable to create UDP socket");

        return false;
    }

    // =====================================================
    // Configure the local endpoint. The adapter listens on the configured UDP port for messages arriving from the SUMO communication bridge.
    // =====================================================

    std::memset(&localAddr, 0, sizeof(sockaddr_in));

    localAddr.sin_family = AF_INET;
    localAddr.sin_port = htons(localPort);
    localAddr.sin_addr.s_addr = INADDR_ANY;

    if (::bind(socketFd, reinterpret_cast<sockaddr*>(&localAddr), sizeof(sockaddr_in)) < 0){

        V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::ERROR, "SUMO", "BIND_FAIL", "-", 0, 0, -1, "Unable to bind UDP socket");

        ::close(socketFd);
        socketFd = -1;

        return false;
    }

    // =====================================================
    // Configure the remote SUMO endpoint. inet_pton() validates the supplied IPv4 address before storing it into the destination socket structure.
    // =====================================================

    std::memset(&sumoAddr, 0, sizeof(sockaddr_in));

    sumoAddr.sin_family = AF_INET;
    sumoAddr.sin_port = htons(sumoPort);

    if (::inet_pton(AF_INET, sumoIp.c_str(), &sumoAddr.sin_addr) != 1){

        V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::ERROR, "SUMO", "INVALID_IP", "-", 0, 0, -1, "Invalid SUMO IPv4 address");

        ::close(socketFd);
        socketFd = -1;

        return false;
    }

    // =====================================================
    // The UDP communication channel has been successfully configured and is ready to exchange messages.
    // =====================================================

    V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::INFO, "SUMO", "INIT_OK", "-", 0, 0, -1, "UDP adapter initialized");

    return true;
}

int SumoSocket::Receive(int socketFd, char* buffer, std::size_t bufferSize, sockaddr_in& sender, socklen_t& senderLen){

    // =====================================================
    // Perform a non-blocking receive operation. MSG_DONTWAIT guarantees that the simulation loop never blocks waiting for external SUMO traffic.
    // =====================================================

    return ::recvfrom(socketFd, buffer, bufferSize, MSG_DONTWAIT, reinterpret_cast<sockaddr*>(&sender), &senderLen);
}

bool SumoSocket::Send(int socketFd, const sockaddr_in& destination, const std::string& message){

    // =====================================================
    // A negative descriptor indicates that the socket was never initialized or has already been closed.
    // =====================================================

    if (socketFd < 0)
        return false;

    const ssize_t bytes = ::sendto(socketFd, message.data(), message.size(), 0, reinterpret_cast<const sockaddr*>(&destination), sizeof(sockaddr_in));

    // =====================================================
    // sendto() should transmit the complete message in a single UDP datagram.
    // =====================================================

    return bytes == static_cast<ssize_t>(message.size());
}

void SumoSocket::Close(int& socketFd){

    // =====================================================
    // Release the UDP socket associated with the SUMO
    // communication adapter.
    //
    // Closing an already invalid descriptor is ignored,
    // making this operation safe to call multiple times.
    //
    // After the operating system resource is released, the
    // descriptor is reset to -1 to prevent accidental reuse.
    // =====================================================

    if (socketFd >= 0){

        ::close(socketFd);

        socketFd = -1;

        V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::INFO, "SUMO", "SOCKET_CLOSED", "-", 0, 0, -1, "UDP adapter socket closed");
    }
}

std::unique_ptr<VehicleState> SumoParser::Parse(const std::string& message){
    
    auto vehicle = std::make_unique<VehicleState>();

    std::stringstream ss(message);
    std::string token;

    try{

        if (!std::getline(ss, vehicle->identity.vehId, ','))
            return nullptr;

        if (!std::getline(ss, token, ','))
            return nullptr;
        vehicle->mobility.position.x = std::stod(token);

        if (!std::getline(ss, token, ','))
            return nullptr;
        vehicle->mobility.position.y = std::stod(token);

        if (!std::getline(ss, token, ','))
            return nullptr;
        vehicle->mobility.speed = std::stod(token);

        if (!std::getline(ss, token, ','))
            return nullptr;
        vehicle->mobility.heading = std::stod(token);

        if (!std::getline(ss, token, ','))
            return nullptr;
        vehicle->mobility.accel = std::stod(token);

        vehicle->initialized = true;
    }
    catch (...)
    {
        //V2XWriteLog(SimulationContext::simulationStep, V2XLogLevel::WARN, "SUMO", "PARSE_FAIL", "-", 0, 0, -1, message);

        return nullptr;
    }

    return vehicle;
}

#pragma once

#include <cstddef>
#include <string>

#include <netinet/in.h>

// ============================================================================
// SumoSocket
//
// Thin wrapper around the operating system UDP socket API.
//
// This module is responsible only for the socket lifecycle and low-level UDP
// communication between ns-3 and the external SUMO bridge.
//
// Design principles:
//   - Stateless implementation.
//   - No ownership of simulation objects.
//   - No knowledge of CAM, DENM or ETSI messages.
//   - No packet processing.
//   - No business logic.
//
// Higher-level modules are responsible for message serialization,
// deserialization and protocol handling.
// ============================================================================

class SumoSocket{

public:

    // ========================================================================
    // Socket lifecycle
    //
    // Creates, configures and releases the UDP socket used by the SUMO
    // communication adapter.
    // ========================================================================

    static bool Init(int& socketFd, sockaddr_in& localAddr, sockaddr_in& sumoAddr, int localPort, const std::string& sumoIp, int sumoPort);

    static void Close(int& socketFd);

    // ========================================================================
    // UDP Input / Output
    //
    // Thin wrappers around the operating system UDP API.
    //
    // Receive() performs a non-blocking read.
    // Send() transmits a complete UDP datagram.
    // ========================================================================

    static int Receive(int socketFd, char* buffer, std::size_t bufferSize, sockaddr_in& sender, socklen_t& senderLen);

    static bool Send(int socketFd, const sockaddr_in& destination, const std::string& message);
};
