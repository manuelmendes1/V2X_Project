std::pair<EVP_PKEY*, X509*> GenerateEcdsaKeyAndCert(const std::string& vehId) {
    EVP_PKEY* pkey = EVP_PKEY_new();
    EC_KEY* eckey = EC_KEY_new_by_curve_name(NID_X9_62_prime256v1);
    EC_KEY_generate_key(eckey);
    EVP_PKEY_assign_EC_KEY(pkey, eckey);
    X509* cert = X509_new();
    ASN1_INTEGER_set(X509_get_serialNumber(cert), std::rand());
    X509_gmtime_adj(X509_get_notBefore(cert), 0);
    X509_gmtime_adj(X509_get_notAfter(cert), 365 * 24 * 3600);
    X509_set_pubkey(cert, pkey);
    X509_NAME* name = X509_get_subject_name(cert);
    X509_NAME_add_entry_by_txt(name, "CN", MBSTRING_ASC, (unsigned char*)vehId.c_str(), -1, -1, 0);
    X509_set_issuer_name(cert, name);
    X509_sign(cert, pkey, EVP_sha256());
    return {pkey, cert};
}

static void RxCallback(Ptr<Socket> socket) {
    Ptr<Packet> packet;
    Address from;
    uint32_t nodeId = socket->GetNode()->GetId();
    
    // CORREÇÃO: Acessar via ponteiro e validar existência
    auto& reg = VehicleRegistry::Instance();
    VehicleState* state = reg.GetByNodeId(nodeId);
    std::string receiverVehId = (state) ? state->identity.vehId : "UNKNOWN";

    while ((packet = socket->RecvFrom(from))) {
        std::vector<uint8_t> buffer(packet->GetSize());
        packet->CopyData(buffer.data(), buffer.size());
        
        InetSocketAddress inetFrom = InetSocketAddress::ConvertFrom(from);
        Ipv4Address srcIp = inetFrom.GetIpv4();
        std::stringstream ss;
        srcIp.Print(ss);
        
        int64_t packetSeqId = -1; 
        if (buffer.size() >= sizeof(int64_t)) {
            std::memcpy(&packetSeqId, buffer.data(), sizeof(int64_t));
        }
        
        std::string rxMeta = "size=" + std::to_string(packet->GetSize()) + " src_ip=" + ss.str();
        V2XWriteLog(V2XLogLevel::INFO, "CAM_RX", "RAW_IN", receiverVehId, nodeId, packetSeqId, rxMeta);
        RxPipeline::Process(socket, buffer);
    }
}


Ptr<Node> CreateVehicleNode(const std::string& vehId) {
    Ptr<Node> node = CreateObject<Node>();
    MobilityHelper mobility;
    mobility.SetMobilityModel("ns3::ConstantVelocityMobilityModel");
    mobility.Install(node);
    InternetStackHelper internet;
    internet.Install(node);
    YansWifiPhyHelper phy;
    WifiHelper wifi;
    WifiMacHelper mac;
    phy.SetChannel(channel);
    wifi.SetStandard(WIFI_STANDARD_80211p);
    wifi.SetRemoteStationManager("ns3::ConstantRateWifiManager", "DataMode", StringValue("OfdmRate6MbpsBW10MHz"), "ControlMode", StringValue("OfdmRate6MbpsBW10MHz"));
    mac.SetType("ns3::AdhocWifiMac");
    NetDeviceContainer devices = wifi.Install(phy, mac, node);
    TrafficControlHelper tch;
    tch.SetRootQueueDisc("ns3::PfifoFastQueueDisc");
    tch.Install(devices);
    ipv4.Assign(devices);
    Ptr<Socket> socket = Socket::CreateSocket(node, UdpSocketFactory::GetTypeId());
    socket->SetAllowBroadcast(true);
    socket->Bind(InetSocketAddress(Ipv4Address::GetAny(), 5000));
    socket->SetRecvCallback(MakeCallback(&RxCallback));
    auto& reg = VehicleRegistry::Instance();
    VehicleState* vs = reg.CreateOrGet(vehId);
    vs->node = node;
    vs->socket = socket;
    vs->identity.vehId = vehId;
    vs->identity.nodeId = node->GetId();
    vs->identity.stationId = static_cast<uint32_t>(std::hash<std::string>{}(vehId) % 100000);
    reg.RegisterNodeId(vehId, node->GetId());
    reg.RegisterStationId(vehId, vs->identity.stationId);
    auto [key, cert] = GenerateEcdsaKeyAndCert(vehId);
    vs->privateKey = key;
    vs->certificate = cert;
    V2XWriteLog(V2XLogLevel::INFO, "CAM_TX", "INIT", vehId, node->GetId(), -1, "stationID=" + std::to_string(vs->identity.stationId));
    return node;
}

void HandleVehicleUpdateFromSumo(const VehicleState& parsedVs) {
    auto& reg = VehicleRegistry::Instance();
    if (reg.GetAllVehicles().find(parsedVs.identity.vehId) == reg.GetAllVehicles().end()) {
        CreateVehicleNode(parsedVs.identity.vehId);
    }
    VehicleState* vs = reg.CreateOrGet(parsedVs.identity.vehId);
    if (vs) {
        vs->position = parsedVs.position;
        vs->speed = parsedVs.speed;
        vs->heading = parsedVs.heading;
        vs->accel = parsedVs.accel;
        vs->initialized = true;
        if (vs->node) {
            auto mob = vs->node->GetObject<ConstantVelocityMobilityModel>();
            if (mob) {
                mob->SetPosition(vs->position);
                double rad = (90.0 - vs->heading) * M_PI / 180.0;
                mob->SetVelocity(Vector(vs->speed * cos(rad), vs->speed * sin(rad), 0));
            }
        }
    }
    sumoStepReceived = true;
}

void SendCam() {
    double now = Simulator::Now().GetSeconds();
    auto& reg = VehicleRegistry::Instance();
    for (auto& [vid, vsPtr] : reg.GetAllVehicles()) {
        VehicleState& vs = *vsPtr;
        if (!vs.initialized || !vs.socket) continue;
        double dt = (vs.lastCamTime < 0.0) ? 999.0 : (now - vs.lastCamTime);
        if (dt < 0.1) continue;
        auto data = CamService::GenerateCAM(vs);
        if (data.empty()) continue;
        Ptr<Packet> p = Create<Packet>(data.data(), data.size());
        Simulator::Schedule(MilliSeconds(1), [socket = vs.socket, p, vid, nid = vs.identity.nodeId, pos = vs.position]() {
            socket->SendTo(p, 0, InetSocketAddress("10.0.0.255", 5000));
            std::string meta = "pos_x=" + std::to_string(pos.x) + " pos_y=" + std::to_string(pos.y);
            V2XWriteLog(V2XLogLevel::INFO, "CAM_TX", "sent_to_channel", vid, nid, 0, meta);
        });
        vs.lastCamTime = now;
    }
}

void SimulationStep() {
    if (!sumoStepReceived) return;
    SendCam();
    auto& reg = VehicleRegistry::Instance();
    for (auto& [vid, vsPtr] : reg.GetAllVehicles()) {
        if (vsPtr->node) SumoAdapter::SendToSumo(*vsPtr);
    }
    sumoStepReceived = false;
}

void SumoPollLoop() {
    SumoAdapter::Poll();
    if (sumoStepReceived) {
        SimulationStep();
        SimulationContext::simulationStep++;
    }
    Simulator::Schedule(STEP_LENGTH, &SumoPollLoop);
}

int main(int argc, char* argv[]) {
    YansWifiChannelHelper ch;
    ch.SetPropagationDelay("ns3::ConstantSpeedPropagationDelayModel");
    ch.AddPropagationLoss("ns3::LogDistancePropagationLossModel", "Exponent", DoubleValue(2.7), "ReferenceDistance", DoubleValue(1.0));
    channel = ch.Create();
    ipv4.SetBase("10.0.0.0", "255.255.255.0");
    SumoAdapter::Init(7400, "127.0.0.1", 7000);
    SumoAdapter::SetVehicleUpdateCallback(HandleVehicleUpdateFromSumo);
    Simulator::ScheduleNow(&SumoPollLoop);
    Simulator::Run();
    Simulator::Destroy();
    SumoAdapter::Close();
    return 0;
}

// ======================================================
// CAM POSITION ENCODING
// ======================================================
// A rede SUMO atual não tem georreferenciação real
// (projParameter="!" em simple.net.xml), pelo que as posições
// vindas do SUMO/TraCI são coordenadas cartesianas locais (metros),
// não latitude/longitude WGS84 reais.
//
// Até existir georreferenciação real (fase futura), mapeamos essas
// coordenadas locais diretamente para os campos ASN.1
// referencePosition.latitude/longitude do CAM, usando uma escala fixa
// que mantém os valores dentro do intervalo válido (constrangido) do
// tipo Latitude/Longitude do ETSI ITS CDD (~±900000000 / ±1800000000),
// evitando overflow / encoding inválido.
//
// IMPORTANTE: este valor TEM de ser idêntico no encoder e no decoder.
// Está centralizado aqui exatamente para evitar a dessincronização
// que já causou bugs anteriormente.
//
// Margem atual: rede até ~500m por eixo -> 500 * 1e6 = 5e8,
// confortavelmente dentro do limite. Se o cenário crescer para
// redes maiores (>900m num eixo), este valor terá de ser revisto.
constexpr double kCamPositionScale = 1e6;

CamMessage CamDecoder::Decode(
    const std::vector<uint8_t>& payload,
    const std::string& receiverVehId,
    uint64_t receiverNodeId,
    int64_t seqId)
{
    if (payload.empty())
    {
        V2XWriteLog(V2XLogLevel::WARN, "CAM_RX", "DECODE_FAIL_EMPTY", receiverVehId, receiverNodeId, seqId, 
                    "Cannot decode an empty payload buffer");
        return {};
    }

    CAM_t* cam = nullptr;

    // Descodificação ASN.1 UPER nativa da Vanetza
    asn_dec_rval_t rval = uper_decode_complete(
        nullptr,
        &asn_DEF_CAM,
        (void**)&cam,
        payload.data(),
        payload.size());

    if (rval.code != RC_OK || !cam)
    {
        if (cam) {
            ASN_STRUCT_FREE(asn_DEF_CAM, cam);
        }
        V2XWriteLog(V2XLogLevel::ERROR, "CAM_RX", "DECODE_FAIL_ASN1", receiverVehId, receiverNodeId, seqId, 
                    "ASN.1 UPER syntax violation or corruption in incoming stream");
        return {};
    }

    CamMessage msg{};
    
    // Extração e desnormalização dos campos segundo especificações ETSI
    msg.stationId = cam->header.stationID;

    auto& pos = cam->cam.camParameters.basicContainer.referencePosition;
    msg.latitude  = pos.latitude  / kCamPositionScale;
    msg.longitude = pos.longitude / kCamPositionScale;

    auto& hf = cam->cam.camParameters.highFrequencyContainer.choice.basicVehicleContainerHighFrequency;
    msg.speed   = hf.speed.speedValue / 100.0;
    msg.heading = hf.heading.headingValue / 10.0;
    
    msg.timestamp = SimulationContext::simulationStep;

    // Log em total conformidade com o Logging Standard v1.0
    V2XWriteLog(
        V2XLogLevel::INFO,
        "CAM_RX",
        "DECODE",
        receiverVehId,
        receiverNodeId,
        seqId,
        "stationId=" + std::to_string(msg.stationId) +
        " pos=(" + std::to_string(msg.longitude) + "," + std::to_string(msg.latitude) + ")" +
        " speed=" + std::to_string(msg.speed) +
        " heading=" + std::to_string(msg.heading)
    );

    ASN_STRUCT_FREE(asn_DEF_CAM, cam);
    return msg;
}

    struct AsnDeleter {

            if (ptr) {
                ASN_STRUCT_FREE(asn_DEF_CAM, ptr);
            }
        }
    };
    using UniqueCamPtr = std::unique_ptr<CAM_t, AsnDeleter>;

    constexpr double kPositionThreshold = 0.0001;
    constexpr size_t kMaxBufferSize = 2048;
}

bool CamEncoder::Validate(const CamInput& input)
{
    return !(std::abs(input.latitude) < kPositionThreshold && 
             std::abs(input.longitude) < kPositionThreshold);
}

std::vector<uint8_t> CamEncoder::Encode(
    const CamInput& v,
    const std::string& vehicleId,
    uint64_t nodeId,
    int64_t seqId)
{
    // 1. Defensiva: Validação estrutural de coordenadas reais
    if (!Validate(v)) {
        V2XWriteLog(V2XLogLevel::WARN, "CAM", "ENCODE_ABORT_INVALID_POS", 
                    vehicleId, nodeId, seqId, 
                    "Skipping generation: Coordinates are too close to reference origin (0,0)");
        return {};
    }

    // 2. Alocação RAII limpa contra fugas de memória
    UniqueCamPtr cam(static_cast<CAM_t*>(calloc(1, sizeof(CAM_t))));
    if (!cam) {
        V2XWriteLog(V2XLogLevel::ERROR, "CAM", "ENCODE_FAIL_ALLOC", 
                    vehicleId, nodeId, seqId, "System calloc failed to provision memory for CAM_t");
        return {};
    }

    // 3. Preenchimento do ASN.1 conforme especificações ETSI
    cam->header.protocolVersion = 1;
    cam->header.messageID = 2; // CAM ID
    cam->header.stationID = v.stationId;
    cam->cam.generationDeltaTime = 0;

    // Basic Container (Estático)
    cam->cam.camParameters.basicContainer.stationType = StationType_passengerCar;
    cam->cam.camParameters.basicContainer.referencePosition.latitude =
        static_cast<long>(std::llround(v.latitude * kCamPositionScale));
    cam->cam.camParameters.basicContainer.referencePosition.longitude =
        static_cast<long>(std::llround(v.longitude * kCamPositionScale));

    // High Frequency Container (Dinâmico)
    cam->cam.camParameters.highFrequencyContainer.present =
        HighFrequencyContainer_PR_basicVehicleContainerHighFrequency;

    auto& hf = cam->cam.camParameters.highFrequencyContainer.choice.basicVehicleContainerHighFrequency;
    hf.speed.speedValue = static_cast<long>(v.speed * 100);
    hf.speed.speedConfidence = SpeedConfidence_unavailable;
    hf.heading.headingValue = static_cast<long>(v.heading * 10);
    hf.heading.headingConfidence = HeadingConfidence_unavailable;
    hf.driveDirection = DriveDirection_forward;
    hf.longitudinalAcceleration.longitudinalAccelerationValue = static_cast<long>(v.acceleration * 100);
    hf.longitudinalAcceleration.longitudinalAccelerationConfidence = 1;
    hf.curvature.curvatureValue = 0;
    hf.curvature.curvatureConfidence = CurvatureConfidence_onePerMeter_0_00002;
    hf.yawRate.yawRateValue = 0;
    hf.yawRate.yawRateConfidence = 1;
    hf.vehicleLength.vehicleLengthValue = 50;
    hf.vehicleWidth = 20;
    hf.vehicleLength.vehicleLengthConfidenceIndication = VehicleLengthConfidenceIndication_noTrailerPresent;

    // 4. Codificação UPER para buffer estático em pilha
    std::array<uint8_t, kMaxBufferSize> buffer;
    asn_enc_rval_t rval = uper_encode_to_buffer(
        &asn_DEF_CAM, nullptr, cam.get(), buffer.data(), buffer.size());

    if (rval.encoded <= 0) {
        V2XWriteLog(V2XLogLevel::ERROR, "CAM", "ENCODE_FAIL_ASN1", 
                    vehicleId, nodeId, seqId, "Vanetza ASN.1 encoder serialization crash");
        return {};
    }

    size_t size = (rval.encoded + 7) / 8;
    
    // Log em perfeita conformidade com a v1.0 da pipeline de análise
    V2XWriteLog(V2XLogLevel::TRACE, "CAM", "ENCODE_OK", 
                vehicleId, nodeId, seqId, 
                "bytes=" + std::to_string(size) + " stationId=" + std::to_string(v.stationId));

    return std::vector<uint8_t>(buffer.begin(), buffer.begin() + size);
}

#include "cam_framing.h"
#include "vanetza_integration/core/v2x_log.h"

#include <cstring>

std::vector<uint8_t> CamFraming::Pack(
    const std::vector<uint8_t>& payload,
    const std::vector<uint8_t>& signature,
    uint32_t timestamp,
    const std::string& vehicleId,
    uint64_t nodeId,
    int64_t seqId)
{
    uint16_t payloadSize   = payload.size();
    uint16_t signatureSize = signature.size();

    std::vector<uint8_t> buffer;

    buffer.resize(sizeof(payloadSize) +
                  sizeof(signatureSize) +
                  sizeof(timestamp));

    size_t offset = 0;

    // Nota: memcpy direto mantido para estabilidade de baixo nível, mas com logging contextualizado
    memcpy(buffer.data() + offset, &payloadSize, sizeof(payloadSize));
    offset += sizeof(payloadSize);

    memcpy(buffer.data() + offset, &signatureSize, sizeof(signatureSize));
    offset += sizeof(signatureSize);

    memcpy(buffer.data() + offset, &timestamp, sizeof(timestamp));
    offset += sizeof(timestamp);

    buffer.insert(buffer.end(), payload.begin(), payload.end());
    buffer.insert(buffer.end(), signature.begin(), signature.end());

    // Log em total conformidade com a v1.0 do interpretador analítico
    V2XWriteLog(
        V2XLogLevel::DEBUG,
        "CAM",
        "FRAME_PACK",
        vehicleId,
        nodeId,
        seqId,
        "payload=" + std::to_string(payloadSize) +
        " signature=" + std::to_string(signatureSize) +
        " total=" + std::to_string(buffer.size()) +
        " timestamp=" + std::to_string(timestamp));

    return buffer;
}

// ------------------------------------------------------

bool CamFraming::Unpack(
    const std::vector<uint8_t>& raw,
    CamFrame& frame,
    const std::string& vehicleId,
    uint64_t nodeId,
    int64_t seqId)
{
    const size_t headerMinimumSize = sizeof(uint16_t) * 2 + sizeof(uint32_t);
    
    if (raw.size() < headerMinimumSize)
    {
        V2XWriteLog(
            V2XLogLevel::WARN,
            "CAM",
            "FRAME_UNPACK_FAIL",
            vehicleId,
            nodeId,
            seqId,
            "raw too small: bytes=" + std::to_string(raw.size()) + 
            " min_required=" + std::to_string(headerMinimumSize));
        return false;
    }

    size_t offset = 0;

    uint16_t payloadSize;
    uint16_t signatureSize;
    uint32_t timestamp;

    memcpy(&payloadSize, raw.data() + offset, sizeof(payloadSize));
    offset += sizeof(payloadSize);

    memcpy(&signatureSize, raw.data() + offset, sizeof(signatureSize));
    offset += sizeof(signatureSize);

    memcpy(&timestamp, raw.data() + offset, sizeof(timestamp));
    offset += sizeof(timestamp);

    size_t expectedSize = offset + payloadSize + signatureSize;

    if (raw.size() < expectedSize)
    {
        V2XWriteLog(
            V2XLogLevel::ERROR,
            "CAM",
            "FRAME_UNPACK_FAIL",
            vehicleId,
            nodeId,
            seqId,
            "size mismatch raw=" + std::to_string(raw.size()) +
            " expected=" + std::to_string(expectedSize));
        return false;
    }

    frame.payload.assign(
        raw.begin() + offset,
        raw.begin() + offset + payloadSize);

    offset += payloadSize;

    frame.signature.assign(
        raw.begin() + offset,
        raw.begin() + offset + signatureSize);

    frame.timestamp = timestamp;

    // Log contextualizado emitindo as métricas corretas do fluxo de entrada de quadros
    V2XWriteLog(
        V2XLogLevel::DEBUG,
        "CAM",
        "FRAME_UNPACK_OK",
        vehicleId,
        nodeId,
        seqId,
        "payload=" + std::to_string(payloadSize) +
        " signature=" + std::to_string(signatureSize) +
        " timestamp=" + std::to_string(timestamp));

    return true;
}

bool CamReceiver::ProcessRawPacket(
    const std::string& recVehId,
    uint32_t recStationId,
    int64_t seqId,
    const std::vector<uint8_t>& raw,
    CamFrame& outFrame)
{

    // EMPTY CHECK
    // ======================================================
    if (raw.empty())
    {
        V2XWriteLog(
            V2XLogLevel::WARN,
            "CAM",
            "RX_EMPTY",
            recVehId,
            static_cast<uint64_t>(recStationId),
            seqId,
            "received empty packet");
        return false;
    }

    // ======================================================
    // UNPACK 
    // Assinatura esperada pelo header:
    // (const std::vector<uint8_t>& raw, CamFrame& frame, const std::string& vehicleId, uint64_t nodeId, int64_t seqId)
    // ======================================================
    if (!CamFraming::Unpack(raw, outFrame, recVehId, static_cast<uint64_t>(recStationId), seqId))
    {

// ======================================================
// GENERATE CAM (CORRIGIDO)
// ======================================================
std::vector<uint8_t> CamService::GenerateCAM(const VehicleState& v)
{
    if (!v.node) return {};

    uint32_t stationID   = v.identity.stationId;
    uint32_t nodeId      = v.identity.nodeId;
    std::string vehId    = v.identity.vehId;
    int64_t currentSeqId = v.camSequenceCounter; 

    if (stationID == 0) return {};

    CamInput input;
    input.stationId    = stationID;
    input.latitude     = v.position.y;
    input.longitude    = v.position.x;
    input.speed        = v.speed;
    input.heading      = v.heading;
    input.acceleration = v.accel;

    // 1. ENCODE (Agora passa os 4 argumentos exigidos)
    std::vector<uint8_t> payload = CamEncoder::Encode(input, vehId, nodeId, currentSeqId);
    if (payload.empty()) return {};

    // 2. SIGN (Agora passa os 4 argumentos exigidos: data, key, vehId, nodeId)
    // Nota: Assume que 'v.privateKey' é do tipo EVP_PKEY*
    std::vector<uint8_t> signature = CamSecurity::Sign(payload, v.privateKey, vehId, nodeId);
    if (signature.empty()) return {};

    // 3. FRAME (Agora passa os 6 argumentos exigidos)
    uint32_t simTimestampEtsi = static_cast<uint32_t>(ns3::Simulator::Now().GetMilliSeconds() % 65536);
    
    // Assinatura: payload, signature, timestamp, vehId, nodeId, seqId
    std::vector<uint8_t> frame = CamFraming::Pack(payload, signature, simTimestampEtsi, vehId, nodeId, currentSeqId);
    
    if (frame.empty()) return {};

    V2XWriteLog(V2XLogLevel::INFO, "CAM", "TX_READY", vehId, nodeId, currentSeqId, "frame_size=" + std::to_string(frame.size()));

    return frame;
}

bool CamService::ShouldSendCAM(const VehicleState& cur, const VehicleState& last)
{
    double dpos = std::sqrt(std::pow(cur.position.x - last.position.x, 2) + std::pow(cur.position.y - last.position.y, 2));
    double dspeed = std::fabs(cur.speed - last.speed);
    double dheading = std::fabs(cur.heading - last.heading);
    return (dpos > 4.0 || dspeed > 0.5 || dheading > 4.0);
}

    std::map<uint32_t, NeighborInfo>& table,
    uint64_t currentStep,
    uint64_t maxAge)

    for (auto it = table.begin(); it != table.end(); )
    {
        if (currentStep - it->second.lastUpdateStep > maxAge)
        {
            V2XWriteLog(
                V2XLogLevel::DEBUG,
                "NEIGHBOR",
                "PURGE",
                std::to_string(it->first), // vehId
                0,                         // nodeId
                -1,                        // seqId
                "age=" + std::to_string(currentStep - it->second.lastUpdateStep));

            it = table.erase(it);
        }
        else
        {
            ++it;
        }
    }

void RxPipeline::Process(ns3::Ptr<ns3::Socket> socket, const std::vector<unsigned char>& buffer)
{
    // 1. Identificação do nó receptor
    const uint32_t rxNodeId = socket->GetNode()->GetId();
    auto& registry = VehicleRegistry::Instance();
    
    VehicleState* receiver = registry.GetByNodeId(rxNodeId);
    std::string rxVehId = receiver ? receiver->identity.vehId : "unknown_rx";

    // Parâmetros para as novas assinaturas (ajustar conforme necessidade da stack)
    const std::string senderId = "unknown"; // Extrair do header se disponível
    const uint32_t timestamp = 0;           // Ou obter de um relógio global
    const int64_t seqId = -1;

    V2XWriteLog(V2XLogLevel::TRACE, "RX", "RAW_IN", rxVehId, rxNodeId, seqId, "size=" + std::to_string(buffer.size()));

    // 2. Processamento da camada de frame
    CamFrame frame;
    // Corrigido: ProcessRawPacket exige 5 argumentos conforme o erro de compilação
    if (!CamReceiver::ProcessRawPacket(senderId, timestamp, seqId, buffer, frame)) {
        V2XWriteLog(V2XLogLevel::WARN, "RX", "FRAME_FAIL", rxVehId, rxNodeId, seqId, "framing failed");
        return;
    }

    // 3. Decodificação
    // Corrigido: Decode exige 4 argumentos
    CamMessage msg = CamDecoder::Decode(frame.payload, senderId, timestamp, seqId);
    
    if (msg.stationId == 0) {
        V2XWriteLog(V2XLogLevel::ERROR, "RX", "DECODE_FAIL", rxVehId, rxNodeId, seqId, "invalid stationId");
        return;
    }

    V2XWriteLog(V2XLogLevel::TRACE, "RX", "DECODE_OK", rxVehId, rxNodeId, seqId, "sid=" + std::to_string(msg.stationId));

    // 4. Validação do Remetente
    VehicleState* sender = registry.GetByStationId(msg.stationId);
    if (!sender) {
        V2XWriteLog(V2XLogLevel::ERROR, "RX", "SENDER_NOT_FOUND", rxVehId, rxNodeId, seqId, "sid_not_in_registry");
        return;
    }

    // 5. Verificação de Segurança
    if (!sender->certificate) {
        V2XWriteLog(V2XLogLevel::WARN, "RX", "NO_CERT", sender->identity.vehId, rxNodeId, seqId, "missing_cert");
        return;
    }

    // Corrigido: VerifyFrame exige 5 argumentos
    if (!CamSecurity::VerifyFrame(frame, sender->certificate, senderId, timestamp, seqId)) {
        V2XWriteLog(V2XLogLevel::ERROR, "RX", "VERIFY_RESULT_FAIL", sender->identity.vehId, rxNodeId, seqId, "auth_failed");
        return;
    }

    // 6. Atualização de Estado
    if (receiver) {
        NeighborInfo info;
        info.stationID = msg.stationId;
        info.x = msg.longitude;
        info.y = msg.latitude;
        info.speed = msg.speed;
        info.heading = msg.heading;
        info.lastUpdateStep = SimulationContext::simulationStep;

        receiver->neighbors[msg.stationId] = info;
        
        V2XWriteLog(V2XLogLevel::INFO, "RX", "PIPELINE_OK", sender->identity.vehId, rxNodeId, seqId, "neighbor_updated");
    }
}

    const std::vector<uint8_t>& data,
    EVP_PKEY* privateKey,
    const std::string& vehId,
    uint32_t nodeId)
{
    if (!privateKey || data.empty())
    {
        V2XWriteLog(V2XLogLevel::ERROR, "SECURITY", "SIGN_FAIL", vehId, nodeId, -1, "invalid input or missing key");
        return {};
    }

    std::unique_ptr<EVP_MD_CTX, OpenSSLDeleter> ctx(EVP_MD_CTX_new());
    if (!ctx)
    {
        V2XWriteLog(V2XLogLevel::ERROR, "SECURITY", "SIGN_FAIL", vehId, nodeId, -1, "EVP_MD_CTX_new failed");
        return {};
    }

    size_t sigLen = EVP_PKEY_size(privateKey);
    std::vector<uint8_t> signature(sigLen);

    bool ok =
        EVP_DigestSignInit(ctx.get(), nullptr, EVP_sha256(), nullptr, privateKey) == 1 &&
        EVP_DigestSignUpdate(ctx.get(), data.data(), data.size()) == 1 &&
        EVP_DigestSignFinal(ctx.get(), signature.data(), &sigLen) == 1;

    if (!ok)
    {
        V2XWriteLog(V2XLogLevel::ERROR, "SECURITY", "SIGN_FAIL", vehId, nodeId, -1, "digest sign failed");
        return {};
    }

    signature.resize(sigLen);

    V2XWriteLog(V2XLogLevel::DEBUG, "SECURITY", "SIGN_OK", vehId, nodeId, -1, "signature_size=" + std::to_string(sigLen));

    return signature;
}

// ======================================================
// VERIFY
// ======================================================
bool CamSecurity::Verify(
    const std::vector<uint8_t>& data,
    const std::vector<uint8_t>& signature,
    X509* certificate,
    const std::string& vehId,
    uint32_t nodeId)
{
    if (!certificate || data.empty() || signature.empty())
    {
        V2XWriteLog(V2XLogLevel::WARN, "SECURITY", "VERIFY_FAIL", vehId, nodeId, -1, "invalid input");
        return false;
    }

    std::unique_ptr<EVP_PKEY, OpenSSLDeleter> pubKey(X509_get_pubkey(certificate));
    if (!pubKey)
    {
        V2XWriteLog(V2XLogLevel::ERROR, "SECURITY", "VERIFY_FAIL", vehId, nodeId, -1, "pubkey extraction failed");
        return false;
    }

    std::unique_ptr<EVP_MD_CTX, OpenSSLDeleter> ctx(EVP_MD_CTX_new());
    if (!ctx)
    {
        V2XWriteLog(V2XLogLevel::ERROR, "SECURITY", "VERIFY_FAIL", vehId, nodeId, -1, "EVP_MD_CTX_new failed");
        return false;
    }

    bool valid = false;
    if (EVP_DigestVerifyInit(ctx.get(), nullptr, EVP_sha256(), nullptr, pubKey.get()) == 1 &&
        EVP_DigestVerifyUpdate(ctx.get(), data.data(), data.size()) == 1 &&
        EVP_DigestVerifyFinal(ctx.get(), signature.data(), signature.size()) == 1)
    {
        valid = true;
    }

    V2XWriteLog(
        valid ? V2XLogLevel::INFO : V2XLogLevel::ERROR,
        "SECURITY",
        "VERIFY_RESULT",
        vehId,
        nodeId,
        -1,
        valid ? "OK" : "FAIL");

    return valid;
}

// ======================================================
// VERIFY FRAME (PIPELINE DE ALTO NÍVEL)
// ======================================================
bool CamSecurity::VerifyFrame(
    const CamFrame& frame,
    X509* certificate,
    const std::string& currentVehId,
    uint32_t currentNodeId,
    uint32_t seqId)
{
    bool ok = Verify(frame.payload, frame.signature, certificate, currentVehId, currentNodeId);

    V2XWriteLog(
        ok ? V2XLogLevel::INFO : V2XLogLevel::ERROR,
        "SECURITY",
        "FRAME_VERIFY",
        currentVehId,
        currentNodeId,
        static_cast<int64_t>(seqId),
        "frame_valid=" + std::string(ok ? "TRUE" : "FALSE"));

    return ok;
}

static int udpRxSocketFd = -1;

// ======================================================
// PARSER: Retorna unique_ptr para evitar cópia proibida
// ======================================================
std::unique_ptr<VehicleState> SumoAdapter::ParseMessage(const std::string& msg)
{
    auto vs = std::make_unique<VehicleState>();

    // Inicialização segura
    vs->identity.nodeId = 0;
    vs->identity.stationId = 0;
    vs->speed = 0.0;
    vs->heading = 0.0;
    vs->accel = 0.0;
    vs->position.x = 0.0;
    vs->position.y = 0.0;
    vs->initialized = false;

    std::stringstream ss(msg);
    std::string token;

    if (!getline(ss, vs->identity.vehId, ',')) return nullptr;
    if (!getline(ss, token, ',')) return nullptr;
    vs->position.x = std::stod(token);
    if (!getline(ss, token, ',')) return nullptr;
    vs->position.y = std::stod(token);
    if (!getline(ss, token, ',')) return nullptr;
    vs->speed = std::stod(token);
    if (!getline(ss, token, ',')) return nullptr;
    vs->heading = std::stod(token);
    if (!getline(ss, token, ',')) return nullptr;
    vs->accel = std::stod(token);

    vs->initialized = true;
    return vs;
}

// ======================================================
// INIT
// ======================================================
bool SumoAdapter::Init(int localPort, const std::string& sumoIp, int sumoPort)
{
    udpRxSocketFd = socket(AF_INET, SOCK_DGRAM, 0);
    if (udpRxSocketFd < 0) {
        V2XWriteLog(V2XLogLevel::ERROR, "SUMO", "SOCKET_FAIL", "-", 0, -1, "UDP socket creation failed");
        return false;
    }

    memset(&localAddr, 0, sizeof(localAddr));
    localAddr.sin_family = AF_INET;
    localAddr.sin_port = htons(localPort);
    localAddr.sin_addr.s_addr = INADDR_ANY;

    if (bind(udpRxSocketFd, (sockaddr*)&localAddr, sizeof(localAddr)) < 0) {
        V2XWriteLog(V2XLogLevel::ERROR, "SUMO", "BIND_FAIL", "-", 0, -1, "UDP bind failed");
        return false;
    }

    memset(&sumoAddr, 0, sizeof(sumoAddr));
    sumoAddr.sin_family = AF_INET;
    sumoAddr.sin_port = htons(sumoPort);
    sumoAddr.sin_addr.s_addr = inet_addr(sumoIp.c_str());

    V2XWriteLog(V2XLogLevel::INFO, "SUMO", "INIT_OK", "-", 0, -1, "adapter initialized");
    return true;
}

// ======================================================
// CALLBACK
// ======================================================
void SumoAdapter::SetVehicleUpdateCallback(std::function<void(const VehicleState&)> cb)
{
    onVehicleUpdate = std::move(cb);
}

// ======================================================
// SEND
// ======================================================
void SumoAdapter::SendToSumo(const VehicleState& vs)
{
    if (udpRxSocketFd < 0) return;

    std::string msg = vs.identity.vehId + "," + std::to_string(vs.position.x) + "," + 
                      std::to_string(vs.position.y) + "," + std::to_string(vs.speed) + "," + 
                      std::to_string(vs.heading) + "," + std::to_string(vs.accel);

    sendto(udpRxSocketFd, msg.c_str(), msg.size(), 0, (sockaddr*)&sumoAddr, sizeof(sumoAddr));

    V2XWriteLog(V2XLogLevel::DEBUG, "SUMO_TX", "VEHICLE_STATE", vs.identity.vehId, 
                vs.identity.nodeId, -1, "pos=(" + std::to_string(vs.position.x) + "," + 
                std::to_string(vs.position.y) + ") speed=" + std::to_string(vs.speed));
}

// ======================================================
// POLL: Processa via ponteiro (unique_ptr)
// ======================================================
void SumoAdapter::Poll()
{
    if (udpRxSocketFd < 0) return;

    char buffer[512];
    sockaddr_in sender{};
    socklen_t len = sizeof(sender);
    auto& reg = VehicleRegistry::Instance();

    while (true) {
        int bytes = recvfrom(udpRxSocketFd, buffer, sizeof(buffer) - 1, MSG_DONTWAIT, (sockaddr*)&sender, &len);
        if (bytes <= 0) break;

        buffer[bytes] = '\0';
        auto vs_ptr = ParseMessage(buffer);
        
        if (!vs_ptr || !vs_ptr->initialized) continue;

        VehicleState* registeredVeh = reg.GetByVehId(vs_ptr->identity.vehId);
        if (registeredVeh) {
            vs_ptr->identity.nodeId = registeredVeh->identity.nodeId;
            vs_ptr->identity.stationId = registeredVeh->identity.stationId;
        }

        V2XWriteLog(V2XLogLevel::DEBUG, "SUMO_RX", "VEHICLE_STATE", vs_ptr->identity.vehId, 
                    vs_ptr->identity.nodeId, -1, "pos=(" + std::to_string(vs_ptr->position.x) + 
                    "," + std::to_string(vs_ptr->position.y) + ") speed=" + std::to_string(vs_ptr->speed));

        if (onVehicleUpdate) {
            onVehicleUpdate(*vs_ptr);
        }
    }
}

void SumoAdapter::Close()
{
    if (udpRxSocketFd >= 0) {
        close(udpRxSocketFd);
        udpRxSocketFd = -1;
    }
}
