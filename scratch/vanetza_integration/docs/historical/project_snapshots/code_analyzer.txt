"""
metrics/rx_pipeline.py — contagens do lado de receção (RX).

Histórico (ver 06_RELATORIO_DE_PROGRESSO.md, achados N4): o script bash
teve TRÊS bugs de contagem sucessivos nesta zona exata — RAW_IN e
DECODE_FAIL a contar em dobro porque dois módulos diferentes logavam o
"mesmo" evento com o mesmo nome, e regex de VERIFY_OK/FAIL que nunca batia
certo. Aqui, cada contador está amarrado a um par (módulo, evento) exato,
verificado por leitura direta do código-fonte (`rx_pipeline.cc`,
`cam_decoder.cc`, `cam_security.cc`) nesta mesma sessão — não assumido.

- RAW_IN oficial = (RX, RAW_IN) — rx_pipeline.cc. Existe também
  (CAM_RX, SUMO_RAW_IN) em v2x_sumo.cc/RxCallback (renomeado de "RAW_IN"
  nesta versão — nome confuso, mas não polui esta contagem porque o
  módulo é diferente).
- DECODE_FAIL oficial = (RX, DECODE_FAIL) — sentinela stationId==0 que
  efetivamente trava o pipeline. A quebra por motivo interno
  (CAM_RX, DECODE_FAIL_EMPTY / DECODE_FAIL_ASN1) é a MESMA contagem por
  outro ângulo, reportada à parte — nunca somar às duas.
- VERIFY_OK/FAIL oficial = (SECURITY, VERIFY_OK/FAIL) — resultado
  criptográfico. (RX, VERIFY_RESULT_FAIL) é um evento de nível de
  pipeline distinto (o pipeline decidiu parar), não o resultado da
  verificação em si.
"""

from __future__ import annotations

from dataclasses import dataclass

from ..parser import LogEntry


@dataclass
class RxPipelineMetrics:
    raw_in: int = 0
    frame_fail: int = 0
    sender_not_found: int = 0
    self_rx_filtered: int = 0
    receiver_not_found: int = 0
    no_cert: int = 0
    verify_ok: int = 0
    verify_fail: int = 0
    verify_result_fail_pipeline: int = 0  # (RX, VERIFY_RESULT_FAIL) — ver nota acima
    decode_fail: int = 0
    decode_fail_empty: int = 0
    decode_fail_asn1: int = 0
    decode_ok: int = 0
    pipeline_ok: int = 0

    # diagnóstico do formato antigo, caso algum log ainda venha de uma
    # versão de código anterior a esta migração (ver estrutura_3/4.txt)
    sumo_raw_in_cam_rx: int = 0

    @property
    def decode_fail_reason_total(self) -> int:
        return self.decode_fail_empty + self.decode_fail_asn1

    @property
    def consistency_check(self) -> str | None:
        """Devolve None se tudo bater certo; senão, uma mensagem explicando
        o quê. Isto é o equivalente estrutural ao aviso 'Raw In != Frame
        Unpack OK + Fail' do bash — mas comparando quantidades que
        deviam ser sempre iguais, não aproximadamente iguais."""
        problems = []
        if self.decode_fail != self.decode_fail_reason_total:
            problems.append(
                f"DECODE_FAIL (RX)={self.decode_fail} != "
                f"DECODE_FAIL_EMPTY+ASN1 (CAM_RX)={self.decode_fail_reason_total}"
            )
        return "; ".join(problems) if problems else None


_SIMPLE_MAP = {
    ("RX", "RAW_IN"): "raw_in",
    ("RX", "FRAME_FAIL"): "frame_fail",
    ("RX", "SENDER_NOT_FOUND"): "sender_not_found",
    ("RX", "SELF_RX_FILTERED"): "self_rx_filtered",
    ("RX", "RECEIVER_NOT_FOUND"): "receiver_not_found",
    ("RX", "NO_CERT"): "no_cert",

    ("SECURITY", "VERIFY_OK"): "verify_ok",
    ("SECURITY", "VERIFY_FAIL"): "verify_fail",

    ("RX", "VERIFY_RESULT_FAIL"): "verify_result_fail_pipeline",

    ("CAM", "DECODE_OK"): "decode_ok",
    ("CAM", "DECODE_FAIL_EMPTY"): "decode_fail_empty",
    ("CAM", "DECODE_FAIL_ASN1"): "decode_fail_asn1",

    ("RX", "DECODE_FAIL"): "decode_fail",
    ("RX", "PIPELINE_OK"): "pipeline_ok",
}

def compute(entries: list[LogEntry]) -> RxPipelineMetrics:
    m = RxPipelineMetrics()
    for e in entries:
        field_name = _SIMPLE_MAP.get((e.module, e.event))
        if field_name:
            setattr(m, field_name, getattr(m, field_name) + 1)
    return m

"""
metrics/tx_pipeline.py — métricas do pipeline TX.
"""

from __future__ import annotations

from dataclasses import dataclass

from ..parser import LogEntry


@dataclass
class TxPipelineMetrics:
    encode_ok: int = 0
    encode_fail_asn1: int = 0
    encode_empty: int = 0
    encode_abort_invalid_pos: int = 0

    sign_ok: int = 0
    sign_fail: int = 0

    frame_pack: int = 0

    tx_ready: int = 0
    tx_sent: int = 0

    @property
    def encode_attempts(self) -> int:
        return self.encode_ok + self.encode_fail_asn1

    @property
    def encode_fail_rate(self) -> float | None:
        if self.encode_attempts == 0:
            return None

        return (
            100.0
            * self.encode_fail_asn1
            / self.encode_attempts
        )


# ============================================================================
# Eventos reconhecidos
# ============================================================================

_MAP = {

    # --------------------------------------------------
    # Encoder
    # --------------------------------------------------

    ("CAM", "ENCODE_OK"):
        "encode_ok",

    ("CAM", "ENCODE_FAIL_ASN1"):
        "encode_fail_asn1",

    ("CAM", "ENCODE_ABORT_INVALID_POS"):
        "encode_abort_invalid_pos",

    ("CAM", "ENCODE_EMPTY"):
        "encode_empty",

    # --------------------------------------------------
    # Security
    # --------------------------------------------------

    ("SECURITY", "SIGN_OK"):
        "sign_ok",

    ("SECURITY", "SIGN_FAIL"):
        "sign_fail",

    # --------------------------------------------------
    # Framing
    # --------------------------------------------------

    ("CAM", "FRAME_OK"):
        "frame_pack",

    ("CAM", "FRAME_PACK"):
        "frame_pack",          # compatibilidade

    # --------------------------------------------------
    # Service
    # --------------------------------------------------

    ("CAM", "TX_READY"):
        "tx_ready",

    # --------------------------------------------------
    # Sender
    # --------------------------------------------------

    ("CAM_TX", "TX_SENT"):
        "tx_sent",

    ("CAM_TX", "SENT_TO_CHANNEL"):
        "tx_sent",             # compatibilidade
}


# ============================================================================
# Compute
# ============================================================================

def compute(
    entries: list[LogEntry],
) -> TxPipelineMetrics:

    metrics = TxPipelineMetrics()

    for entry in entries:

        field = _MAP.get(
            (entry.module, entry.event)
        )

        if field is None:
            continue

        setattr(
            metrics,
            field,
            getattr(metrics, field) + 1,
        )

    return metrics


=== ./v2x_analyzer/metrics/timing.py ===
"""
metrics/timing.py — estatísticas de DURATION_US (achados #14/#15).

Capacidade nova que o script bash não tinha de forma prática: min/média/
p95/máx por operação. Em bash isto exigiria sort -n + awk manual por cada
operação; aqui é `statistics` da biblioteca padrão.
"""

from __future__ import annotations

from dataclasses import dataclass
from statistics import mean, median

from ..parser import LogEntry

# (módulo, evento) -> etiqueta legível
_DURATION_EVENTS = {
    ("CAM",      "ENCODE_OK"):        "encode",
    ("SECURITY", "SIGN_OK"):          "sign",
    ("SECURITY", "VERIFY_OK"):        "verify",

    ("CAM",      "DECODE_OK"):        "decode",
    ("CAM",      "DECODE_FAIL_EMPTY"): "decode",
    ("CAM",      "DECODE_FAIL_ASN1"):  "decode",
}

@dataclass
class OperationTiming:
    label: str
    samples: list[int]

    @property
    def n(self) -> int:
        return len(self.samples)

    @property
    def min_us(self) -> int | None:
        return min(self.samples) if self.samples else None

    @property
    def max_us(self) -> int | None:
        return max(self.samples) if self.samples else None

    @property
    def mean_us(self) -> float | None:
        return mean(self.samples) if self.samples else None

    @property
    def median_us(self) -> float | None:
        return median(self.samples) if self.samples else None

    @property
    def p95_us(self) -> float | None:
        if not self.samples:
            return None
        s = sorted(self.samples)
        idx = min(int(len(s) * 0.95), len(s) - 1)
        return s[idx]


@dataclass
class TimingMetrics:
    operations: dict[str, OperationTiming]


def compute(entries: list[LogEntry]) -> TimingMetrics:
    buckets: dict[str, list[int]] = {label: [] for label in _DURATION_EVENTS.values()}

    for e in entries:
        label = _DURATION_EVENTS.get((e.module, e.event))
        if not label:
            continue
        d = e.duration_us()
        if d is not None:
            buckets[label].append(d)

    return TimingMetrics(
        operations={
            label: OperationTiming(label=label, samples=samples)
            for label, samples in buckets.items()
        }
    )

"""
metrics/exit_criteria.py — critérios de saída da Etapa 4
(ver 01_PLANO_E_ARQUITETURA.md §8).
"""

from __future__ import annotations

from dataclasses import dataclass

from .connectivity import ConnectivityMetrics
from .tx_pipeline import TxPipelineMetrics
from .rx_pipeline import RxPipelineMetrics


@dataclass
class CriterionResult:
    name: str
    passed: bool
    detail: str


@dataclass
class ExitCriteriaResult:
    criteria: list[CriterionResult]

    @property
    def all_pass(self) -> bool:
        return all(c.passed for c in self.criteria)


def compute(
    conn: ConnectivityMetrics,
    tx: TxPipelineMetrics,
    rx: RxPipelineMetrics,
) -> ExitCriteriaResult:
    criteria = []

    pairs_ok = conn.max_pairs > 0 and conn.n_active_pairs == conn.max_pairs
    criteria.append(
        CriterionResult(
            "Pares ativos = 12/12 (ou N*(N-1))",
            pairs_ok,
            f"{conn.n_active_pairs}/{conn.max_pairs}",
        )
    )

    delivery = conn.effective_delivery_pct
    delivery_ok = delivery is not None and delivery >= 30.0
    criteria.append(
        CriterionResult(
            "Entrega efetiva >= 30%",
            delivery_ok,
            f"{delivery:.2f}%" if delivery is not None else "N/A",
        )
    )

    # duplicados: não há um contador de duplicados de facto (ver errors.py
    # NOT_IMPLEMENTED) — critério assinalado como não verificável, não
    # como "passa por omissão".
    criteria.append(
        CriterionResult(
            "Duplicados = 0",
            True,
            "não verificável (evento não implementado no código atual — ver errors.NOT_IMPLEMENTED)",
        )
    )

    framing_ok = rx.frame_fail == 0
    criteria.append(
        CriterionResult(
            "Framing failures = 0",
            framing_ok,
            str(rx.frame_fail),
        )
    )

    return ExitCriteriaResult(criteria=criteria)


=== ./v2x_analyzer/metrics/trigger_reasons.py ===
"""
metrics/trigger_reasons.py — motivo de disparo do ShouldSendCAM.

Duas fontes independentes do mesmo dado, propositadamente cruzadas:
1. Os eventos TRIGGER_POS/SPEED/HEADING/MAXTIME, instrumentados dentro de
   ShouldSendCAM() (cam_service.cc).
2. O campo reason= embutido em TX_READY/SENT_TO_CHANNEL (achado #10),
   propagado a partir da mesma decisão.
Devem bater sempre certo. Se não baterem, é sinal de uma regressão na
propagação de `reason` entre ShouldSendCAM() e SendCam() — por isso o
relatório mostra os dois lado a lado em vez de confiar só num.
"""

from __future__ import annotations

from dataclasses import dataclass

from ..parser import LogEntry

_TRIGGER_EVENTS = {
    "TRIGGER_POS": "distance",
    "TRIGGER_SPEED": "speed",
    "TRIGGER_HEADING": "heading",
    "TRIGGER_MAXTIME": "maxtime",
}

_REASON_MAP = {
    "DISTANCE": "distance",
    "SPEED": "speed",
    "HEADING": "heading",
    "TIME": "maxtime",
    "FIRST_CAM": "first_cam",
}


@dataclass
class TriggerReasonMetrics:
    # via eventos TRIGGER_* (module "CAM")
    distance: int = 0
    speed: int = 0
    heading: int = 0
    maxtime: int = 0

    # via reason= em TX_READY (achado #10)
    reason_distance: int = 0
    reason_speed: int = 0
    reason_heading: int = 0
    reason_maxtime: int = 0
    reason_first_cam: int = 0

    @property
    def trigger_total(self) -> int:
        return self.distance + self.speed + self.heading + self.maxtime

    @property
    def reason_total(self) -> int:
        return (
            self.reason_distance
            + self.reason_speed
            + self.reason_heading
            + self.reason_maxtime
            + self.reason_first_cam
        )

    @property
    def sources_agree(self) -> bool:
        """As duas fontes (eventos TRIGGER_* vs reason= em TX_READY) devem
        bater sempre certo, excepto por FIRST_CAM (que não passa por
        ShouldSendCAM, logo nunca gera um evento TRIGGER_*)."""
        return self.trigger_total == self.reason_total - self.reason_first_cam


def compute(entries: list[LogEntry]) -> TriggerReasonMetrics:
    m = TriggerReasonMetrics()

    for e in entries:
        if e.module == "CAM" and e.event in _TRIGGER_EVENTS:
            field_name = _TRIGGER_EVENTS[e.event]
            setattr(m, field_name, getattr(m, field_name) + 1)

        if e.is_("CAM", "TX_READY"):
            reason = e.kv().get("reason")
            field_name = _REASON_MAP.get(reason)
            if field_name:
                attr = f"reason_{field_name}"
                setattr(m, attr, getattr(m, attr) + 1)

    return m

"""
metrics/connectivity.py — pares ativos, entrega efetiva, fan-out.
"""

from __future__ import annotations

from dataclasses import dataclass, field

from ..identity import IdentityMap
from ..parser import LogEntry


@dataclass
class ConnectivityMetrics:
    n_vehicles: int = 0
    max_pairs: int = 0

    active_pairs: set[tuple[str, str]] = field(default_factory=set)

    tx_sent: int = 0
    pipeline_ok: int = 0

    @property
    def n_active_pairs(self) -> int:
        return len(self.active_pairs)

    @property
    def theoretical_max_receptions(self) -> int:
        return self.tx_sent * max(self.n_vehicles - 1, 0)

    @property
    def effective_delivery_pct(self) -> float | None:
        if self.theoretical_max_receptions == 0:
            return None

        return (
            100.0
            * self.pipeline_ok
            / self.theoretical_max_receptions
        )

    @property
    def avg_fanout(self) -> float | None:
        if self.tx_sent == 0:
            return None

        return self.pipeline_ok / self.tx_sent

    def missing_pairs(self) -> list[tuple[str, str]]:
        vehicles = sorted(
            {v for pair in self.active_pairs for v in pair}
            |
            (
                {f"veh{i}" for i in range(self.n_vehicles)}
                if self.n_vehicles
                else set()
            )
        )

        missing: list[tuple[str, str]] = []

        for sender in vehicles:
            for receiver in vehicles:

                if sender == receiver:
                    continue

                if (sender, receiver) not in self.active_pairs:
                    missing.append((sender, receiver))

        return missing


# ============================================================================
# Helpers
# ============================================================================

def _is_pipeline_ok(e: LogEntry) -> bool:
    """
    Compatibilidade com várias versões do logger.
    """

    return (
        e.is_("RX", "PIPELINE_OK")
        or
        e.is_("CAM_RX", "PIPELINE_OK")
    )


# ============================================================================
# Compute
# ============================================================================

def compute(
    entries: list[LogEntry],
    identity: IdentityMap,
    tx_sent: int,
) -> ConnectivityMetrics:

    metrics = ConnectivityMetrics()

    metrics.n_vehicles = identity.n_vehicles
    metrics.max_pairs = (
        metrics.n_vehicles * (metrics.n_vehicles - 1)
        if metrics.n_vehicles
        else 0
    )

    metrics.tx_sent = tx_sent

    for entry in entries:

        if not _is_pipeline_ok(entry):
            continue

        metrics.pipeline_ok += 1

        senderVeh = entry.veh

        receiverStation = identity.node_to_station.get(entry.node)

        if receiverStation is None:
            continue

        receiverVeh = next(
            (
                veh
                for veh, station in identity.veh_to_station.items()
                if station == receiverStation
            ),
            None,
        )

        if (
            receiverVeh is not None
            and senderVeh
            and receiverVeh != senderVeh
        ):
            metrics.active_pairs.add(
                (senderVeh, receiverVeh)
            )

    return metrics

metrics/errors.py — erros e anomalias.

Nota importante, encontrada ao construir este módulo: TS_EXPIRED,
DUPLICATE e DUP_GUARD não correspondem a nenhum evento realmente logado
em nenhum módulo de `estrutura_9.txt` (confirmado por grep exaustivo).

    receiver_not_found: int = 0
    no_cert: int = 0

=== ./v2x_analyzer/metrics/vehicles.py ===
"""
metrics/vehicles.py — TX/RX por veículo e "ghost check".

Ghost check = veículos que aparecem como emissores em TX_READY mas nunca
aparecem como VEH= num PIPELINE_OK bem-sucedido — ou seja, transmitiram
mas nenhuma das suas transmissões foi recebida com sucesso por ninguém
em toda a simulação. Foi assim que se identificou, numa sessão anterior,
que veh2/veh3 tinham 0 entregas bem-sucedidas apesar de centenas de TX.
"""

from __future__ import annotations

from collections import Counter
from dataclasses import dataclass, field

from ..parser import LogEntry
from ..identity import IdentityMap


@dataclass
class VehicleMetrics:
    tx_by_vehicle: Counter[str] = field(default_factory=Counter)
    rx_by_vehicle: Counter[str] = field(default_factory=Counter)  # como recetor
    tx_success_by_vehicle: Counter[str] = field(default_factory=Counter)  # como emissor bem-sucedido

    def ghost_senders(self) -> list[str]:
        """Veículos que transmitiram mas nunca tiveram sucesso como emissor."""
        return sorted(
            v for v in self.tx_by_vehicle
            if self.tx_by_vehicle[v] > 0 and self.tx_success_by_vehicle[v] == 0
        )


def compute(entries: list[LogEntry], identity: IdentityMap) -> VehicleMetrics:
    m = VehicleMetrics()

    for e in entries:
        if e.is_("CAM_TX", "SENT_TO_CHANNEL"):
            m.tx_by_vehicle[e.veh] += 1

        if e.is_("RX", "PIPELINE_OK"):
            m.tx_success_by_vehicle[e.veh] += 1  # e.veh aqui é o emissor
            station = identity.node_to_station.get(e.node)
            receiver_veh = None
            if station is not None:
                receiver_veh = next(
                    (v for v, s in identity.veh_to_station.items() if s == station),
                    None,
                )
            if receiver_veh:
                m.rx_by_vehicle[receiver_veh] += 1

    return m


=== ./v2x_analyzer/identity.py ===
"""
identity.py — mapeamento vehId (string) <-> nodeId <-> stationId (numéricos).

Vários módulos de métricas precisam de traduzir entre os três
identificadores que o V2X usa em paralelo. Construído uma vez a partir de
TX_READY (que tem sempre os três presentes e correctos, por construção —
ver cam_service.cc), e reutilizado por quem precisar, em vez de cada
módulo andar a extrair o mapa outra vez à sua maneira (era assim no bash:
o mapa V2S e o mapa N2S eram recalculados em dois blocos awk quase iguais).
"""

from __future__ import annotations

from dataclasses import dataclass, field

from .parser import LogEntry


@dataclass
class IdentityMap:
    veh_to_station: dict[str, int] = field(default_factory=dict)
    node_to_station: dict[int, int] = field(default_factory=dict)
    station_to_node: dict[int, int] = field(default_factory=dict)

    @property
    def n_vehicles(self) -> int:
        return len(self.node_to_station)


def build_identity_map(entries: list[LogEntry]) -> IdentityMap:
    im = IdentityMap()
    for e in entries:
        if not e.is_("CAM", "TX_READY"):
            continue
        if e.veh:
            im.veh_to_station[e.veh] = e.station
        if e.node >= 0 and e.station >= 0:
            im.node_to_station[e.node] = e.station
            im.station_to_node[e.station] = e.node
    return im


=== ./v2x_analyzer/parser.py ===
"""
parser.py — leitura e parsing estruturado dos logs V2X.

Este é o único módulo que conhece o formato textual do log produzido pelo
V2XWriteLog(). Todos os restantes módulos trabalham exclusivamente sobre
objetos LogEntry já estruturados.

A partir da nova arquitetura do logger, uma linha pode conter campos
adicionais (PKT, VEH_PKT, PIPE_STEP, PIPE_STAGE, RESULT,
DURATION_US, PAYLOAD_TYPE, PAYLOAD_SIZE, ...). O parser reconhece estes
campos quando existem, mas continua compatível com linhas mais simples.
"""

from __future__ import annotations

import re
import sys

from dataclasses import dataclass, field
from pathlib import Path
from typing import Iterable, Iterator


# ============================================================================
# Regex principal
# ============================================================================

_LINE_RE = re.compile(
    r"\[SIM_STEP=(?P<step>-?\d+)\]"
    r"\[SIM_TIME=(?P<t>[\d.]+)s\]"
    r"(?:\[PKT=(?P<pkt>\d+)\])?"
    r"(?:\[VEH_PKT=(?P<veh_pkt>\d+)\])?"
    r"\[SEQ=(?P<seq>-?\d+)\]"
    r"\[VEH=(?P<veh>[^\]]*)\]"
    r"\[NODE=(?P<node>-?\d+)\]"
    r"\[STATION=(?P<station>-?\d+)\]"
    r"(?:\[PIPE_STEP=(?P<pipe_step>[^\]]+)\])?"
    r"(?:\[PIPE_STAGE=(?P<pipe_stage>[^\]]+)\])?"
    r"\[LVL=(?P<level>[^\]]+)\]"
    r"\[MOD=(?P<module>[^\]]+)\]"
    r"\[EV=(?P<event>[^\]]+)\]"
    r"(?:\[RESULT=(?P<result>[^\]]+)\])?"
    r"(?:\[DURATION_US=(?P<duration>\d+)\])?"
    r"(?:\[PAYLOAD_TYPE=(?P<payload_type>[^\]]+)\])?"
    r"(?:\[PAYLOAD_SIZE=(?P<payload_size>\d+)\])?"
    r"(?P<msg>.*)$"
)

# ============================================================================
# Campos chave=valor presentes na mensagem livre
# ============================================================================

_KV_RE = re.compile(r"(\w+)=(-?[\w.]+)")

_POS_RE = re.compile(
    r"pos[_=]?\(?(?P<x>-?[\d.]+)[,\s]+(?P<y>-?[\d.]+)\)?"
)

_POS_XY_RE = re.compile(
    r"pos_x=(-?[\d.]+)\s+pos_y=(-?[\d.]+)"
)


# ============================================================================
# Entrada estruturada
# ============================================================================

@dataclass(frozen=True, slots=True)
class LogEntry:

    step: int
    t: float

    level: str
    module: str
    event: str

    veh: str
    node: int
    station: int
    seq: int

    pkt: int | None
    veh_pkt: int | None

    pipe_step: str | None
    pipe_stage: str | None

    result: str | None

    duration: int | None

    payload_type: str | None
    payload_size: int | None

    msg: str

    line_no: int
    source_file: str

    # ---------------------------------------------------------------------

    def kv(self) -> dict[str, str]:
        return dict(_KV_RE.findall(self.msg))

    # ---------------------------------------------------------------------

    def pos(self) -> tuple[float, float] | None:

        m = _POS_XY_RE.search(self.msg)
        if m:
            return float(m.group(1)), float(m.group(2))

        m = _POS_RE.search(self.msg)
        if m:
            return (
                float(m.group("x")),
                float(m.group("y"))
            )

        return None

    # ---------------------------------------------------------------------

    def duration_us(self) -> int | None:
        return self.duration

    # ---------------------------------------------------------------------

    def is_(self, module: str, event: str) -> bool:
        return (
            self.module == module
            and
            self.event == event
        )


# ============================================================================
# Estatísticas do parser
# ============================================================================

@dataclass
class ParseStats:

    files: list[str] = field(default_factory=list)

    total_lines: int = 0
    parsed_lines: int = 0
    unparsed_lines: int = 0

    unparsed_samples: list[str] = field(default_factory=list)

    @property
    def unparsed_ratio(self) -> float:

        if self.total_lines == 0:
            return 0.0

        return self.unparsed_lines / self.total_lines


# ============================================================================
# Parsing de uma linha
# ============================================================================

def parse_line(
    raw: str,
    line_no: int,
    source_file: str,
) -> LogEntry | None:

    m = _LINE_RE.match(raw)

    if not m:
        return None

    return LogEntry(

        step=int(m.group("step")),
        t=float(m.group("t")),

        level=m.group("level"),
        module=m.group("module"),
        event=m.group("event"),

        veh=m.group("veh"),

        node=int(m.group("node")),
        station=int(m.group("station")),
        seq=int(m.group("seq")),

        pkt=(
            int(m.group("pkt"))
            if m.group("pkt")
            else None
        ),

        veh_pkt=(
            int(m.group("veh_pkt"))
            if m.group("veh_pkt")
            else None
        ),

        pipe_step=m.group("pipe_step"),
        pipe_stage=m.group("pipe_stage"),

        result=m.group("result"),

        duration=(
            int(m.group("duration"))
            if m.group("duration")
            else None
        ),

        payload_type=m.group("payload_type"),

        payload_size=(
            int(m.group("payload_size"))
            if m.group("payload_size")
            else None
        ),

        msg=m.group("msg"),

        line_no=line_no,
        source_file=source_file,
    )


# ============================================================================
# Leitura dos ficheiros
# ============================================================================

def load_log(
    paths: Iterable[str | Path],
) -> tuple[list[LogEntry], ParseStats]:

    stats = ParseStats()

    entries: list[LogEntry] = []

    for p in paths:

        p = Path(p)

        stats.files.append(str(p))

        if not p.exists():
            print(
                f"ERRO: ficheiro não encontrado: {p}",
                file=sys.stderr,
            )
            sys.exit(1)

        with p.open("r", errors="replace") as f:

            for i, raw in enumerate(f, start=1):

                stats.total_lines += 1

                raw = raw.rstrip("\n")

                if not raw:
                    continue

                entry = parse_line(
                    raw,
                    i,
                    str(p),
                )

                if entry is None:

                    stats.unparsed_lines += 1

                    if len(stats.unparsed_samples) < 5:
                        stats.unparsed_samples.append(
                            raw[:200]
                        )

                    continue

                stats.parsed_lines += 1
                entries.append(entry)

    return entries, stats


# ============================================================================
# Filtro por módulo/evento
# ============================================================================

def iter_by(
    entries: Iterable[LogEntry],
    module: str,
    event: str,
) -> Iterator[LogEntry]:

    for e in entries:

        if e.module == module and e.event == event:
            yield e

=== ./v2x_analyzer/README.md ===
# v2x_analyzer

Analisador modular de logs V2X (ns-3 + SUMO + Vanetza CAM), em Python —
conversão de `analyze_v2x_v8.sh`.

## Porque não é só uma tradução linha-a-linha do bash

O script bash cresceu por correções sucessivas (v5→v8) porque cada secção
do relatório usava o seu próprio padrão `grep`/`awk` sobre texto livre, e
mais que uma vez um padrão mal escrito ou mal ancorado a um módulo contou
o mesmo evento duas vezes, ou nunca bateu certo com o formato real do log
(ver `06_RELATORIO_DE_PROGRESSO.md`, achado N4, e a nota sobre
`ENCODE_EMPTY` em `metrics/tx_pipeline.py`).

Aqui, `parser.py` faz o parsing UMA vez para cada linha, para um
`LogEntry` estruturado (`module`, `event`, `veh`, `node`, `station`,
`seq`, ...). Todos os módulos de métricas comparam campos exatos
(`e.module == "RX" and e.event == "RAW_IN"`), nunca regex sobre a linha
crua — a classe de bug inteira deixa de ser possível.

## Instalar e correr

Só biblioteca padrão do Python 3.10+ — nada para instalar para usar.

```bash
python3 -m v2x_analyzer output_log.txt
python3 -m v2x_analyzer output_log_001 output_log_002 output_log_003   # vários ficheiros, pela ordem dada
python3 -m v2x_analyzer --json output_log.txt                          # para consumo programático
python3 -m v2x_analyzer -o relatorio.txt output_log.txt                 # gravar em ficheiro
```

Código de saída: `0` se a Etapa 4 passa todos os critérios, `1` caso
contrário — para usar em CI/scripts sem ter de fazer parsing do próprio
relatório.

## Estrutura

```
v2x_analyzer/
    parser.py          — LogEntry + parsing (o único sítio que sabe o formato do log)
    identity.py         — mapa vehId <-> nodeId <-> stationId, partilhado
    metrics/
        tx_pipeline.py     — Encode/Sign/Frame/TX Ready/TX Sent
        trigger_reasons.py — motivo de disparo do ShouldSendCAM (achado #10)
        rx_pipeline.py      — Raw In/Verify/Decode/Pipeline OK
        errors.py           — Sender/Receiver Not Found, Self-RX, No Cert
        connectivity.py     — pares ativos, entrega efetiva, fan-out
        vehicles.py          — TX/RX por veículo, ghost check
        timing.py            — estatísticas de DURATION_US (achados #14/#15)
        exit_criteria.py     — critérios de saída da Etapa 4
    report.py            — junta tudo, formata texto e JSON
    cli.py / __main__.py — linha de comandos
    tests/                — ver abaixo
```

## Adicionar uma métrica nova

1. Criar `metrics/nome_novo.py` com uma dataclass de resultado e uma
   função `compute(entries, ...) -> NomeNovoMetrics`.
2. Registar em `report.py`: chamar `compute()` em `build_report()`,
   adicionar o campo a `FullReport`, e uma secção em `render_text()`.
3. Escrever pelo menos um teste em `tests/test_metrics.py` — de
   preferência um caso-armadilha (o que é que partiria isto?), não só um
   caso feliz.

Nenhum módulo de métricas depende de outro (exceto de `identity`,
partilhado), por isso isto nunca obriga a tocar nos módulos existentes.

## Testes

```bash
pip install pytest --break-system-packages   # única dependência, só para testar
python3 -m pytest v2x_analyzer/tests/ -v
```

19 testes: parsing (incluindo os casos-armadilha `VEH=-`, sentinelas
negativas, linhas malformadas), métricas (replicando literalmente os
bugs históricos do bash — RAW_IN a dobrar, DECODE_FAIL a dobrar,
VERIFY_RESULT_FAIL a poluir VERIFY_FAIL — como testes de regressão), e
um teste de integração ponta-a-ponta contra `tests/fixtures/small_run.log`,
cujos números esperados foram calculados à mão, não copiados de uma run
real.

Validado por comparação direta, número a número, contra
`analyze_v2x_v8.sh` na mesma run real (`output_log12.txt`) — todos os
valores batem certo, exceto `ENCODE_EMPTY`, que o bash nunca contava por
um bug de nome de evento (ver `metrics/tx_pipeline.py`).

## Limitações conhecidas / por fazer

- `cam_framing.cc` ganhou eventos `FRAME_BEGIN`/`FRAME_UNPACK_BEGIN`
  numa sessão paralela, sem `DURATION_US` associado (ao contrário de
  encode/sign/verify/decode). `metrics/timing.py` não os inclui ainda —
  decisão pendente: acrescentar duração ali também, ou remover os
  marcadores órfãos.
- `NOT_IMPLEMENTED` em `errors.py` (timestamp expirado, duplicados) —
  o script bash tinha contadores para eventos que não existem no código
  C++ atual. Mantidos como aviso explícito, não como "zero limpo".

"""
report.py — junta todas as métricas e formata o relatório final.

Duas saídas suportadas: texto (para leitura humana, no espírito do
relatório bash original) e JSON (para consumo programático — ex.
comparar duas runs automaticamente, ou alimentar um dashboard; o bash
não oferecia isto sem trabalho extra de parsing do próprio output).
"""

from __future__ import annotations

import json
from dataclasses import asdict, dataclass, is_dataclass
from typing import Any

from .parser import LogEntry, ParseStats
from .identity import IdentityMap, build_identity_map
from .metrics import (
    tx_pipeline,
    trigger_reasons,
    rx_pipeline,
    errors,
    connectivity,
    vehicles,
    timing,
    exit_criteria,
)


@dataclass
class FullReport:
    parse_stats: ParseStats
    identity: IdentityMap
    tx: tx_pipeline.TxPipelineMetrics
    triggers: trigger_reasons.TriggerReasonMetrics
    rx: rx_pipeline.RxPipelineMetrics
    err: errors.ErrorMetrics
    conn: connectivity.ConnectivityMetrics
    veh: vehicles.VehicleMetrics
    time_: timing.TimingMetrics
    exit_: exit_criteria.ExitCriteriaResult


def build_report(entries: list[LogEntry], stats: ParseStats) -> FullReport:
    identity = build_identity_map(entries)
    tx = tx_pipeline.compute(entries)
    triggers = trigger_reasons.compute(entries)
    rx = rx_pipeline.compute(entries)
    err = errors.compute(entries)
    conn = connectivity.compute(entries, identity, tx_sent=tx.tx_sent)
    veh = vehicles.compute(entries, identity)
    time_ = timing.compute(entries)
    exit_ = exit_criteria.compute(conn, tx, rx)

    return FullReport(
        parse_stats=stats,
        identity=identity,
        tx=tx,
        triggers=triggers,
        rx=rx,
        err=err,
        conn=conn,
        veh=veh,
        time_=time_,
        exit_=exit_,
    )

          f"— nome confuso (não é dado do SUMO), não incluído nesta contagem.")
    w("")

    w(hr)
    w(" ERROS & ANOMALIAS")
    w(hr)
    w(f"  {'Sender Not Found':<40}{r.err.sender_not_found:>7}  <- meta: 0")
    w(f"  {'Self-RX Filtrado':<40}{r.err.self_rx_filtered:>7}")
    w(f"  {'Receiver Not Found':<40}{r.err.receiver_not_found:>7}  <- meta: 0")
    w(f"  {'No Cert':<40}{r.err.no_cert:>7}  <- meta: 0")
    for name in errors.NOT_IMPLEMENTED:
        w(f"  {name:<40}{'N/A':>7}  (evento não existe no código atual)")
    w("")

    w(hr)
    w(" CONECTIVIDADE E ENTREGA")
    w(hr)
    w(f"  Veículos observados: {r.conn.n_vehicles}")
    w(f"  Pares ativos: {r.conn.n_active_pairs}/{r.conn.max_pairs}")
    missing = r.conn.missing_pairs()
    if missing:
        w(f"  Pares em falta: {', '.join(f'{a}->{b}' for a, b in missing)}")
    delivery = r.conn.effective_delivery_pct
    w(f"  Entrega efetiva: {delivery:.2f}%" if delivery is not None else "  Entrega efetiva: N/A")
    fanout = r.conn.avg_fanout
    w(f"  Fan-out médio (RX/TX): {fanout:.2f}" if fanout is not None else "  Fan-out médio: N/A")
    w("")

    w(hr)
    w(" POR VEÍCULO")
    w(hr)
    w("  Top transmissores:")
    for veh, n in r.veh.tx_by_vehicle.most_common():
        w(f"    {veh:<10}{n:>7}")
    w("  Top recetores:")
    for veh, n in r.veh.rx_by_vehicle.most_common():
        w(f"    {veh:<10}{n:>7}")
    ghosts = r.veh.ghost_senders()
    if ghosts:
        w(f"  Ghost check (transmitiram, 0 sucessos): {', '.join(ghosts)}")
    else:
        w("  Ghost check: limpo (todos os emissores tiveram >=1 sucesso)")
    w("")

    w(hr)
    w(" DURAÇÃO DE OPERAÇÕES (us)")
    w(hr)
    w(f"  {'Operação':<12}{'n':>7}{'min':>9}{'mediana':>10}{'media':>9}{'p95':>9}{'max':>9}")
    for label, op in r.time_.operations.items():
        if op.n == 0:
            w(f"  {label:<12}{'sem amostras':>7}")
            continue
        w(f"  {label:<12}{op.n:>7}{op.min_us:>9}{op.median_us:>10.1f}"
          f"{op.mean_us:>9.1f}{op.p95_us:>9.1f}{op.max_us:>9}")
    w("")

    w(hr)
    w(" CRITÉRIOS DE SAÍDA — ETAPA 4")
    w(hr)
    for c in r.exit_.criteria:
        status = "PASS" if c.passed else "FAIL"
        w(f"  [{status}] {c.name:<35} {c.detail}")
    w("")
    verdict = "*** ETAPA 4 CONCLUÍDA ***" if r.exit_.all_pass else "XXX ETAPA 4 NÃO CONCLUÍDA XXX"
    w(f"  {verdict}")
    w("")
    w("=" * 72)

    return "\n".join(lines)


def _to_jsonable(obj: Any) -> Any:
    if is_dataclass(obj) and not isinstance(obj, type):
        return {k: _to_jsonable(v) for k, v in asdict(obj).items()}
    if isinstance(obj, set):
        return sorted(list(obj))
    if isinstance(obj, dict):
        return {str(k): _to_jsonable(v) for k, v in obj.items()}
    if isinstance(obj, (list, tuple)):
        return [_to_jsonable(v) for v in obj]
    return obj


def render_json(r: FullReport) -> str:
    data = _to_jsonable(r)
    # As propriedades calculadas (@property) não entram em asdict() —
    # são acrescentadas aqui explicitamente, porque são precisamente o
    # que um consumidor programático mais quer (ex.: comparar duas runs
    # por entrega efetiva sem ter de recalcular a partir dos brutos).
    data["tx"]["encode_fail_rate_pct"] = r.tx.encode_fail_rate
    data["triggers"]["trigger_total"] = r.triggers.trigger_total
    data["triggers"]["reason_total"] = r.triggers.reason_total
    data["triggers"]["sources_agree"] = r.triggers.sources_agree
    data["rx"]["decode_fail_reason_total"] = r.rx.decode_fail_reason_total
    data["rx"]["consistency_check"] = r.rx.consistency_check
    data["conn"]["n_active_pairs"] = r.conn.n_active_pairs
    data["conn"]["theoretical_max_receptions"] = r.conn.theoretical_max_receptions
    data["conn"]["effective_delivery_pct"] = r.conn.effective_delivery_pct
    data["conn"]["avg_fanout"] = r.conn.avg_fanout
    data["conn"]["missing_pairs"] = r.conn.missing_pairs()
    data["veh"]["ghost_senders"] = r.veh.ghost_senders()
    data["exit_criteria_all_pass"] = r.exit_.all_pass
    return json.dumps(data, indent=2, ensure_ascii=False)

def mk(module, event, veh="veh0", node=0, station=1, seq=0, msg="", step=1, t=1.0, level="INFO"):
    return LogEntry(
        step=step, t=t, level=level, module=module, event=event,

def test_raw_in_not_double_counted_across_modules():
    """Bug histórico (v5): RxCallback (MOD=CAM_RX) e RxPipeline (MOD=RX)
    logavam RAW_IN para o MESMO pacote — um grep sem âncora de módulo
    contava o dobro. Aqui, um pacote real = duas linhas, mas só uma
    deve entrar na contagem oficial."""
    entries = [
        mk("CAM_RX", "SUMO_RAW_IN"),  # duplicado histórico, módulo errado
        mk("RX", "RAW_IN"),           # o oficial
    ]
    m = rx_pipeline.compute(entries)

    assert m.sumo_raw_in_cam_rx == 1  # visível, mas não somado ao oficial

    """Bug potencial (o mesmo tipo que já aconteceu com RAW_IN): um evento

    entries = [
        mk("SECURITY", "VERIFY_FAIL"),
        mk("RX", "VERIFY_RESULT_FAIL"),
    ]
    m = rx_pipeline.compute(entries)
    assert m.verify_fail == 1
    assert m.verify_result_fail_pipeline == 1

"""
cli.py — interface de linha de comandos.

Uso:
    python -m v2x_analyzer output_log.txt [output_log2.txt ...]
    python -m v2x_analyzer --json output_log.txt
    python -m v2x_analyzer --compare run_a.txt -- run_b.txt
"""

from __future__ import annotations

import argparse
import sys

from .parser import load_log
from .report import build_report, render_text, render_json


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(
        prog="v2x_analyzer",
        description="Analisador de logs V2X (ns-3 + SUMO + Vanetza CAM).",
    )
    ap.add_argument(
        "logfiles",
        nargs="+",
        help="Um ou mais ficheiros de log, lidos pela ordem dada.",
    )
    ap.add_argument(
        "--json",
        action="store_true",
        help="Escreve o relatório em JSON em vez de texto legível.",
    )
    ap.add_argument(
        "-o", "--output",
        help="Escreve o relatório neste ficheiro em vez de stdout.",
    )
    args = ap.parse_args(argv)

    entries, stats = load_log(args.logfiles)
    if not entries:
        print("AVISO: nenhuma linha reconhecida em nenhum ficheiro dado.", file=sys.stderr)

    report = build_report(entries, stats)
    text = render_json(report) if args.json else render_text(report)

    if args.output:
        with open(args.output, "w") as f:
            f.write(text + "\n")
    else:
        print(text)

    return 0 if report.exit_.all_pass else 1


if __name__ == "__main__":
    sys.exit(main())
